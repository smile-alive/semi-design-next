import ApiType from '#/components/ApiType.astro';

**Props**

| Property | Description | Type | Default |
| --- | --- | --- | --- |
| boundElement | Element that bounds the size: 'parent', 'window' or an HTMLElement. | `'parent' \| 'window' \| HTMLElement` | - |
| boundsByDirection | Whether bounds are computed by drag direction. When true, dragging from a left or top handle limits the size by the distance from the side that stays put to the opposite edge of the bound element. | `boolean` | - |
| children | Content of the resizable box. | `Snippet` | - |
| defaultSize | Initial width and height, as a number or a string (px/vw/vh/%). It is also the fallback of an axis that `size` leaves unset. | `Size` | - |
| enable | Directions that can be resized. A direction not set to false is enabled; false renders no handles. | `Enable \| false` | <ApiType detail={"{ top: true, right: true, bottom: true, left: true, topRight: true, bottomRight: true, bottomLeft: true, topLeft: true }"}>Type</ApiType> |
| grid | Increment that resizing snaps to. | `[number, number]` | `[1, 1]` |
| handle | Snippet for the content of the handle in each enabled direction. It renders once per enabled direction, with `{ direction }` set to that handle's direction. | <ApiType detail={"Snippet\u003c[ResizableHandleSnippetProps]>"}>Type</ApiType> | - |
| handleClass | Class names of the handle in each direction. | `HandleClassName` | - |
| handleStyle | Styles of the handle in each direction, as style strings. | `HandleStyle` | - |
| handleWrapperClass | Class name of the element that wraps all handles. | `string` | - |
| handleWrapperStyle | Style string of the element that wraps all handles. | `string` | - |
| lockAspectRatio | Locks the aspect ratio: true keeps the initial one, a number is the width to height ratio. | `boolean \| number` | `false` |
| lockAspectRatioExtraHeight | Extra height in pixels (such as padding) left out of the ratio when the aspect ratio is locked. | `number` | `0` |
| lockAspectRatioExtraWidth | Extra width in pixels (such as padding) left out of the ratio when the aspect ratio is locked. | `number` | `0` |
| maxHeight | Maximum height. | `string \| number` | - |
| maxWidth | Maximum width. | `string \| number` | - |
| minHeight | Minimum height. | `string \| number` | - |
| minWidth | Minimum width. | `string \| number` | - |
| onChange | Called when a drag or arrow key changes the size, with the new size, the event and the direction. | `ResizeCallback` | - |
| onResizeEnd | Called when resizing ends, with the final size, the event and the direction. | `ResizeCallback` | - |
| onResizeStart | Called when resizing starts. Return false to cancel. | `ResizeStartCallback` | - |
| ratio | Ratio of pointer movement to size change; pass [width, height] for per-axis ratios. Arrow keys always change the size by 10px, regardless of ratio and scale. | `number \| [number, number]` | `1` |
| scale | Scale applied to the element from outside, used to correct the pointer movement. | `number` | `1` |
| size | Controlled size, applied per axis: an axis left unset follows the drag. A one-way prop that cannot be bound with `bind:size`: write the new size back in `onChange`. | `Size` | - |
| snap | Absolute pixel values that resizing snaps to. Both x and y are optional. | `{ x?: number[]; y?: number[] }` | - |
| snapGap | Minimum gap required to move to the next snap target. | `number` | `0` |
