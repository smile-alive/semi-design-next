import ApiType from '#/components/ApiType.astro';

**Props**

| Property | Description | Type | Default |
| --- | --- | --- | --- |
| classNames | Class names of the inner parts, keyed by `expand` (expand or collapse button) and `copy` (copy button; the wrapping span still takes `copyable.className`); merged with the built-in classes. | <ApiType detail={"Partial\u003cRecord\u003cTypographyParagraphPart, string>>"}>Type</ApiType> | - |
| component | Tag name or component the root element renders as. Defaults to `p`. | `React.ElementType` | - |
| copyable | Whether the text can be copied; pass an object to configure it, see Copyable Config. | `CopyableConfig \| boolean` | - |
| delete | Adds a strikethrough. | `boolean` | - |
| disabled | Disables the text. A link renders as a `span` that keeps only its style. | `boolean` | - |
| ellipsis | Truncates overflowing text; pass an object to configure it, see Ellipsis Config. | `Ellipsis \| boolean` | - |
| link | Whether the text is a link. With an object, its attributes (`href`, `target`, `className`, `style`, `onClick` and so on) are applied to the inner `a` element. When disabled, only `className` and `style` are kept. | `LinkType` | - |
| mark | Adds a highlight. | `boolean` | - |
| ref | Reference to the root element. | <ApiType detail={"React.Ref\u003cHTMLParagraphElement>"}>Type</ApiType> | - |
| size | Text size: `normal`, `small` or `inherit`. `inherit` takes the size of the enclosing Typography component. Defaults to `normal`. | `TypographyBaseSize` | - |
| spacing | Line spacing; `extended` is looser. Defaults to `normal`. | `TypographyBaseSpacing` | - |
| strong | Whether the text is bold. | `boolean` | - |
| styles | Inline styles of the inner parts, keyed like `classNames`. | <ApiType detail={"Partial\u003cRecord\u003cTypographyParagraphPart, React.CSSProperties>>"}>Type</ApiType> | - |
| type | Text type: `primary`, `secondary`, `danger`, `warning`, `success`, `tertiary` or `quaternary`. Defaults to `primary`. | `TypographyBaseType` | - |
| underline | Adds an underline. Link text shows it only on hover and active. | `boolean` | - |
