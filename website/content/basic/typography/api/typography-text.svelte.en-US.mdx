import ApiType from '#/components/ApiType.astro';

**Props**

| Property | Description | Type | Default |
| --- | --- | --- | --- |
| code | Whether the text is wrapped in a code element. | `boolean` | - |
| component | Tag name of the element to render. Defaults to span. | `string` | `'span'` |
| copyable | Whether the text can be copied; pass an object for the Copyable Config. | `TypographyCopyableConfig \| boolean` | - |
| copyAction | Snippet that replaces the whole copy node, with the argument `{ copied, copy, config }`; `copy` takes the click event. | <ApiType detail={"Snippet\u003c[TypographyCopyActionProps]>"}>Type</ApiType> | - |
| copyIcon | Snippet for the content of the copy button; a single node with an aria-label names the button, otherwise the locale's "Copy" does. | `Snippet` | - |
| copySuccessTip | Snippet for the content shown after a successful copy. | `Snippet` | - |
| copyTip | Snippet for the tooltip content of the copy button. | `Snippet` | - |
| delete | Adds a strikethrough. | `boolean` | - |
| disabled | Disables the text. | `boolean` | - |
| ellipsis | Truncates overflowing text; pass an object for the Ellipsis Config. | `Ellipsis \| boolean` | - |
| ellipsisContent | Snippet for the content of the floating layer when truncated; defaults to the full text. | `Snippet` | - |
| ellipsisTooltip | Snippet for a custom floating layer, rendered only when the text overflows; its argument is `{ content, trigger }`, where `content` is the full text and `trigger` is a snippet of the text element, to be placed in your own layer with `{@render trigger()}`. | <ApiType detail={"Snippet\u003c[TypographyEllipsisTooltipProps]>"}>Type</ApiType> | - |
| icon | Snippet for the prefix icon; a Semi icon is set to small or default according to size. | `Snippet` | - |
| link | Whether the text is a link; with an object, its attributes (href, target, class, style, onclick and so on) are applied to the a element. | `LinkType` | - |
| mark | Adds a highlight. | `boolean` | - |
| size | Text size; inherit takes the size of the enclosing Text or Paragraph. | `TypographyBaseSize` | - |
| strong | Whether the text is bold. | `boolean` | - |
| type | Text type. | `TypographyBaseType` | - |
| underline | Adds an underline. | `boolean` | - |
| weight | Font weight as a number. | `number` | - |
