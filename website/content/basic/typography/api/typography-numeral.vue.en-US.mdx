import ApiType from '#/components/ApiType.astro';

**Props**

| Property | Description | Type | Default |
| --- | --- | --- | --- |
| code | Whether the text is wrapped in a code element. | `boolean` | `false` |
| parser | Custom number parser. | <ApiType detail={"((value: string) => string)"}>Type</ApiType> | `undefined` |
| precision | Number of decimal places to keep. | `number` | `0` |
| rule | Parsing rule: text only rounds, numbers filters out non-numeric characters, bytes-decimal and bytes-binary convert byte units, percentages converts to percent, exponential converts to scientific notation. | <ApiType detail={"\"text\" \u007c \"numbers\" \u007c \"bytes-decimal\" \u007c \"bytes-binary\" \u007c \"percentages\" \u007c \"exponential\""}>Type</ApiType> | `"text"` |
| size | Text size; inherit takes the size of the enclosing Text or Paragraph. | `"normal" \| "small" \| "inherit"` | `"normal"` |
| truncate | How the kept decimals are rounded, matching Math.round, Math.ceil and Math.floor. | `"round" \| "ceil" \| "floor"` | `"round"` |
| component | Element to render: a tag name or a Vue component. Defaults to span; use div when the content has block elements such as p. | `string \| Component` | `undefined` |
| copyable | Whether the text can be copied; pass an object for the Copyable Config. | `boolean \| TypographyCopyableConfig` | `false` |
| delete | Adds a strikethrough. | `boolean` | `false` |
| disabled | Disables the text. | `boolean` | `false` |
| link | Whether the text is a link; with an object, its attributes (href, target, class, style, onClick and so on) are applied to the a element. | `LinkType` | `false` |
| mark | Adds a highlight. | `boolean` | `false` |
| strong | Whether the text is bold. | `boolean` | `false` |
| type | Text type. | <ApiType detail={"\"primary\" \u007c \"secondary\" \u007c \"danger\" \u007c \"warning\" \u007c \"success\" \u007c \"tertiary\" \u007c \"quaternary\""}>Type</ApiType> | `"primary"` |
| underline | Adds an underline. | `boolean` | `false` |

**Events**

| Event | Description                                                                                                                                                                  | Callback parameters                              |
| ----- | ---------------------------------------------------------------------------------------------------------------------------------------------------------------------------- | ------------------------------------------------ |
| copy  | Emitted after copying, with the click event, the copied content and whether the write succeeded. When the write is rejected it is false and the copied state is not entered. | `[e: MouseEvent, content: string, res: boolean]` |

**Slots**

| Slot           | Description                                                                                                        | Scope parameters            |
| -------------- | ------------------------------------------------------------------------------------------------------------------ | --------------------------- |
| default        | Content to format; text and elements are processed recursively, text inside components is not.                     | `any`                       |
| icon           | Prefix icon; a Semi icon is set to small or default according to size.                                             | `any`                       |
| copyIcon       | Content of the copy button; a single node with an aria-label names the button, otherwise the locale's "Copy" does. | `any`                       |
| copyTip        | Tooltip content of the copy button.                                                                                | `any`                       |
| copySuccessTip | Content shown after a successful copy.                                                                             | `any`                       |
| copyAction     | Replaces the whole copy node; receives copied, copy and the resolved config.                                       | `TypographyCopyActionProps` |
