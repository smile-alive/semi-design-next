import ApiType from '#/components/ApiType.astro';

**Props**

| 属性      | 说明                                                                                                                                                 | 类型                                                         | 默认值 |
| --------- | ---------------------------------------------------------------------------------------------------------------------------------------------------- | ------------------------------------------------------------ | ------ |
| code      | 是否被 code 元素包裹。                                                                                                                               | `boolean`                                                    | -      |
| component | 自定义渲染元素，可以是标签名或组件，默认 span；内容含 p 等块级元素时使用 div。                                                                       | `ValidComponent`                                             | -      |
| copyable  | 是否可复制；传对象时见 Copyable Config。                                                                                                             | `CopyableConfig \| boolean`                                  | -      |
| delete    | 添加删除线样式。                                                                                                                                     | `boolean`                                                    | -      |
| disabled  | 禁用文本。                                                                                                                                           | `boolean`                                                    | -      |
| icon      | 前缀图标，可传元素或图标组件；传图标组件时按文本大小取尺寸。                                                                                         | `TypographyIcon`                                             | -      |
| link      | 是否为链接；传对象时，其中的属性（href、target、class、style、onClick 等）挂在 a 元素上。                                                            | `LinkType`                                                   | -      |
| mark      | 添加标记样式。                                                                                                                                       | `boolean`                                                    | -      |
| parser    | 自定义数值解析函数。                                                                                                                                 | <ApiType detail={"(value: string) => string"}>Type</ApiType> | -      |
| precision | 小数点后保留的位数，默认 0。                                                                                                                         | `number`                                                     | -      |
| rule      | 解析规则：text 仅取整，numbers 过滤非数字字符，bytes-decimal、bytes-binary 换算字节单位，percentages 转百分比，exponential 转科学计数法，默认 text。 | `TypographyBaseRule`                                         | -      |
| size      | 文本大小，inherit 继承外层 Text 或 Paragraph 的尺寸，默认 normal。                                                                                   | `TypographyBaseSize`                                         | -      |
| strong    | 是否加粗。                                                                                                                                           | `boolean`                                                    | -      |
| truncate  | 小数点后保留位的取整方式，对应 Math.round、Math.ceil、Math.floor，默认 round。                                                                       | `TypographyBaseTruncate`                                     | -      |
| type      | 文本类型，默认 primary。                                                                                                                             | `TypographyBaseType`                                         | -      |
| underline | 添加下划线样式。                                                                                                                                     | `boolean`                                                    | -      |
