import ApiType from '#/components/ApiType.astro';

**Props**

| 属性      | 说明                                                                                                                                                                                                           | 类型                                                               | 默认值    |
| --------- | -------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------- | ------------------------------------------------------------------ | --------- |
| children  | 需要格式化的内容，递归遍历其中的文本，包括子元素内的数字。                                                                                                                                                     | `React.ReactNode`                                                  | -         |
| code      | 是否被 `code` 元素包裹。                                                                                                                                                                                       | `boolean`                                                          | -         |
| component | 根元素渲染的标签名或组件，默认 `span`；内容含 `p` 等块级元素时使用 `div`。                                                                                                                                     | `React.ElementType`                                                | -         |
| copyable  | 是否可复制；传对象时可配置 Copyable Config。                                                                                                                                                                   | `CopyableConfig \| boolean`                                        | -         |
| delete    | 添加删除线样式。                                                                                                                                                                                               | `boolean`                                                          | -         |
| disabled  | 禁用文本；链接渲染为 `span`，只保留样式。                                                                                                                                                                      | `boolean`                                                          | -         |
| icon      | 前缀图标，可传节点；Semi 图标会按文字大小设置尺寸。                                                                                                                                                            | `React.ReactNode \| string`                                        | -         |
| link      | 是否为链接；传对象时，其中的属性（`href`、`target`、`className`、`style`、`onClick` 等）挂在内层 `a` 元素上。禁用时只保留 `className` 与 `style`。                                                             | `LinkType`                                                         | -         |
| mark      | 添加标记样式。                                                                                                                                                                                                 | `boolean`                                                          | -         |
| parser    | 自定义数值解析函数，接收文本并返回要显示的字符串；设置后 `rule` 不再生效。                                                                                                                                     | <ApiType detail={"(value: string) => string"}>Type</ApiType>       | -         |
| precision | 小数点后保留的位数。                                                                                                                                                                                           | `number`                                                           | `0`       |
| ref       | 根元素的引用。                                                                                                                                                                                                 | <ApiType detail={"React.Ref\u003cHTMLSpanElement>"}>Type</ApiType> | -         |
| rule      | 解析规则：`text` 仅取整，`numbers` 过滤非数字字符，`bytes-decimal` 与 `bytes-binary` 换算字节单位（前者 1 KB 为 1000 字节，后者 1 KiB 为 1024 字节），`percentages` 转为百分比，`exponential` 转为科学计数法。 | `TypographyBaseRule`                                               | `"text"`  |
| size      | 文本大小，可选 `normal`、`small` 或 `inherit`；`inherit` 继承外层 Typography 组件的尺寸，默认 `normal`。                                                                                                       | `TypographyBaseSize`                                               | -         |
| strong    | 是否加粗。                                                                                                                                                                                                     | `boolean`                                                          | -         |
| truncate  | 小数点后保留位的取整方式，对应 `Math.ceil`、`Math.floor` 与 `Math.round`。                                                                                                                                     | `TypographyBaseTruncate`                                           | `"round"` |
| type      | 文本类型，可选 `primary`、`secondary`、`danger`、`warning`、`success`、`tertiary` 或 `quaternary`，默认 `primary`。                                                                                            | `TypographyBaseType`                                               | -         |
| underline | 添加下划线样式；链接文本只在 hover 与 active 时显示下划线。                                                                                                                                                    | `boolean`                                                          | -         |
