import ApiType from '#/components/ApiType.astro';

**Props**

| 属性 | 说明 | 类型 | 默认值 |
| --- | --- | --- | --- |
| component | 自定义渲染元素的标签名，默认由 heading 决定（h1 到 h6）。 | `string` | - |
| copyable | 是否可复制；传对象时见 Copyable Config。 | `TypographyCopyableConfig \| boolean` | - |
| copyAction | 完全自定义复制节点的 snippet，参数为 `{ copied, copy, config }`，`copy` 接收点击事件。 | <ApiType detail={"Snippet\u003c[TypographyCopyActionProps]>"}>Type</ApiType> | - |
| copyIcon | 复制按钮内容的 snippet；内容为单个节点且带 aria-label 时以它作为按钮的可访问名称，否则用语言包的「复制」。 | `Snippet` | - |
| copySuccessTip | 复制成功后展示内容的 snippet。 | `Snippet` | - |
| copyTip | 复制图标提示内容的 snippet。 | `Snippet` | - |
| delete | 添加删除线样式。 | `boolean` | - |
| disabled | 禁用文本。 | `boolean` | - |
| ellipsis | 设置自动溢出省略；传对象时见 Ellipsis Config。 | `Ellipsis \| boolean` | - |
| ellipsisContent | 截断时浮层内容的 snippet，默认是全文。 | `Snippet` | - |
| ellipsisTooltip | 自定义浮层的 snippet，仅在文本溢出时渲染；参数 `{ content, trigger }`，`content` 是全文，`trigger` 是文字元素的 snippet，用 `{@render trigger()}` 放入自己的浮层。 | <ApiType detail={"Snippet\u003c[TypographyEllipsisTooltipProps]>"}>Type</ApiType> | - |
| heading | 标题级别，对应 h1 到 h6。 | `1 \| 2 \| 3 \| 4 \| 5 \| 6` | `1` |
| link | 是否为链接；传对象时，其中的属性（href、target、class、style、onclick 等）挂在 a 元素上。 | `LinkType` | - |
| mark | 添加标记样式。 | `boolean` | - |
| strong | 是否加粗。 | `boolean` | - |
| type | 文本类型。 | `TypographyBaseType` | - |
| underline | 添加下划线样式。 | `boolean` | - |
| weight | 字重，可选 light、regular、medium、semibold、bold、default，或数字。 | <ApiType detail={"'light' \u007c 'regular' \u007c 'medium' \u007c 'semibold' \u007c 'bold' \u007c 'default' \u007c number"}>Type</ApiType> | - |
