import ApiType from '#/components/ApiType.astro';

**Props**

| Property | Description | Type | Default |
| --- | --- | --- | --- |
| children | Content to format. Numbers are formatted in text at any depth, including inside child elements. | `React.ReactNode` | - |
| code | Whether the text is wrapped in a `code` element. | `boolean` | - |
| component | Tag name or component the root element renders as. Defaults to `span`; use `div` when the content has block elements such as `p`. | `React.ElementType` | - |
| copyable | Whether the text can be copied; pass an object to configure it, see Copyable Config. | `CopyableConfig \| boolean` | - |
| delete | Adds a strikethrough. | `boolean` | - |
| disabled | Disables the text. A link renders as a `span` that keeps only its style. | `boolean` | - |
| icon | Prefix icon, a node. A Semi icon is sized to match the text. | `React.ReactNode \| string` | - |
| link | Whether the text is a link. With an object, its attributes (`href`, `target`, `className`, `style`, `onClick` and so on) are applied to the inner `a` element. When disabled, only `className` and `style` are kept. | `LinkType` | - |
| mark | Adds a highlight. | `boolean` | - |
| parser | Custom number parser that takes the text and returns the string to show; `rule` no longer applies once it is set. | <ApiType detail={"(value: string) => string"}>Type</ApiType> | - |
| precision | Number of decimal places to keep. | `number` | `0` |
| ref | Reference to the root element. | <ApiType detail={"React.Ref\u003cHTMLSpanElement>"}>Type</ApiType> | - |
| rule | Parsing rule: `text` only rounds, `numbers` filters out non-numeric characters, `bytes-decimal` and `bytes-binary` convert byte units (1 KB is 1000 bytes, 1 KiB is 1024 bytes), `percentages` converts to percent, `exponential` converts to scientific notation. | `TypographyBaseRule` | `"text"` |
| size | Text size: `normal`, `small` or `inherit`. `inherit` takes the size of the enclosing Typography component. Defaults to `normal`. | `TypographyBaseSize` | - |
| strong | Whether the text is bold. | `boolean` | - |
| truncate | How the kept decimals are rounded, matching `Math.ceil`, `Math.floor` and `Math.round`. | `TypographyBaseTruncate` | `"round"` |
| type | Text type: `primary`, `secondary`, `danger`, `warning`, `success`, `tertiary` or `quaternary`. Defaults to `primary`. | `TypographyBaseType` | - |
| underline | Adds an underline. Link text shows it only on hover and active. | `boolean` | - |
