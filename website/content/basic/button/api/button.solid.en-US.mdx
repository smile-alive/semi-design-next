import ApiType from '#/components/ApiType.astro';

**Props**

| Property            | Description                                                                                                                            | Type                                                                                      | Default    |
| ------------------- | -------------------------------------------------------------------------------------------------------------------------------------- | ----------------------------------------------------------------------------------------- | ---------- |
| block               | Renders the button as a block-level button.                                                                                            | `boolean`                                                                                 | `false`    |
| colorful            | Colorful button, supported for the primary and tertiary types only. Falls back to the ButtonGroup it is in, and finally to false.      | `boolean`                                                                                 | -          |
| contentClassName    | Class name of the content area.                                                                                                        | `string`                                                                                  | -          |
| disabled            | Disabled state, which takes precedence over loading. Falls back to the ButtonGroup it is in, and finally to false.                     | `boolean`                                                                                 | -          |
| htmlType            | Native type of the button: button, reset or submit.                                                                                    | `HtmlType`                                                                                | `"button"` |
| icon                | Button icon, an element or an icon component; only a component (such as `icon={IconAIFilledLevel1}`) receives the colorful fill.       | <ApiType detail={"JSX.Element \u007c Component\u003c{ fill?: IconFill }>"}>Type</ApiType> | -          |
| iconPosition        | Icon position: left or right.                                                                                                          | `'left' \| 'right'`                                                                       | `"left"`   |
| loading             | Loading state. A loading button stays focusable, its onClick is ignored, and a submit button does not submit its form.                 | `boolean`                                                                                 | `false`    |
| noHorizontalPadding | Removes horizontal padding: true, left, right or an array of both. Applies only to a button that shows an icon or the loading spinner. | `boolean \| HorizontalPaddingType \| HorizontalPaddingType[]`                             | `false`    |
| size                | Button size: large, default or small. Falls back to the ButtonGroup it is in, and finally to default.                                  | `Size`                                                                                    | -          |
| theme               | Button theme: solid, borderless, light or outline. Falls back to the ButtonGroup it is in, and finally to light.                       | `Theme`                                                                                   | -          |
| type                | Button type: primary, secondary, tertiary, warning or danger. Falls back to the ButtonGroup it is in, and finally to primary.          | `Type`                                                                                    | -          |
