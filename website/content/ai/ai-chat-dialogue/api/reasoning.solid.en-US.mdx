import ApiType from '#/components/ApiType.astro';

**Props**

| Property            | Description                                                                                                                              | Type                                                                            | Default |
| ------------------- | ---------------------------------------------------------------------------------------------------------------------------------------- | ------------------------------------------------------------------------------- | ------- |
| content             | Reasoning content. When summary is empty, its texts joined by line breaks form the reasoning text.                                       | `{ text?: string; type?: string }[]`                                            | -       |
| customRenderer      | Replaces the rendering of the reasoning text. It receives all props of the component, and its result goes into the content area.         | <ApiType detail={"(props: ReasoningWidgetProps) => JSX.Element"}>Type</ApiType> | -       |
| markdownRenderProps | Props passed to the MarkdownRender of the reasoning text; raw comes from the reasoning text.                                             | `MarkdownRenderProps`                                                           | -       |
| status              | Reasoning status. completed shows the finished title and starts collapsed; any other value shows the thinking title and starts expanded. | `string`                                                                        | -       |
| summary             | Reasoning summary. When it is not empty, its texts joined by line breaks form the reasoning text.                                        | `{ text?: string; type?: string }[]`                                            | -       |
