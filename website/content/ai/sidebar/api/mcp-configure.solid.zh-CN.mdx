import ApiType from '#/components/ApiType.astro';

**Props**

| 属性               | 说明                                                                                                              | 类型                                                                                       | 默认值 |
| ------------------ | ----------------------------------------------------------------------------------------------------------------- | ------------------------------------------------------------------------------------------ | ------ |
| afterVisibleChange | 可见状态变化完成后的回调，参数为当前是否可见；有动画时在动画结束后触发。                                          | <ApiType detail={"(isVisible: boolean) => void"}>Type</ApiType>                            | -      |
| children           | 不生效，内容由 MCP 工具列表渲染。                                                                                 | `JSX.Element`                                                                              | -      |
| closeOnEsc         | 按下 Esc 键时是否触发 onCancel。                                                                                  | `boolean`                                                                                  | -      |
| containerRef       | 接收 `.semi-sidebar-container` 元素，作用同 `ref`。                                                               | <ApiType detail={"Ref\u003cHTMLDivElement>"}>Type</ApiType>                                | -      |
| customOptions      | 自定义 MCP 工具列表，在自定义页签中展示。                                                                         | `MCPSolidOption[]`                                                                         | -      |
| defaultSize        | 初始尺寸（宽度与高度），仅在 resizable 为 true 时生效。                                                           | `{ width?: number \| string; height?: number \| string }`                                  | -      |
| filter             | 自定义搜索过滤，参数为转为小写的输入值和选项；默认按 label 与 desc 匹配。                                         | <ApiType detail={"(inputValue: string, option: MCPSolidOption) => boolean"}>Type</ApiType> | -      |
| maxWidth           | 可调整的最大宽度，仅在 resizable 为 true 时生效。                                                                 | `string \| number`                                                                         | -      |
| minWidth           | 可调整的最小宽度，仅在 resizable 为 true 时生效，默认 150。                                                       | `string \| number`                                                                         | -      |
| motion             | 显示与隐藏时是否使用展开动画，默认 true。                                                                         | `boolean`                                                                                  | -      |
| onAddClick         | 点击自定义页签中的新增按钮时触发。                                                                                | <ApiType detail={"(e: MouseEvent) => void"}>Type</ApiType>                                 | -      |
| onCancel           | 点击关闭按钮，或在 closeOnEsc 为 true 时按下 Esc 触发。                                                           | <ApiType detail={"(e: MouseEvent \u007c KeyboardEvent) => void"}>Type</ApiType>            | -      |
| onConfigureClick   | 点击内置 MCP 工具的配置按钮时触发，参数为事件和该选项。                                                           | <ApiType detail={"(e: MouseEvent, option: MCPSolidOption) => void"}>Type</ApiType>         | -      |
| onEditClick        | 点击自定义 MCP 工具的编辑按钮时触发，参数为事件和该选项。                                                         | <ApiType detail={"(e: MouseEvent, option: MCPSolidOption) => void"}>Type</ApiType>         | -      |
| onSearch           | 搜索时触发，参数为输入值和是否在自定义列表中搜索。                                                                | <ApiType detail={"(inputValue: string, custom: boolean) => void"}>Type</ApiType>           | -      |
| onStatusChange     | 切换启用或关闭时触发，参数为切换后的完整选项列表和是否为自定义列表；需自行把新列表写回 options 或 customOptions。 | <ApiType detail={"(options: MCPSolidOption[], custom: boolean) => void"}>Type</ApiType>    | -      |
| options            | 内置 MCP 工具列表；每项的 value 必填且唯一。                                                                      | `MCPSolidOption[]`                                                                         | -      |
| placeholder        | 搜索框的占位文字，默认使用语言包文案。                                                                            | `string`                                                                                   | -      |
| renderDesc         | 渲染工具项的描述，参数为 `{ option, custom }`；不传时显示选项的 desc 文本。                                       | <ApiType detail={"(props: MCPRenderItemProps) => JSX.Element"}>Type</ApiType>              | -      |
| renderHeader       | 自定义整个头部；返回内容为空时使用内置头部。                                                                      | <ApiType detail={"() => JSX.Element"}>Type</ApiType>                                       | -      |
| renderIcon         | 渲染工具项的图标，参数为 `{ option, custom }`；不传时选项的 icon 作为图片地址渲染。                               | <ApiType detail={"(props: MCPRenderItemProps) => JSX.Element"}>Type</ApiType>              | -      |
| renderItem         | 自定义整行工具项，参数为 `{ option, custom }`。                                                                   | <ApiType detail={"(props: MCPRenderItemProps) => JSX.Element"}>Type</ApiType>              | -      |
| resizable          | 宽度是否可拖拽调整，默认 true；为 true 时 minWidth、maxWidth 与 defaultSize 生效。                                | `boolean`                                                                                  | -      |
| showClose          | 是否显示头部的关闭按钮，默认 true。                                                                               | `boolean`                                                                                  | -      |
| style              | 容器元素的内联样式，可传样式对象或样式字符串。                                                                    | `JSX.CSSProperties \| string`                                                              | -      |
| title              | 头部标题，默认使用语言包中的 MCP 配置。                                                                           | `JSX.Element`                                                                              | -      |
| visible            | 是否显示。                                                                                                        | `boolean`                                                                                  | -      |
