import ApiType from '#/components/ApiType.astro';

**Props**

| Property           | Description                                                                                                                                                             | Type                                                                                       | Default |
| ------------------ | ----------------------------------------------------------------------------------------------------------------------------------------------------------------------- | ------------------------------------------------------------------------------------------ | ------- |
| afterVisibleChange | Called after a visibility change has completed, with whether the container is visible; with motion it runs when the animation ends.                                     | <ApiType detail={"(isVisible: boolean) => void"}>Type</ApiType>                            | -       |
| children           | Has no effect; the content is the MCP tool list.                                                                                                                        | `JSX.Element`                                                                              | -       |
| closeOnEsc         | Whether pressing Esc calls onCancel.                                                                                                                                    | `boolean`                                                                                  | -       |
| containerRef       | Receives the `.semi-sidebar-container` element, as `ref` does.                                                                                                          | <ApiType detail={"Ref\u003cHTMLDivElement>"}>Type</ApiType>                                | -       |
| customOptions      | Custom MCP tools, listed on the custom tab.                                                                                                                             | `MCPSolidOption[]`                                                                         | -       |
| defaultSize        | Initial size (width and height); applies only when resizable is true.                                                                                                   | `{ width?: number \| string; height?: number \| string }`                                  | -       |
| filter             | Custom search filter, called with the lower-cased input and an option; the label and desc are matched by default.                                                       | <ApiType detail={"(inputValue: string, option: MCPSolidOption) => boolean"}>Type</ApiType> | -       |
| maxWidth           | Maximum width when resizing; applies only when resizable is true.                                                                                                       | `string \| number`                                                                         | -       |
| minWidth           | Minimum width when resizing; applies only when resizable is true, 150 by default.                                                                                       | `string \| number`                                                                         | -       |
| motion             | Whether showing and hiding use the expand animation, true by default.                                                                                                   | `boolean`                                                                                  | -       |
| onAddClick         | Called when the add button on the custom tab is clicked.                                                                                                                | <ApiType detail={"(e: MouseEvent) => void"}>Type</ApiType>                                 | -       |
| onCancel           | Called when the close button is clicked, or when Esc is pressed while closeOnEsc is true.                                                                               | <ApiType detail={"(e: MouseEvent \u007c KeyboardEvent) => void"}>Type</ApiType>            | -       |
| onConfigureClick   | Called when the configure button of a built-in tool is clicked, with the event and the option.                                                                          | <ApiType detail={"(e: MouseEvent, option: MCPSolidOption) => void"}>Type</ApiType>         | -       |
| onEditClick        | Called when the edit button of a custom tool is clicked, with the event and the option.                                                                                 | <ApiType detail={"(e: MouseEvent, option: MCPSolidOption) => void"}>Type</ApiType>         | -       |
| onSearch           | Called on search, with the input value and whether the custom list is searched.                                                                                         | <ApiType detail={"(inputValue: string, custom: boolean) => void"}>Type</ApiType>           | -       |
| onStatusChange     | Called when a tool is enabled or disabled, with the whole updated list and whether it is the custom list; write the new list back to options or customOptions yourself. | <ApiType detail={"(options: MCPSolidOption[], custom: boolean) => void"}>Type</ApiType>    | -       |
| options            | Built-in MCP tools; each item needs a unique value.                                                                                                                     | `MCPSolidOption[]`                                                                         | -       |
| placeholder        | Placeholder of the search input, the locale text by default.                                                                                                            | `string`                                                                                   | -       |
| renderDesc         | Renders the description of a tool row, called with `{ option, custom }`; without it the option's desc text is shown.                                                    | <ApiType detail={"(props: MCPRenderItemProps) => JSX.Element"}>Type</ApiType>              | -       |
| renderHeader       | Renders the whole header; the built-in header is used when it returns no content.                                                                                       | <ApiType detail={"() => JSX.Element"}>Type</ApiType>                                       | -       |
| renderIcon         | Renders the icon of a tool row, called with `{ option, custom }`; without it the option's icon is rendered as an image URL.                                             | <ApiType detail={"(props: MCPRenderItemProps) => JSX.Element"}>Type</ApiType>              | -       |
| renderItem         | Renders a whole tool row, called with `{ option, custom }`.                                                                                                             | <ApiType detail={"(props: MCPRenderItemProps) => JSX.Element"}>Type</ApiType>              | -       |
| resizable          | Whether the width can be resized by dragging, true by default; minWidth, maxWidth and defaultSize apply only when true.                                                 | `boolean`                                                                                  | -       |
| showClose          | Whether the header shows the close button, true by default.                                                                                                             | `boolean`                                                                                  | -       |
| style              | Inline style of the container element, a style object or a string.                                                                                                      | `JSX.CSSProperties \| string`                                                              | -       |
| title              | Title in the header, the locale's MCP configure text by default.                                                                                                        | `JSX.Element`                                                                              | -       |
| visible            | Whether the container is shown.                                                                                                                                         | `boolean`                                                                                  | -       |
