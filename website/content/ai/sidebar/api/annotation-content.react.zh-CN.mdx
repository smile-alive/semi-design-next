import ApiType from '#/components/ApiType.astro';

**Props**

| 属性       | 说明                                                    | 类型                                                                                   | 默认值 |
| ---------- | ------------------------------------------------------- | -------------------------------------------------------------------------------------- | ------ |
| activeKey  | 当前展开的分组 key，支持单个或多个。                    | `string \| string[]`                                                                   | -      |
| className  | Collapse 根元素的类名。                                 | `string`                                                                               | -      |
| info       | 参考来源分组列表，每组包含 header、key 与 annotations。 | `AnnotationInfo[]`                                                                     | `[]`   |
| onChange   | 展开的分组变化时触发，参数为当前展开的 key。            | <ApiType detail={"(key: string \u007c string[]) => void"}>Type</ApiType>               | -      |
| onClick    | 点击某条参考来源时触发，参数为事件和该条目。            | <ApiType detail={"(e: React.MouseEvent, item: AnnotationItem) => void"}>Type</ApiType> | -      |
| renderItem | 自定义参考来源条目的渲染，参数为该条目。                | <ApiType detail={"(annotation: AnnotationItem) => ReactNode"}>Type</ApiType>           | -      |
| style      | Collapse 根元素的内联样式。                             | `React.CSSProperties`                                                                  | -      |
