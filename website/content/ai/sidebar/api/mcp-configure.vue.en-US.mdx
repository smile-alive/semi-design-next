import ApiType from '#/components/ApiType.astro';

**Props**

| Property | Description | Type | Default |
| --- | --- | --- | --- |
| closeOnEsc | Whether pressing Esc emits cancel. | `boolean` | `semiGlobalDefault("MCPConfigure", "closeOnEsc")` |
| customOptions | Custom MCP tools; bind it with v-model:customOptions. | `MCPConfigureOption[]` | `semiGlobalDefault("MCPConfigure", "customOptions")` |
| defaultSize | Initial width and height; applies only when resizable is true. | <ApiType detail={"{ width?: string \u007c number \u007c undefined; height?: string \u007c number \u007c undefined; }"}>Type</ApiType> | `semiGlobalDefault("MCPConfigure", "defaultSize")` |
| filter | Custom search filter, called with the lower-cased input and an option; the label and desc are matched by default. | <ApiType detail={"((inputValue: string, option: MCPConfigureOption) => boolean)"}>Type</ApiType> | `undefined` |
| maxWidth | Maximum width when resizing; applies only when resizable is true. | `string \| number` | `semiGlobalDefault("MCPConfigure", "maxWidth")` |
| minWidth | Minimum width when resizing; applies only when resizable is true. | `string \| number` | `semiGlobalDefault("MCPConfigure", "minWidth")` |
| motion | Whether showing and hiding use the expand animation. | `boolean` | `semiGlobalDefault("MCPConfigure", "motion")` |
| options | Built-in MCP tools; bind it with v-model:options. Each item needs a unique value. | `MCPConfigureOption[]` | `semiGlobalDefault("MCPConfigure", "options")` |
| placeholder | Placeholder of the search input, the locale text by default. | `string` | `semiGlobalDefault("MCPConfigure", "placeholder")` |
| resizable | Whether the width can be resized by dragging; minWidth, maxWidth and defaultSize apply only when true. | `boolean` | `semiGlobalDefault("MCPConfigure", "resizable")` |
| showClose | Whether the header shows the close button. | `boolean` | `semiGlobalDefault("MCPConfigure", "showClose")` |
| title | Plain-text title, the locale's MCP configure text by default; the title slot takes precedence. | `string` | `semiGlobalDefault("MCPConfigure", "title")` |
| visible | Whether it is shown; bind it with v-model:visible. | `boolean` | `semiGlobalDefault("MCPConfigure", "visible")` |

**Events**

| Event                | Description                                                                                                  | Callback parameters                                |
| -------------------- | ------------------------------------------------------------------------------------------------------------ | -------------------------------------------------- |
| addClick             | Emitted when the add button of the custom page is clicked.                                                   | `[event: MouseEvent]`                              |
| afterVisibleChange   | Emitted after the show or hide animation ends, with whether it is visible.                                   | `[visible: boolean]`                               |
| cancel               | Emitted when the close button is clicked, or when Esc is pressed while closeOnEsc is true.                   | `[event: KeyboardEvent \| MouseEvent]`             |
| configureClick       | Emitted when the configure button is clicked, with the event and its option.                                 | `[event: MouseEvent, option: MCPConfigureOption]`  |
| editClick            | Emitted when the edit button of a custom tool is clicked, with the event and its option.                     | `[event: MouseEvent, option: MCPConfigureOption]`  |
| search               | Emitted when the search input changes, with the input value and whether the custom list is shown.            | `[inputValue: string, custom: boolean]`            |
| statusChange         | Emitted after a tool is switched, with the list whose active flag changed and whether it is the custom list. | `[options: MCPConfigureOption[], custom: boolean]` |
| update:customOptions | Emitted before statusChange when a custom tool is switched, so v-model:customOptions follows.                | `[options: MCPConfigureOption[]]`                  |
| update:options       | Emitted before statusChange when a built-in tool is switched, so v-model:options follows.                    | `[options: MCPConfigureOption[]]`                  |
| update:visible       | Emitted before cancel when the close button or Esc cancels, with false, so v-model:visible closes it.        | `[visible: false]`                                 |

**Slots**

| Slot   | Description                                                                                                                                       | Scope parameters            |
| ------ | ------------------------------------------------------------------------------------------------------------------------------------------------- | --------------------------- |
| desc   | Description of an option, with item, index and custom; the desc field is shown when absent.                                                       | `MCPConfigureItemSlotProps` |
| header | Replaces the whole header; empty content falls back to the built-in header.                                                                       | `any`                       |
| icon   | Icon of an option, with item, index and custom. When provided it is used for every option; the icon field is an image URL only when it is absent. | `MCPConfigureItemSlotProps` |
| item   | Replaces the whole built-in row, with item, index and custom.                                                                                     | `MCPConfigureItemSlotProps` |
| title  | Header title; it takes precedence over the title prop.                                                                                            | `any`                       |

**Exposes**

| Name | Description                                                  | Type                     |
| ---- | ------------------------------------------------------------ | ------------------------ |
| root | The .semi-sidebar-container root element; null while hidden. | `HTMLDivElement \| null` |
