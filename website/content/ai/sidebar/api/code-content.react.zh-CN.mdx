import ApiType from '#/components/ApiType.astro';

**Props**

| 属性      | 说明                                                                                      | 类型                                                                                                | 默认值 |
| --------- | ----------------------------------------------------------------------------------------- | --------------------------------------------------------------------------------------------------- | ------ |
| activeKey | 当前展开的代码面板 key，支持单个或多个。                                                  | `string \| string[]`                                                                                | -      |
| className | Collapse 根元素的类名。                                                                   | `string`                                                                                            | -      |
| codes     | 代码条目列表，每项为 CodeItem 的属性加用于标识面板的 key。                                | `CodeItemProps[]`                                                                                   | `[]`   |
| onChange  | 展开的面板变化时触发，参数为当前展开的 key。                                              | <ApiType detail={"(activeKey: string \u007c string[]) => void"}>Type</ApiType>                      | -      |
| onExpand  | 点击面板头部的展开按钮时触发，参数为事件、该代码条目和模式 `code`；设置后才显示展开按钮。 | <ApiType detail={"(e: React.MouseEvent, code: CodeItemProps, mode: string) => void"}>Type</ApiType> | -      |
| style     | Collapse 根元素的内联样式。                                                               | `React.CSSProperties`                                                                               | -      |
