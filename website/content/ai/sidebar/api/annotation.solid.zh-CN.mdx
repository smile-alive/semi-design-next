import ApiType from '#/components/ApiType.astro';

**Props**

| 属性               | 说明                                                                                                                                     | 类型                                                                             | 默认值 |
| ------------------ | ---------------------------------------------------------------------------------------------------------------------------------------- | -------------------------------------------------------------------------------- | ------ |
| activeKey          | 当前展开的分组 key，支持单个或多个。                                                                                                     | `string \| string[]`                                                             | -      |
| afterVisibleChange | 可见状态变化完成后的回调，参数为当前是否可见；有动画时在动画结束后触发。                                                                 | <ApiType detail={"(isVisible: boolean) => void"}>Type</ApiType>                  | -      |
| children           | 不生效，内容由参考来源列表渲染。                                                                                                         | `JSX.Element`                                                                    | -      |
| closeOnEsc         | 按下 Esc 键时是否触发 onCancel。                                                                                                         | `boolean`                                                                        | -      |
| containerRef       | 接收 `.semi-sidebar-container` 元素，作用同 `ref`。                                                                                      | <ApiType detail={"Ref\u003cHTMLDivElement>"}>Type</ApiType>                      | -      |
| defaultSize        | 初始尺寸（宽度与高度），仅在 resizable 为 true 时生效。                                                                                  | `{ width?: number \| string; height?: number \| string }`                        | -      |
| info               | 参考来源分组列表，每组包含 header、key 与 annotations。                                                                                  | `AnnotationInfo[]`                                                               | -      |
| maxWidth           | 可调整的最大宽度，仅在 resizable 为 true 时生效。                                                                                        | `string \| number`                                                               | -      |
| minWidth           | 可调整的最小宽度，仅在 resizable 为 true 时生效，默认 150。                                                                              | `string \| number`                                                               | -      |
| motion             | 显示与隐藏时是否使用展开动画，默认 true。                                                                                                | `boolean`                                                                        | -      |
| onCancel           | 点击关闭按钮，或在 closeOnEsc 为 true 时按下 Esc 触发。                                                                                  | <ApiType detail={"(e: MouseEvent \u007c KeyboardEvent) => void"}>Type</ApiType>  | -      |
| onChange           | 展开的分组变化时触发，参数为当前展开的 key。                                                                                             | <ApiType detail={"(key: string \u007c string[]) => void"}>Type</ApiType>         | -      |
| onClick            | 点击某条参考来源时触发，参数为事件和该条目。                                                                                             | <ApiType detail={"(e: MouseEvent, item: AnnotationItem) => void"}>Type</ApiType> | -      |
| renderHeader       | 自定义整个头部；返回内容为空时使用内置头部。                                                                                             | <ApiType detail={"() => JSX.Element"}>Type</ApiType>                             | -      |
| renderInfoHeader   | 渲染分组标题图标后的内容，参数为该分组；不传时显示分组的 header 文本。即 AnnotationContent 的 renderHeader，与容器的 renderHeader 区分。 | `ContentProps['renderHeader']`                                                   | -      |
| renderItem         | 自定义参考来源条目的渲染，参数为该条目。                                                                                                 | <ApiType detail={"(annotation: AnnotationItem) => JSX.Element"}>Type</ApiType>   | -      |
| resizable          | 宽度是否可拖拽调整，默认 true；为 true 时 minWidth、maxWidth 与 defaultSize 生效。                                                       | `boolean`                                                                        | -      |
| showClose          | 是否显示头部的关闭按钮，默认 true。                                                                                                      | `boolean`                                                                        | -      |
| style              | 容器元素的内联样式，可传样式对象或样式字符串。                                                                                           | `JSX.CSSProperties \| string`                                                    | -      |
| title              | 头部标题，默认使用语言包中的参考来源。                                                                                                   | `JSX.Element`                                                                    | -      |
| visible            | 是否显示。                                                                                                                               | `boolean`                                                                        | -      |
