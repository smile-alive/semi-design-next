import ApiType from '#/components/ApiType.astro';

**Props**

| Property           | Description                                                                                                                         | Type                                                                                   | Default |
| ------------------ | ----------------------------------------------------------------------------------------------------------------------------------- | -------------------------------------------------------------------------------------- | ------- |
| activeKey          | Keys of the expanded groups, one or several.                                                                                        | `string \| string[]`                                                                   | -       |
| afterVisibleChange | Called after a visibility change has completed, with whether the container is visible; with motion it runs when the animation ends. | <ApiType detail={"(isVisible: boolean) => void"}>Type</ApiType>                        | -       |
| children           | Has no effect; the content is the reference list.                                                                                   | `React.ReactNode`                                                                      | -       |
| closeOnEsc         | Whether pressing Esc calls onCancel, false by default.                                                                              | `boolean`                                                                              | -       |
| containerRef       | Receives the root element (`.semi-sidebar-container`), as `ref` does.                                                               | <ApiType detail={"React.RefObject\u003cHTMLDivElement>"}>Type</ApiType>                | -       |
| defaultSize        | Initial size (width and height); applies only when resizable is true.                                                               | `{ width?: number \| string; height?: number \| string }`                              | -       |
| info               | Groups of references; each has a header, a key and its annotations.                                                                 | `AnnotationInfo[]`                                                                     | -       |
| maxWidth           | Maximum width when resizing; applies only when resizable is true.                                                                   | `string \| number`                                                                     | -       |
| minWidth           | Minimum width when resizing; applies only when resizable is true, 150 by default.                                                   | `string \| number`                                                                     | -       |
| motion             | Whether showing and hiding use the expand animation, true by default.                                                               | `boolean`                                                                              | -       |
| onCancel           | Called when the close button is clicked, or when Esc is pressed while closeOnEsc is true, with the event.                           | <ApiType detail={"(e: any) => void"}>Type</ApiType>                                    | -       |
| onChange           | Called when the expanded groups change, with the expanded keys.                                                                     | <ApiType detail={"(key: string \u007c string[]) => void"}>Type</ApiType>               | -       |
| onClick            | Called when a reference is clicked, with the event and the item.                                                                    | <ApiType detail={"(e: React.MouseEvent, item: AnnotationItem) => void"}>Type</ApiType> | -       |
| ref                | Reference to the root element (`.semi-sidebar-container`); `null` after it is hidden and its leave animation has ended.             | <ApiType detail={"React.Ref\u003cHTMLDivElement>"}>Type</ApiType>                      | -       |
| renderHeader       | Renders the whole header; the built-in header is used when it returns no content.                                                   | <ApiType detail={"() => ReactNode"}>Type</ApiType>                                     | -       |
| renderItem         | Renders a reference item, called with the item.                                                                                     | <ApiType detail={"(annotation: AnnotationItem) => ReactNode"}>Type</ApiType>           | -       |
| resizable          | Whether the width can be resized by dragging, true by default; minWidth, maxWidth and defaultSize apply only when true.             | `boolean`                                                                              | -       |
| showClose          | Whether the header shows the close button, true by default.                                                                         | `boolean`                                                                              | -       |
| style              | Inline style of the root element.                                                                                                   | `CSSProperties`                                                                        | -       |
| title              | Title in the header, the locale's references text by default.                                                                       | `ReactNode`                                                                            | -       |
| visible            | Whether the container is shown.                                                                                                     | `boolean`                                                                              | -       |
