import ApiType from '#/components/ApiType.astro';

**Props**

| 属性               | 说明                                                                                                                                | 类型                                                                                                             | 默认值 |
| ------------------ | ----------------------------------------------------------------------------------------------------------------------------------- | ---------------------------------------------------------------------------------------------------------------- | ------ |
| afterVisibleChange | 可见状态变化完成后的回调，参数为当前是否可见；有动画时在动画结束后触发。                                                            | <ApiType detail={"(isVisible: boolean) => void"}>Type</ApiType>                                                  | -      |
| children           | 不生效，内容由 MCP 工具列表渲染。                                                                                                   | `React.ReactNode`                                                                                                | -      |
| closeOnEsc         | 按下 Esc 键时是否触发 onCancel，默认 false。                                                                                        | `boolean`                                                                                                        | -      |
| containerRef       | 接收根元素（`.semi-sidebar-container`），作用同 `ref`。                                                                             | <ApiType detail={"React.RefObject\u003cHTMLDivElement>"}>Type</ApiType>                                          | -      |
| customOptions      | 自定义 MCP 工具列表，在自定义页签中展示；每项的 value 必填且唯一。                                                                  | `MCPReactOption[]`                                                                                               | -      |
| defaultSize        | 初始尺寸（宽度与高度），仅在 resizable 为 true 时生效。                                                                             | `{ width?: number \| string; height?: number \| string }`                                                        | -      |
| filter             | 自定义搜索过滤，参数为转为小写的输入值和选项；默认按 label 与 desc 匹配。                                                           | <ApiType detail={"(inputValue: string, option: MCPReactOption) => boolean"}>Type</ApiType>                       | -      |
| maxWidth           | 可调整的最大宽度，仅在 resizable 为 true 时生效。                                                                                   | `string \| number`                                                                                               | -      |
| minWidth           | 可调整的最小宽度，仅在 resizable 为 true 时生效，默认 150。                                                                         | `string \| number`                                                                                               | -      |
| motion             | 显示与隐藏时是否使用展开动画，默认 true。                                                                                           | `boolean`                                                                                                        | -      |
| onAddClick         | 点击自定义页签中的新增按钮时触发，参数为事件。                                                                                      | <ApiType detail={"(e: React.MouseEvent\u003cHTMLButtonElement>) => void"}>Type</ApiType>                         | -      |
| onCancel           | 点击关闭按钮，或在 closeOnEsc 为 true 时按下 Esc 触发，参数为事件。                                                                 | <ApiType detail={"(e: any) => void"}>Type</ApiType>                                                              | -      |
| onConfigureClick   | 点击内置 MCP 工具的配置按钮时触发，参数为事件和该选项。                                                                             | <ApiType detail={"(e: React.MouseEvent\u003cHTMLButtonElement>, option: MCPReactOption) => void"}>Type</ApiType> | -      |
| onEditClick        | 点击自定义 MCP 工具的编辑按钮时触发，参数为事件和该选项。                                                                           | <ApiType detail={"(e: React.MouseEvent\u003cHTMLButtonElement>, option: MCPReactOption) => void"}>Type</ApiType> | -      |
| onSearch           | 搜索时触发，参数为输入值和是否在自定义列表中搜索。                                                                                  | <ApiType detail={"(inputValue: string, custom: boolean) => void"}>Type</ApiType>                                 | -      |
| onStatusChange     | 切换启用或关闭时触发，参数为切换后的完整选项列表和是否为自定义列表；组件不会自行修改列表，需把新列表写回 options 或 customOptions。 | <ApiType detail={"(options: MCPReactOption[], custom: boolean) => void"}>Type</ApiType>                          | -      |
| options            | 内置 MCP 工具列表；每项的 value 必填且唯一。                                                                                        | `MCPReactOption[]`                                                                                               | -      |
| placeholder        | 搜索框的占位文字，默认使用语言包中的文案。                                                                                          | `string`                                                                                                         | -      |
| ref                | 根元素（`.semi-sidebar-container`）的引用，隐藏且离场动画结束后为 `null`。                                                          | <ApiType detail={"React.Ref\u003cHTMLDivElement>"}>Type</ApiType>                                                | -      |
| renderHeader       | 自定义整个头部，返回内容为空时使用内置头部。                                                                                        | <ApiType detail={"() => ReactNode"}>Type</ApiType>                                                               | -      |
| renderItem         | 自定义整行工具项，参数为 `{ option, custom }`。                                                                                     | <ApiType detail={"(props: { option: MCPReactOption; custom: boolean }) => ReactNode"}>Type</ApiType>             | -      |
| resizable          | 宽度是否可拖拽调整，默认 true；为 true 时 minWidth、maxWidth 与 defaultSize 生效。                                                  | `boolean`                                                                                                        | -      |
| showClose          | 是否显示头部的关闭按钮，默认 true。                                                                                                 | `boolean`                                                                                                        | -      |
| style              | 根元素的内联样式。                                                                                                                  | `CSSProperties`                                                                                                  | -      |
| title              | 头部标题，默认使用语言包中的 MCP 配置文案。                                                                                         | `ReactNode`                                                                                                      | -      |
| visible            | 是否显示。                                                                                                                          | `boolean`                                                                                                        | -      |
