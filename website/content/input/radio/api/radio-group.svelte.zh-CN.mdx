import ApiType from '#/components/ApiType.astro';

**Props**

| 属性       | 说明                                                                                                                                                                  | 类型                                                                             | 默认值         |
| ---------- | --------------------------------------------------------------------------------------------------------------------------------------------------------------------- | -------------------------------------------------------------------------------- | -------------- |
| buttonSize | type 为 button 时按钮的尺寸，可选 small、middle、large。                                                                                                              | `RadioGroupButtonSize`                                                           | `"middle"`     |
| children   | 组内的 Radio；设置 options 时忽略。                                                                                                                                   | `Snippet`                                                                        | -              |
| direction  | 组内 Radio 的排列方向，可选 horizontal 或 vertical；type 为 button 时不支持。                                                                                         | `RadioDirection`                                                                 | `"horizontal"` |
| disabled   | 是否禁用整组。                                                                                                                                                        | `boolean`                                                                        | `false`        |
| extra      | 自定义 options 中每一项辅助文本的 snippet，参数为 `{ option, index }`。                                                                                               | <ApiType detail={"Snippet\u003c[RadioGroupOptionSnippetParams]>"}>Type</ApiType> | -              |
| label      | 自定义 options 中每一项文本的 snippet，参数为 `{ option, index }`。                                                                                                   | <ApiType detail={"Snippet\u003c[RadioGroupOptionSnippetParams]>"}>Type</ApiType> | -              |
| mode       | 设为 advanced 时，已选中的 Radio 可以通过点击或空格取消选中，此时选中值变为 undefined。                                                                               | `RadioMode`                                                                      | `""`           |
| name       | 组内所有原生 input 的 name；未设置时每组生成唯一值。                                                                                                                  | `string`                                                                         | -              |
| onBlur     | 焦点离开整个分组时的回调，在组内 Radio 之间移动焦点不触发。                                                                                                           | <ApiType detail={"(event: FocusEvent) => void"}>Type</ApiType>                   | -              |
| onChange   | 选中的值变化时的回调，参数的 `target.value` 为新的选中值；mode 为 advanced 且取消选中时为 undefined。                                                                 | <ApiType detail={"(event: RadioChangeEvent) => void"}>Type</ApiType>             | -              |
| options    | 以配置形式声明选项，元素为字符串或 `{ label, value, disabled, extra, class, style, addonId, addonStyle, addonClass, extraId }`，其中 `style`、`addonStyle` 为字符串。 | <ApiType detail={"Array\u003cstring \u007c RadioOptionItem>"}>Type</ApiType>     | -              |
| prefixCls  | CSS 类名前缀。                                                                                                                                                        | `string`                                                                         | -              |
| type       | 组内所有 Radio 的样式类型，可选 default、button、card、pureCard。                                                                                                     | `RadioType`                                                                      | `"default"`    |
| value      | 选中的值，可用 `bind:value` 双向绑定，与 Radio 的 value 匹配。`bind:value={get, set}` 时完全受控；只传 `value` 时作为初值。                                           | `RadioValue`                                                                     | -              |
