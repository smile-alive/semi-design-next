import ApiType from '#/components/ApiType.astro';

**Props**

| 属性           | 说明                                                                                            | 类型                                                                                                     | 默认值      |
| -------------- | ----------------------------------------------------------------------------------------------- | -------------------------------------------------------------------------------------------------------- | ----------- |
| addonId        | addon 节点的 id，aria-labelledby 指向这个 id；未设置时随机生成。                                | `string`                                                                                                 | -           |
| checked        | 是否选中，受控使用；在 RadioGroup 中使用时无效。                                                | `boolean`                                                                                                | -           |
| classNames     | 内部各部位的类名，键为 addon（包裹内容的容器），与组件自带的类名合并。                          | <ApiType detail={"Partial\u003cRecord\u003cRadioPart, string>>"}>Type</ApiType>                          | -           |
| defaultChecked | 初始是否选中，非受控使用；在 RadioGroup 中使用时无效。                                          | `boolean`                                                                                                | `false`     |
| disabled       | 是否禁用。                                                                                      | `boolean`                                                                                                | -           |
| displayMode    | 设为 vertical 时 Radio 以块级元素独占一行。                                                     | `RadioDisplayMode`                                                                                       | -           |
| extra          | 辅助文本，aria-describedby 指向它；type 为 button 时不渲染。                                    | `JSX.Element`                                                                                            | -           |
| extraId        | extra 节点的 id，aria-describedby 指向这个 id；未设置时随机生成。                               | `string`                                                                                                 | -           |
| mode           | 设为 advanced 时，已选中的 Radio 可以通过点击或空格取消选中；在 RadioGroup 中以组的 mode 为准。 | `RadioMode`                                                                                              | `""`        |
| name           | 原生 input 的 name；在 RadioGroup 中以组的 name 为准。                                          | `string`                                                                                                 | -           |
| onblur         | 与 `onBlur` 相同，挂在原生 input 上。                                                           | <ApiType detail={"JSX.EventHandlerUnion\u003cHTMLInputElement, FocusEvent>"}>Type</ApiType>              | -           |
| onBlur         | 原生 input 失去焦点时的回调。                                                                   | <ApiType detail={"JSX.EventHandlerUnion\u003cHTMLInputElement, FocusEvent>"}>Type</ApiType>              | -           |
| onChange       | 选中状态变化时的回调，event.target 为 `{ checked, value, name, disabled }`。                    | <ApiType detail={"(event: RadioChangeEvent) => void"}>Type</ApiType>                                     | -           |
| onfocus        | 与 `onFocus` 相同，挂在原生 input 上。                                                          | <ApiType detail={"JSX.EventHandlerUnion\u003cHTMLInputElement, FocusEvent>"}>Type</ApiType>              | -           |
| onFocus        | 原生 input 获得焦点时的回调。                                                                   | <ApiType detail={"JSX.EventHandlerUnion\u003cHTMLInputElement, FocusEvent>"}>Type</ApiType>              | -           |
| preventScroll  | 聚焦时是否阻止浏览器滚动到该元素，作用于 focus 方法。                                           | `boolean`                                                                                                | -           |
| styles         | 内部各部位的内联样式，键为 addon（包裹内容的容器）。                                            | <ApiType detail={"Partial\u003cRecord\u003cRadioPart, JSX.CSSProperties \u007c string>>"}>Type</ApiType> | -           |
| type           | 样式类型，可选 default、button、card、pureCard；在 RadioGroup 中以组的 type 为准。              | `RadioType`                                                                                              | `"default"` |
| value          | Radio 的值，RadioGroup 据此判断是否选中；同时作为原生 input 的 value 参与表单提交。             | `string \| number \| boolean`                                                                            | -           |
