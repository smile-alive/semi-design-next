import ApiType from '#/components/ApiType.astro';

**Props**

| Property     | Description                                                                                                                                                                                             | Type                                                                 | Default        |
| ------------ | ------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------- | -------------------------------------------------------------------- | -------------- |
| buttonSize   | Button size when `type` is `button`: `small`, `middle` or `large`.                                                                                                                                      | `RadioGroupButtonSize`                                               | `"middle"`     |
| children     | The Radios in the group. Ignored when `options` is set.                                                                                                                                                 | `React.ReactNode`                                                    | -              |
| className    | Class name of the root element.                                                                                                                                                                         | `string`                                                             | -              |
| defaultValue | The value selected by default when uncontrolled.                                                                                                                                                        | `string \| number \| boolean`                                        | -              |
| direction    | Layout direction of the radios: `horizontal` or `vertical`. Has no effect when type is `button`.                                                                                                        | `'horizontal' \| 'vertical'`                                         | `"horizontal"` |
| disabled     | Whether the whole group is disabled.                                                                                                                                                                    | `boolean`                                                            | `false`        |
| id           | Id of the root element.                                                                                                                                                                                 | `string`                                                             | -              |
| mode         | When set to `advanced`, a checked radio can be unchecked by clicking it or pressing Space. On unchecking, `event.target.value` is `undefined`.                                                          | `RadioMode`                                                          | `""`           |
| name         | The name of every native input in the group. A unique value is generated per group when omitted.                                                                                                        | `string`                                                             | -              |
| onChange     | Called when the selected value changes, and on every click when `mode` is `advanced`. `event.target` contains `checked`, `value`, `name` and `disabled`.                                                | <ApiType detail={"(event: RadioChangeEvent) => void"}>Type</ApiType> | -              |
| options      | Declares the options as data. Each item is a string or `{ label, value, disabled, extra, className, style, addonId, addonStyle, addonClassName, extraId }`, where `label` and `extra` can be ReactNode. | `Options`                                                            | -              |
| prefixCls    | CSS class name prefix.                                                                                                                                                                                  | `string`                                                             | -              |
| ref          | Reference to the root div element.                                                                                                                                                                      | <ApiType detail={"React.Ref\u003cHTMLDivElement>"}>Type</ApiType>    | -              |
| style        | Inline style of the root element.                                                                                                                                                                       | `React.CSSProperties`                                                | -              |
| type         | Style type of every Radio in the group: `default`, `button`, `card` or `pureCard`.                                                                                                                      | `RadioType`                                                          | `"default"`    |
| value        | The selected value, for controlled use.                                                                                                                                                                 | `string \| number \| boolean`                                        | -              |
