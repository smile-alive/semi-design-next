import ApiType from '#/components/ApiType.astro';

**Props**

| 属性         | 说明                                                                                                                                                                          | 类型                                                                 | 默认值         |
| ------------ | ----------------------------------------------------------------------------------------------------------------------------------------------------------------------------- | -------------------------------------------------------------------- | -------------- |
| buttonSize   | `type` 为 `button` 时按钮的尺寸，可选 `small`、`middle` 或 `large`。                                                                                                          | `RadioGroupButtonSize`                                               | `"middle"`     |
| children     | 组内的 Radio，设置了 `options` 时忽略。                                                                                                                                       | `React.ReactNode`                                                    | -              |
| className    | 根元素的类名。                                                                                                                                                                | `string`                                                             | -              |
| defaultValue | 非受控时默认选中的值。                                                                                                                                                        | `string \| number \| boolean`                                        | -              |
| direction    | 组内 Radio 的排列方向，可选 `horizontal` 或 `vertical`；`type` 为 `button` 时不生效。                                                                                         | `'horizontal' \| 'vertical'`                                         | `"horizontal"` |
| disabled     | 是否禁用整组。                                                                                                                                                                | `boolean`                                                            | `false`        |
| id           | 根元素的 id。                                                                                                                                                                 | `string`                                                             | -              |
| mode         | 设为 `advanced` 时，已选中的 Radio 可以通过点击或空格取消选中；取消选中时 `event.target.value` 为 `undefined`。                                                               | `RadioMode`                                                          | `""`           |
| name         | 组内所有原生 input 的 name；未设置时每组生成唯一值。                                                                                                                          | `string`                                                             | -              |
| onChange     | 选中的值变化时的回调，`mode` 为 `advanced` 时每次点击都会触发；`event.target` 包含 `checked`、`value`、`name` 和 `disabled`。                                                 | <ApiType detail={"(event: RadioChangeEvent) => void"}>Type</ApiType> | -              |
| options      | 以配置形式声明选项，元素为字符串或 `{ label, value, disabled, extra, className, style, addonId, addonStyle, addonClassName, extraId }`，`label` 与 `extra` 可以是 ReactNode。 | `Options`                                                            | -              |
| prefixCls    | CSS 类名前缀。                                                                                                                                                                | `string`                                                             | -              |
| ref          | 根 div 元素的引用。                                                                                                                                                           | <ApiType detail={"React.Ref\u003cHTMLDivElement>"}>Type</ApiType>    | -              |
| style        | 根元素的内联样式。                                                                                                                                                            | `React.CSSProperties`                                                | -              |
| type         | 组内所有 Radio 的样式类型，可选 `default`、`button`、`card` 或 `pureCard`。                                                                                                   | `RadioType`                                                          | `"default"`    |
| value        | 选中的值，受控使用。                                                                                                                                                          | `string \| number \| boolean`                                        | -              |
