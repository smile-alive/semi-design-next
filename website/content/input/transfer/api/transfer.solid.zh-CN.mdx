import ApiType from '#/components/ApiType.astro';

**Props**

| 属性 | 说明 | 类型 | 默认值 |
| --- | --- | --- | --- |
| dataSource | 数据源。type 为 `list` 时是数据项数组，`groupList` 时是分组数组，`treeList` 时是带 `children` 的树形数据。数据按数组引用计算，原地修改数组不会生效，需要传入新数组。 | `DataSource` | `[]` |
| defaultValue | 默认已选中的 value 数组，只在创建时读取。 | <ApiType detail={"Array\u003cstring \u007c number>"}>Type</ApiType> | `[]` |
| disabled | 是否禁用。 | `boolean` | - |
| draggable | 是否开启右侧已选项的拖拽排序。排序后 `onChange` 的数组顺序随之改变。 | `boolean` | - |
| emptyContent | 自定义空状态的提示：`left` 为左侧没有数据时，`right` 为右侧没有已选项时，`search` 为搜索没有结果时。某一项为 `null` 时不渲染对应的空状态；未提供时显示语言包文案。 | `EmptyContent` | `{}` |
| filter | 筛选逻辑。为 `false` 时不显示搜索框；传入函数时自定义匹配，返回 `true` 表示保留该项；默认按 `label` 不区分大小写匹配。`treeList` 想自定义搜索时需设为 `true`，并通过 `treeProps` 的 `filterTreeNode` 传入匹配函数。 | <ApiType detail={"boolean \u007c ((searchValue: string, item: DataItem) => boolean)"}>Type</ApiType> | - |
| inputProps | 传给搜索框 Input 的属性，可配置项参考 Input；`value` 与 `onChange` 由 Transfer 占用，需要外部驱动搜索时调用 ref 上的 `search` 方法。 | `InputProps` | - |
| loading | 左侧选项是否正在加载，加载中显示 Spin 代替列表。 | `boolean` | - |
| onBlur | 焦点从 Transfer 内部移到外部时触发一次，参数为 FocusEvent；在搜索框、复选框、列表项、移除按钮、拖拽手柄之间移动不触发，窗口失焦（`relatedTarget` 为空）时触发。 | <ApiType detail={"(e: FocusEvent & { currentTarget: HTMLDivElement; target: Element }) => void"}>Type</ApiType> | - |
| onChange | 选中值变化时触发，拖拽排序后也会触发。参数依次为 value 数组与对应的数据项数组。 | <ApiType detail={"(values: Array\u003cstring \u007c number>, items: Array\u003cDataItem>) => void"}>Type</ApiType> | - |
| onDeselect | 取消勾选一项时触发，参数为该数据项。 | <ApiType detail={"(item: DataItem) => void"}>Type</ApiType> | - |
| onSearch | 搜索框输入内容变化时触发，参数为搜索值。通过 ref 的 `search` 方法驱动的搜索不触发。 | <ApiType detail={"(searchValue: string) => void"}>Type</ApiType> | - |
| onSelect | 勾选一项时触发，参数为该数据项。 | <ApiType detail={"(item: DataItem) => void"}>Type</ApiType> | - |
| pagination | 左侧面板的分页配置，仅对 `list` 与 `groupList` 生效。`onPageChange` 写在该对象内。 | `PaginationProps` | - |
| ref | 接收根 `div` 元素，类型为 TransferElement，元素上带有 `search` 方法。 | <ApiType detail={"Ref\u003cTransferElement>"}>Type</ApiType> | - |
| renderLabel | 自定义左右两侧每一项的内容与 treeList 中节点的内容，参数为数据项；未设置时显示 label。`showPath` 的路径仍是 label 文本。 | <ApiType detail={"(item: DataItem) => JSX.Element"}>Type</ApiType> | - |
| renderSelectedHeader | 自定义右侧面板头部，参数为实时对象（SelectedHeaderProps），只调用一次。 | <ApiType detail={"(headProps: SelectedHeaderProps) => JSX.Element"}>Type</ApiType> | - |
| renderSelectedItem | 自定义右侧单个已选项的渲染，每个选项调用一次。参数为数据项，另含 `onRemove`；`draggable` 时还有 `sortableHandle`。`treeList` 且 `showPath` 时可通过 `fullPath` 获取完整路径。 | <ApiType detail={"(item: RenderSelectedItemProps) => JSX.Element"}>Type</ApiType> | - |
| renderSelectedPanel | 完全自定义右侧已选面板，参数为实时对象（SelectedPanelProps），只调用一次。读取其字段需写在 JSX 中。 | <ApiType detail={"(selectedPanelProps: SelectedPanelProps) => JSX.Element"}>Type</ApiType> | - |
| renderSourceHeader | 自定义左侧面板头部，参数为实时对象（SourceHeaderProps），只调用一次。`leafOnlyNum` 仅 `treeList` 提供。 | <ApiType detail={"(headProps: SourceHeaderProps) => JSX.Element"}>Type</ApiType> | - |
| renderSourceItem | 自定义左侧单个候选项的渲染，每个选项调用一次。参数为数据项，另含 `onChange` 与 `checked`；`checked` 是 getter，需在返回的 JSX 中读取。 | <ApiType detail={"(item: RenderSourceItemProps) => JSX.Element"}>Type</ApiType> | - |
| renderSourcePanel | 完全自定义左侧候选面板，参数为实时对象（SourcePanelProps），只调用一次。读取其字段需写在 JSX 中，解构只取首次渲染时的值。 | <ApiType detail={"(sourcePanelProps: SourcePanelProps) => JSX.Element"}>Type</ApiType> | - |
| showPath | 仅 `treeList` 有效，控制右侧已选项是否显示从根节点到自身的选择路径。 | `boolean` | `false` |
| tabIndex | 设置在搜索框上，包括外层 Tooltip 注入的值；根元素本身不是焦点目标。 | `number \| string` | - |
| treeProps | 仅 `treeList` 有效，传给左侧 Tree 的属性，会覆盖默认的 `multiple`、`disableStrictly`、`leafOnly`、`filterTreeNode`、`defaultExpandAll` 与 `searchRender`（为 `false`）。`value`、`treeData`、`keyMaps`、`treeDataSimpleJson`、`onChangeWithObject`、`onChange` 与 `ref` 由 Transfer 决定，传入会被忽略。 | <ApiType detail={"Omit\u003cTreeProps, 'value' \u007c 'ref' \u007c 'onChange' \u007c 'keyMaps' \u007c 'treeData' \u007c 'treeDataSimpleJson' \u007c 'onChangeWithObject'>"}>Type</ApiType> | - |
| type | 穿梭框类型，可选 `list`、`groupList`、`treeList`。 | `Type` | `"list"` |
| value | 已选中的 value 数组，传入后即为受控组件，数组顺序就是右侧的显示顺序；传入 `undefined` 回到非受控。 | <ApiType detail={"Array\u003cstring \u007c number>"}>Type</ApiType> | - |
| virtualize | 右侧已选列表虚拟化，仅在默认右侧面板且 `draggable` 为 `false` 时生效。`height` 为数字时按像素生效，为其他值时按面板剩余高度测量；`itemSize` 为每行高度。 | `VirtualizeProps` | - |

**Methods**

| 方法   | 说明                                                        | 类型                                                       |
| ------ | ----------------------------------------------------------- | ---------------------------------------------------------- |
| search | 按输入了该值来搜索，该值会被置给搜索框；不触发 `onSearch`。 | <ApiType detail={"(value: string) => void"}>Type</ApiType> |
