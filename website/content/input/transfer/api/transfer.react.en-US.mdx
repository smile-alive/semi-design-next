import ApiType from '#/components/ApiType.astro';

**Props**

| Property | Description | Type | Default |
| --- | --- | --- | --- |
| className | Class name of the root element. | `string` | - |
| dataSource | Data source: an array of items for `list`, an array of groups for `groupList`, or tree data with `children` for `treeList`. | `DataSource` | `[]` |
| defaultValue | Values selected initially, read once when the transfer is created. | <ApiType detail={"Array\u003cstring \u007c number>"}>Type</ApiType> | `[]` |
| disabled | Whether the whole transfer is disabled, including the search box, the options and the action buttons. | `boolean` | - |
| draggable | Enables drag sorting of the selected items on the right. After a sort, the order of the array passed to `onChange` changes. | `boolean` | - |
| emptyContent | Custom empty-state content: `left` when the left list has no data, `right` when nothing is selected, `search` when the search matches nothing. A `null` entry hides that empty state; a missing one shows the locale text. | `EmptyContent` | `{}` |
| filter | Filtering of the search. `false` hides the search box; a function replaces the default case-insensitive match on `label` and returns `true` to keep an item. For a custom search in `treeList`, set it to `true` and pass the matching function as `filterTreeNode` in `treeProps`. | <ApiType detail={"boolean \u007c ((searchValue: string, item: DataItem) => boolean)"}>Type</ApiType> | - |
| inputProps | Props passed to the search Input, see Input for the options. `value` and `onChange` belong to Transfer; to search from outside, call the `search` method on the ref. | `InputProps` | - |
| loading | Whether the options on the left are loading; a Spin replaces the list while they load. | `boolean` | - |
| onChange | Called when the selection changes, including after a drag sort, with the array of values and the array of their items. | <ApiType detail={"(values: Array\u003cstring \u007c number>, items: Array\u003cDataItem>) => void"}>Type</ApiType> | - |
| onDeselect | Called when an item is unchecked, with that item. | <ApiType detail={"(item: DataItem) => void"}>Type</ApiType> | - |
| onSearch | Called when the text of the search box changes, with the search value. A search driven by the ref's `search` method does not call it. | <ApiType detail={"(searchValue: string) => void"}>Type</ApiType> | - |
| onSelect | Called when an item is checked, with that item. | <ApiType detail={"(item: DataItem) => void"}>Type</ApiType> | - |
| pagination | Paging of the left panel, for `list` and `groupList` only. `onPageChange` lives inside the object. | `PaginationProps` | - |
| ref | Receives a `TransferHandle` that exposes only the `search` method; it is not the root element. | <ApiType detail={"React.Ref\u003cTransferHandle>"}>Type</ApiType> | - |
| renderSelectedHeader | Replaces the header of the right panel; called on every render with SelectedHeaderProps. | <ApiType detail={"(headProps: { num: number; showButton: boolean; onClear: () => void; }) => React.ReactNode"}>Type</ApiType> | - |
| renderSelectedItem | Renders one selected item on the right, called once per item on every render. The argument is the item plus `onRemove`; with `draggable` it also has `sortableHandle`. With `treeList` and `showPath`, `fullPath` holds the whole path. | <ApiType detail={"(item: RenderSelectedItemProps) => React.ReactNode"}>Type</ApiType> | - |
| renderSelectedPanel | Replaces the whole right panel; called on every render with SelectedPanelProps. | <ApiType detail={"(selectedPanelProps: SelectedPanelProps) => React.ReactNode"}>Type</ApiType> | - |
| renderSourceHeader | Replaces the header of the left panel; called on every render with SourceHeaderProps. `leafOnlyNum` is provided for `treeList` only. | <ApiType detail={"(headProps: { num: number; showButton: boolean; allChecked: boolean; onAllClick: () => void; leafOnlyNum?: number; }) => React.ReactNode"}>Type</ApiType> | - |
| renderSourceItem | Renders one candidate on the left, called once per item on every render. The argument is the item plus `checked` and `onChange`, which toggles the checked state of the item. | <ApiType detail={"(item: RenderSourceItemProps) => React.ReactNode"}>Type</ApiType> | - |
| renderSourcePanel | Replaces the whole left panel; called on every render with SourcePanelProps. | <ApiType detail={"(sourcePanelProps: SourcePanelProps) => React.ReactNode"}>Type</ApiType> | - |
| showPath | `treeList` only: whether the selected items on the right show their path from the root. | `boolean` | `false` |
| style | Inline style of the root element. | `React.CSSProperties` | - |
| treeProps | `treeList` only: props passed to the Tree on the left, overriding the defaults `multiple`, `disableStrictly`, `leafOnly`, `filterTreeNode`, `defaultExpandAll` and `searchRender` (`false`). `value`, `treeData`, `keyMaps`, `treeDataSimpleJson`, `onChangeWithObject`, `onChange` and `ref` belong to Transfer and are ignored. | <ApiType detail={"Omit\u003cTreeProps, 'value' \u007c 'ref' \u007c 'onChange' \u007c 'keyMaps' \u007c 'treeData' \u007c 'treeDataSimpleJson' \u007c 'onChangeWithObject'>"}>Type</ApiType> | - |
| type | Transfer type: `list`, `groupList` or `treeList`. | `Type` | `"list"` |
| value | Values of the selected items. Passing it makes the component controlled, and its order is the order on the right; `undefined` returns to uncontrolled. | <ApiType detail={"Array\u003cstring \u007c number>"}>Type</ApiType> | - |
| virtualize | Virtualizes the selected list on the right, only with the default right panel and `draggable` off. A numeric `height` applies in pixels; any other value is measured from the remaining panel height. `itemSize` is the height of each row. | `VirtualizeProps` | - |

**Methods**

| Method | Description                                                                                                  | Type                                                       |
| ------ | ------------------------------------------------------------------------------------------------------------ | ---------------------------------------------------------- |
| search | Searches as if the value were typed into the search box, and puts the value there; `onSearch` is not called. | <ApiType detail={"(value: string) => void"}>Type</ApiType> |
