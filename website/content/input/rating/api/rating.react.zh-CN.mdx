import ApiType from '#/components/ApiType.astro';

**Props**

| 属性          | 说明                                                                                                                                              | 类型                                                                                                       | 默认值          |
| ------------- | ------------------------------------------------------------------------------------------------------------------------------------------------- | ---------------------------------------------------------------------------------------------------------- | --------------- |
| allowClear    | 是否允许再次点击已选中的星后清除数值。                                                                                                            | `boolean`                                                                                                  | `true`          |
| allowHalf     | 是否允许半选。                                                                                                                                    | `boolean`                                                                                                  | `false`         |
| aria-invalid  | 是否校验失败，设置在根元素的 `aria-invalid` 上。                                                                                                  | `boolean`                                                                                                  | -               |
| aria-required | 是否必填，设置在根元素的 `aria-required` 上。                                                                                                     | `boolean`                                                                                                  | -               |
| autoFocus     | 挂载时自动获取焦点，落点与 `ref` 的 `focus()` 相同。                                                                                              | `boolean`                                                                                                  | -               |
| character     | 每颗星中绘制的字符，可传入 React 节点，如 `<IconLikeHeart />`，未设置时为星形图标。传入纯文本字符串时，该文本作为各星可访问名称与整组摘要的单位。 | `React.ReactNode`                                                                                          | -               |
| className     | 根元素的类名。                                                                                                                                    | `string`                                                                                                   | -               |
| count         | 星的总数。                                                                                                                                        | `number`                                                                                                   | `5`             |
| defaultValue  | 非受控时的初始值。                                                                                                                                | `number`                                                                                                   | `0`             |
| disabled      | 只读，无法进行交互，也无法获得焦点。                                                                                                              | `boolean`                                                                                                  | -               |
| id            | 根元素的 id。                                                                                                                                     | `string`                                                                                                   | -               |
| onBlur        | 焦点离开整个评分组件时的回调，在星之间移动不触发。                                                                                                | <ApiType detail={"(e: React.FocusEvent) => void"}>Type</ApiType>                                           | -               |
| onChange      | 选择变化时的回调，参数为新的值。                                                                                                                  | <ApiType detail={"(value: number) => void"}>Type</ApiType>                                                 | -               |
| onClick       | 点击星或按 Enter、Space 选中时的回调，参数为事件与星的序号；不会作为根元素的原生 click 监听。                                                     | <ApiType detail={"(e: React.MouseEvent \u007c React.KeyboardEvent, index: number) => void"}>Type</ApiType> | -               |
| onFocus       | 焦点进入整个评分组件时的回调，在星之间移动不触发。                                                                                                | <ApiType detail={"(e: React.FocusEvent) => void"}>Type</ApiType>                                           | -               |
| onHoverChange | 指针经过时数值变化的回调；指针移出或点击清除时参数为 `undefined`。                                                                                | <ApiType detail={"(value?: number) => void"}>Type</ApiType>                                                | -               |
| onKeyDown     | 方向键使值移动时的回调，其他按键不触发。                                                                                                          | <ApiType detail={"(e: React.KeyboardEvent) => void"}>Type</ApiType>                                        | -               |
| prefixCls     | 组件的样式类名前缀。                                                                                                                              | `string`                                                                                                   | `"semi-rating"` |
| preventScroll | 聚焦组件内的星（包括 `autoFocus` 与 `focus()`）时，是否阻止浏览器滚动文档以显示新聚焦的星。                                                       | `boolean`                                                                                                  | -               |
| ref           | 组件句柄，暴露 `focus()` 与 `blur()` 方法。                                                                                                       | <ApiType detail={"React.Ref\u003cRatingHandle>"}>Type</ApiType>                                            | -               |
| size          | 尺寸，可选 `default`、`small`，也可传入 number 自定义尺寸。                                                                                       | `'small' \| 'default' \| number`                                                                           | `"default"`     |
| style         | 根元素的样式。                                                                                                                                    | `React.CSSProperties`                                                                                      | `{}`            |
| tooltips      | 每颗星的提示信息，数组下标与星的序号对应；空字符串或缺省的项不显示提示。                                                                          | `string[]`                                                                                                 | -               |
| value         | 当前受控值，传入后由该值决定显示；变回 `undefined` 时回到 `defaultValue` 并转为非受控。                                                           | `number`                                                                                                   | -               |

**Methods**

| 方法  | 说明                                                             | 类型                                          |
| ----- | ---------------------------------------------------------------- | --------------------------------------------- |
| blur  | 焦点位于评分内时移除焦点；禁用时无效。                           | <ApiType detail={"() => void"}>Type</ApiType> |
| focus | 聚焦 Tab 会到达的 radio，与 `autoFocus` 的落点相同；禁用时无效。 | <ApiType detail={"() => void"}>Type</ApiType> |
