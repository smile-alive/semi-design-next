import ApiType from '#/components/ApiType.astro';

**Props**

| 属性          | 说明                                                                                                                                            | 类型                                                                                                                             | 默认值          |
| ------------- | ----------------------------------------------------------------------------------------------------------------------------------------------- | -------------------------------------------------------------------------------------------------------------------------------- | --------------- |
| allowClear    | 是否允许再次点击后清除。                                                                                                                        | `boolean`                                                                                                                        | `true`          |
| allowHalf     | 是否允许半选。                                                                                                                                  | `boolean`                                                                                                                        | `false`         |
| autofocus     | 挂载时自动获取焦点。                                                                                                                            | `boolean`                                                                                                                        | -               |
| character     | 每颗星中绘制的字符，可传入 JSX 元素，如 `<IconLikeHeart />`，未设置时为星形图标。传入纯文本字符串时，该文本作为各星可访问名称与整组摘要的单位。 | `JSX.Element`                                                                                                                    | -               |
| count         | 星的总数。                                                                                                                                      | `number`                                                                                                                         | `5`             |
| defaultValue  | 非受控时的初始值。                                                                                                                              | `number`                                                                                                                         | `0`             |
| disabled      | 只读，无法进行交互，也无法获得焦点。                                                                                                            | `boolean`                                                                                                                        | -               |
| name          | 不生效：radiogroup 不能带 name，传入的值被丢弃。                                                                                                | `string`                                                                                                                         | -               |
| onBlur        | 焦点离开整个评分组件时的回调，在星之间移动不触发。                                                                                              | <ApiType detail={"(event: FocusEvent & { currentTarget: HTMLElement; relatedTarget: Node \u007c null }) => void"}>Type</ApiType> | -               |
| onChange      | 选择变化时的回调，参数为新的值。                                                                                                                | <ApiType detail={"(value: number) => void"}>Type</ApiType>                                                                       | -               |
| onClick       | 点击星或按 Enter、Space 选中时的回调，参数为事件与星的序号；不会作为根元素的原生 click 监听。                                                   | <ApiType detail={"(event: MouseEvent \u007c KeyboardEvent, index: number) => void"}>Type</ApiType>                               | -               |
| onFocus       | 焦点进入整个评分组件时的回调，在星之间移动不触发。                                                                                              | <ApiType detail={"(event: FocusEvent & { currentTarget: HTMLElement; relatedTarget: Node \u007c null }) => void"}>Type</ApiType> | -               |
| onHoverChange | 指针经过时数值变化的回调；指针移出或点击清除时参数为 `undefined`。                                                                              | <ApiType detail={"(value?: number) => void"}>Type</ApiType>                                                                      | -               |
| onKeyDown     | 仅在方向键使值移动时的回调。                                                                                                                    | <ApiType detail={"(event: KeyboardEvent & { currentTarget: HTMLElement }) => void"}>Type</ApiType>                               | -               |
| prefixCls     | 组件的样式类名前缀。                                                                                                                            | `string`                                                                                                                         | `"semi-rating"` |
| preventScroll | 指示浏览器是否应滚动文档以显示新聚焦的元素，作用于组件内的聚焦。                                                                                | `boolean`                                                                                                                        | -               |
| size          | 尺寸，可选 default、small，也可传入 number 自定义尺寸。                                                                                         | `'small' \| 'default' \| number`                                                                                                 | `"default"`     |
| tooltips      | 每一项的提示信息。                                                                                                                              | `string[]`                                                                                                                       | -               |
| value         | 当前受控值，传入后由该值决定显示；变回 undefined 时回到 defaultValue 并转为非受控。                                                             | `number`                                                                                                                         | -               |
