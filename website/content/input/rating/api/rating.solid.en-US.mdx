import ApiType from '#/components/ApiType.astro';

**Props**

| Property | Description | Type | Default |
| --- | --- | --- | --- |
| allowClear | Whether clicking the selected star again clears the value. | `boolean` | `true` |
| allowHalf | Whether half stars can be selected. | `boolean` | `false` |
| autofocus | Whether to focus on mount. | `boolean` | - |
| character | Character drawn in every star, a JSX element such as `<IconLikeHeart />`; the star icon when not set. When it is a plain text string, that text is the unit in each star's accessible name and the group summary. | `JSX.Element` | - |
| count | Total number of stars. | `number` | `5` |
| defaultValue | Initial value when uncontrolled. | `number` | `0` |
| disabled | Read-only: no interaction and no focus. | `boolean` | - |
| name | Ineffective: a radiogroup cannot carry a name, so the value is dropped. | `string` | - |
| onBlur | Called when focus leaves the whole rating; moving between stars does not call it. | <ApiType detail={"(event: FocusEvent & { currentTarget: HTMLElement; relatedTarget: Node \u007c null }) => void"}>Type</ApiType> | - |
| onChange | Called when the selection changes, with the new value. | <ApiType detail={"(value: number) => void"}>Type</ApiType> | - |
| onClick | Called when a star is clicked or selected with Enter or Space, with the event and the star index. It is not a native click listener on the root element. | <ApiType detail={"(event: MouseEvent \u007c KeyboardEvent, index: number) => void"}>Type</ApiType> | - |
| onFocus | Called when focus enters the whole rating; moving between stars does not call it. | <ApiType detail={"(event: FocusEvent & { currentTarget: HTMLElement; relatedTarget: Node \u007c null }) => void"}>Type</ApiType> | - |
| onHoverChange | Called when the hovered value changes; receives `undefined` when the pointer leaves or a click clears the value. | <ApiType detail={"(value?: number) => void"}>Type</ApiType> | - |
| onKeyDown | Called only for an arrow key that moved the value. | <ApiType detail={"(event: KeyboardEvent & { currentTarget: HTMLElement }) => void"}>Type</ApiType> | - |
| prefixCls | Class name prefix of the component. | `string` | `"semi-rating"` |
| preventScroll | Whether the browser should avoid scrolling the document to show the newly focused element, for focus inside the component. | `boolean` | - |
| size | Size: default or small, or a number for a custom size. | `'small' \| 'default' \| number` | `"default"` |
| tooltips | Prompt text for each item. | `string[]` | - |
| value | Controlled value. When given, this value decides what is shown; back to undefined it returns to defaultValue and becomes uncontrolled. | `number` | - |
