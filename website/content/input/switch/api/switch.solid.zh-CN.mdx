import ApiType from '#/components/ApiType.astro';

**Props**

| 属性           | 说明                                                                                                                                    | 类型                                                                                                                                        | 默认值      |
| -------------- | --------------------------------------------------------------------------------------------------------------------------------------- | ------------------------------------------------------------------------------------------------------------------------------------------- | ----------- |
| checked        | 选中状态。不为 `undefined` 时为受控：显示由该值决定，点击只触发 onChange，需在回调中更新该值。变回 `undefined` 后取消选中并转为非受控。 | `boolean`                                                                                                                                   | -           |
| checkedText    | 选中时展示的内容，size 为 small 时不显示。                                                                                              | `JSX.Element`                                                                                                                               | -           |
| defaultChecked | 非受控时的初始选中状态，只在首次渲染时读取。                                                                                            | `boolean`                                                                                                                                   | -           |
| disabled       | 是否禁用。                                                                                                                              | `boolean`                                                                                                                                   | `false`     |
| loading        | 是否处于加载中；加载中 input 仍可聚焦并带 `aria-disabled`，切换被忽略且不触发 onChange。                                                | `boolean`                                                                                                                                   | `false`     |
| name           | 原生 input 的 name，用于表单提交。                                                                                                      | `string`                                                                                                                                    | -           |
| onBlur         | 原生 input 失去焦点时的回调。焦点事件不冒泡，因此挂在 input 上。                                                                        | <ApiType detail={"JSX.EventHandlerUnion\u003cHTMLInputElement, FocusEvent>"}>Type</ApiType>                                                 | -           |
| onChange       | 切换状态时的回调，参数为新的选中状态与原生 change 事件；加载中不触发。                                                                  | <ApiType detail={"(checked: boolean, event: Event & { currentTarget: HTMLInputElement; target: HTMLInputElement }) => void"}>Type</ApiType> | -           |
| onFocus        | 原生 input 获得焦点时的回调。焦点事件不冒泡，因此挂在 input 上。                                                                        | <ApiType detail={"JSX.EventHandlerUnion\u003cHTMLInputElement, FocusEvent>"}>Type</ApiType>                                                 | -           |
| size           | 开关尺寸，可选 large、default 或 small。                                                                                                | `Size`                                                                                                                                      | `"default"` |
| uncheckedText  | 未选中时展示的内容，size 为 small 时不显示。                                                                                            | `JSX.Element`                                                                                                                               | -           |
