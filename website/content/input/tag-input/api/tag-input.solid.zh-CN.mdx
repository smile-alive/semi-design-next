import ApiType from '#/components/ApiType.astro';

**Props**

| 属性 | 说明 | 类型 | 默认值 |
| --- | --- | --- | --- |
| addOnBlur | 焦点离开组件时，是否将输入框内容自动创建为标签。 | `boolean` | `false` |
| allowDuplicates | 是否允许添加相同的标签。 | `boolean` | `true` |
| aria-label | 输入框的可访问名称，落在输入框上，根节点不带。 | `string` | - |
| autofocus | 挂载时是否自动聚焦输入框并激活组件，禁用时不生效。 | `boolean` | `false` |
| clearIcon | 替换清除按钮中的图标；内容为空时使用默认图标。 | `JSX.Element` | - |
| defaultValue | 非受控时的初始标签。 | `string[]` | - |
| disabled | 是否禁用。 | `boolean` | - |
| draggable | 是否开启拖拽排序。 | `boolean` | `false` |
| expandRestTagsOnClick | 不可拖拽时，焦点进入组件（点击、Tab 或 focus 方法）后是否展开被折叠的标签。 | `boolean` | `true` |
| inputValue | 当前输入框内容，配合 onInputChange 实现受控。 | `string` | - |
| insetLabel | 内嵌标签，显示在输入框前；同时提供 prefix 时只显示 prefix。显示时，没有 aria-label、aria-labelledby 的输入框以它命名。 | `JSX.Element` | - |
| insetLabelId | 内嵌标签容器的 id，供 aria-labelledby 引用；未设置时组件自动生成。 | `string` | - |
| max | 允许的标签最大数量。 | `number` | - |
| maxLength | 单个标签的最大长度。 | `number` | - |
| maxTagCount | 标签的最大展示数量，超出后以 +N 形式展示。 | `number` | - |
| name | 输入框的 name，供表单使用。 | `string` | - |
| onAdd | 添加标签时触发，参数为新增的标签。 | <ApiType detail={"(addedValue: string[]) => void"}>Type</ApiType> | - |
| onBlur | 焦点离开整个组件时触发一次；焦点在组件内部移动不触发。 | <ApiType detail={"(e: FocusEvent & { currentTarget: HTMLDivElement; target: Element }) => void"}>Type</ApiType> | - |
| onChange | 标签变化时触发，参数为完整的标签数组。 | <ApiType detail={"(value: string[]) => void"}>Type</ApiType> | - |
| onExceed | 添加后超过 max 时触发，参数为截断前的完整列表。 | <ApiType detail={"(value: string[]) => void"}>Type</ApiType> | - |
| onFocus | 焦点从组件外进入时触发一次；焦点在输入框、标签、清除按钮之间移动不触发。 | <ApiType detail={"(e: FocusEvent & { currentTarget: HTMLDivElement; target: Element }) => void"}>Type</ApiType> | - |
| onInputChange | 输入框内容变化时触发；事件可能是输入事件、KeyboardEvent（Enter 添加）、FocusEvent（addOnBlur）、MouseEvent（清除）或 CompositionEvent（输入法截断），读取专有字段前先按 `event.type` 收窄。 | <ApiType detail={"(value: string, e: Event) => void"}>Type</ApiType> | - |
| onInputExceed | 输入内容超过 maxLength 时触发，参数为被拒绝的输入内容。 | <ApiType detail={"(value: string) => void"}>Type</ApiType> | - |
| onKeyDown | 输入框按键时在组件处理之后触发；在其中调用 `preventDefault` 不会阻止 Enter 添加标签。 | <ApiType detail={"(e: InputElementEvent\u003cKeyboardEvent>) => void"}>Type</ApiType> | - |
| onRemove | 移除标签时触发，参数为被移除的标签及其序号。 | <ApiType detail={"(removedValue: string, idx: number) => void"}>Type</ApiType> | - |
| placeholder | 没有标签时输入框的占位文字。 | `string` | - |
| prefix | 输入框前缀，文字或图标会自动带上左右间隔；自定义元素的左右间隔为 0，可在元素中自行设置。 | `JSX.Element` | - |
| preventScroll | 组件内 focus 方法是否阻止浏览器滚动到新聚焦的元素。 | `boolean` | - |
| renderTagItem | 自定义标签渲染，返回的元素替换默认标签。第三个参数 onClose 关闭该标签，可直接作为按钮的 `onClick`。 | <ApiType detail={"(value: string, index: number, onClose: () => void) => JSX.Element"}>Type</ApiType> | - |
| restTagsPopoverProps | +N 浮层的 Popover 配置属性，可以控制弹出方向、trigger 等；不要传 `content`，它会替换剩余标签的列表。 | `RestTagsPopoverProps` | - |
| separator | 批量输入时的分隔符，可传多个；传 null 不拆分。 | `string \| string[] \| null` | `","` |
| showClear | 是否显示一键清除所有标签和输入内容的按钮。 | `boolean` | `false` |
| showContentTooltip | 标签文字被截断时，hover 标签是否通过浮层显示全部内容；传对象时用 `type` 指定浮层类型（`tooltip` 或 `popover`），用 `opts` 透传给浮层的属性。 | `boolean \| ShowTooltip` | `true` |
| showRestTagsPopover | 超过 maxTagCount 后，hover 到 +N 时是否通过 Popover 显示剩余标签。 | `boolean` | `true` |
| size | 尺寸，可选 small、default、large。 | `TagInputSize` | `"default"` |
| split | 自定义分隔符处理函数，参数为原始字符串与 separator，返回拆分后的字符串数组。 | <ApiType detail={"(originString: string, separators: string \u007c string[] \u007c null) => string[]"}>Type</ApiType> | - |
| suffix | 输入框后缀，文字或图标会自动带上左右间隔；自定义元素的左右间隔为 0，可在元素中自行设置。 | `JSX.Element` | - |
| validateStatus | 校验状态，仅影响样式；error 时输入框带 aria-invalid。 | `TagInputValidateStatus` | `"default"` |
| value | 当前标签，配合 onChange 实现受控；undefined 时按非受控处理，null 表示受控且没有标签。 | `string[] \| null` | - |
