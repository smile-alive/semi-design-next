import ApiType from '#/components/ApiType.astro';

**Props**

| 属性 | 说明 | 类型 | 默认值 |
| --- | --- | --- | --- |
| addOnBlur | 焦点离开组件时，是否将输入框内容自动创建为标签。 | `boolean` | `false` |
| allowDuplicates | 是否允许添加相同的标签。 | `boolean` | `true` |
| autofocus | 挂载时是否自动聚焦输入框并激活组件，禁用时不生效。 | `boolean` | `false` |
| clearIcon | 替换清除按钮中图标的 snippet；内容为空时使用默认图标。 | `Snippet` | - |
| disabled | 是否禁用。 | `boolean` | - |
| draggable | 是否开启拖拽排序。 | `boolean` | `false` |
| expandRestTagsOnClick | 不可拖拽时，焦点进入组件（点击、Tab 或 focus 方法）后是否展开被折叠的标签。 | `boolean` | `true` |
| inputValue | 当前输入框内容，可 `bind:inputValue`。不传时由组件自行管理；要完全受控用函数绑定 `bind:inputValue={get, set}`；绑成 `undefined` 会清空输入。 | `string` | - |
| insetLabel | 内嵌标签的 snippet，显示在输入框前；同时提供 prefix 时只显示 prefix。显示时，没有 aria-label、aria-labelledby 的输入框以它命名。 | `Snippet` | - |
| insetLabelId | 内嵌标签容器的 id，供 aria-labelledby 引用。 | `string` | - |
| max | 允许的标签最大数量。 | `number` | - |
| maxLength | 单个标签的最大长度。 | `number` | - |
| maxTagCount | 标签的最大展示数量，超出后以 +N 形式展示。 | `number` | - |
| onAdd | 添加标签时调用，参数为新增的标签。 | <ApiType detail={"(addedValue: string[]) => void"}>Type</ApiType> | - |
| onBlur | 焦点离开整个组件时调用一次；焦点在组件内部移动不调用。 | <ApiType detail={"(event: FocusEvent) => void"}>Type</ApiType> | - |
| onChange | 标签变化时调用。 | <ApiType detail={"(value: string[]) => void"}>Type</ApiType> | - |
| onclick | 点击根元素时在组件自己的激活处理之后调用。 | <ApiType detail={"MouseEventHandler\u003cT> \u007c null"}>Type</ApiType> | - |
| onExceed | 添加后超过 max 时调用，参数为截断前的完整列表。 | <ApiType detail={"(value: string[]) => void"}>Type</ApiType> | - |
| onFocus | 焦点从组件外进入时调用一次；焦点在输入框、标签、清除按钮之间移动不调用。 | <ApiType detail={"(event: FocusEvent) => void"}>Type</ApiType> | - |
| onInputChange | 输入框内容变化时调用；事件对象可能是 input、keydown、focusout、click 或 composition 事件，读取专有字段前先按 event.type 收窄。 | <ApiType detail={"(value: string, event: Event) => void"}>Type</ApiType> | - |
| onInputExceed | 输入内容超过 maxLength 时调用，参数为被拒绝的输入内容。 | <ApiType detail={"(value: string) => void"}>Type</ApiType> | - |
| onKeyDown | 输入框按键时在组件处理之后调用；其中调用 preventDefault 不会阻止 Enter 添加标签。 | <ApiType detail={"(event: KeyboardEvent) => void"}>Type</ApiType> | - |
| onRemove | 移除标签时调用，参数为被移除的标签及其序号。 | <ApiType detail={"(removedValue: string, index: number) => void"}>Type</ApiType> | - |
| placeholder | 没有标签时输入框的占位文字。 | `string` | - |
| prefix | 输入框前缀的 snippet，文字或图标会自动带上左右间隔。 | `Snippet` | - |
| preventScroll | 组件内 focus 方法是否阻止浏览器滚动到新聚焦的元素。 | `boolean` | - |
| restTagsPopoverProps | +N 浮层的 Popover 属性，不含 content、child、children。 | `RestTagsPopoverProps` | - |
| separator | 批量输入时的分隔符，可传多个；传 null 不拆分。 | `string \| string[] \| null` | `","` |
| showClear | 是否显示一键清除所有标签和输入内容的按钮。 | `boolean` | `false` |
| showContentTooltip | 标签文字被截断时，hover 标签是否通过浮层显示全部内容；传对象时可指定浮层类型与透传给浮层的属性。 | `boolean \| ShowTooltip` | `true` |
| showRestTagsPopover | 超过 maxTagCount 后，hover 到 +N 时是否通过 Popover 显示剩余标签。 | `boolean` | `true` |
| size | 尺寸，可选 small、default、large。 | `TagInputSize` | - |
| split | 自定义分隔符处理函数，返回拆分后的字符串数组。 | <ApiType detail={"(originString: string, separators: string \u007c string[] \u007c null) => string[]"}>Type</ApiType> | - |
| suffix | 输入框后缀的 snippet，文字或图标会自动带上左右间隔。 | `Snippet` | - |
| tag | 整体替换单个标签的 snippet，参数为 `{ value, index, onClose }`；内容为空时不渲染该标签，不回落默认标签。 | <ApiType detail={"Snippet\u003c[TagInputTagSnippetProps]>"}>Type</ApiType> | - |
| validateStatus | 校验状态，仅影响样式；error 时输入框带 aria-invalid。 | `TagInputValidateStatus` | `"default"` |
| value | 当前标签，可 `bind:value`。不传时由组件自行管理；`null` 表示受控且没有标签；单向 `value={x}` 只作初值，要完全受控用函数绑定 `bind:value={get, set}`；绑成 `undefined` 会清空标签。没有 defaultValue。 | `string[] \| null` | - |

**Methods**

| 方法  | 说明                                                           | 类型                                          |
| ----- | -------------------------------------------------------------- | --------------------------------------------- |
| blur  | 使组件内任意持有焦点的元素（输入框、标签、清除按钮）失去焦点。 | <ApiType detail={"() => void"}>Type</ApiType> |
| focus | 使输入框获得焦点并激活组件。                                   | <ApiType detail={"() => void"}>Type</ApiType> |
