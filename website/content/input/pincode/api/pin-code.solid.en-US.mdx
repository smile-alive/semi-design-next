import ApiType from '#/components/ApiType.astro';

**Props**

| Property       | Description                                                                                                                                                  | Type                                                                                                         | Default    |
| -------------- | ------------------------------------------------------------------------------------------------------------------------------------------------------------ | ------------------------------------------------------------------------------------------------------------ | ---------- |
| count          | Number of digits.                                                                                                                                            | `number`                                                                                                     | `6`        |
| defaultValue   | Initial content when uncontrolled.                                                                                                                           | `string`                                                                                                     | -          |
| disabled       | Whether the input is disabled.                                                                                                                               | `boolean`                                                                                                    | -          |
| format         | Format limit for a single character: number allows digits only, mixed allows digits and letters, and a regular expression or function judges each character. | <ApiType detail={"'number' \u007c 'mixed' \u007c RegExp \u007c ((value: string) => boolean)"}>Type</ApiType> | `"number"` |
| onBlur         | Called when focus leaves the whole code, with the native `FocusEvent`.                                                                                       | <ApiType detail={"(e: FocusEvent) => void"}>Type</ApiType>                                                   | -          |
| onChange       | Called when the content changes, with the current content.                                                                                                   | <ApiType detail={"(value: string) => void"}>Type</ApiType>                                                   | -          |
| onComplete     | Called once all digits are entered, with the current content.                                                                                                | <ApiType detail={"(value: string) => void"}>Type</ApiType>                                                   | -          |
| onFocus        | Called when focus enters the code from outside, with the native `FocusEvent`; moving between the inputs does not call it.                                    | <ApiType detail={"(e: FocusEvent) => void"}>Type</ApiType>                                                   | -          |
| size           | Input size: large, default or small.                                                                                                                         | `InputSize`                                                                                                  | -          |
| validateStatus | Validation status, passed to every input; error shows the error style.                                                                                       | `ValidateStatus`                                                                                             | -          |
| value          | Input content. Passing it makes the component controlled, so update it in onChange.                                                                          | `string`                                                                                                     | -          |
