import ApiType from '#/components/ApiType.astro';

**Props**

| Property                  | Description                                                                                                                                                          | Type                                                                                     | Default |
| ------------------------- | -------------------------------------------------------------------------------------------------------------------------------------------------------------------- | ---------------------------------------------------------------------------------------- | ------- |
| autosize                  | Whether the height follows the content; pass `{ minRows, maxRows }` to limit the rows.                                                                               | `boolean \| AutosizeRow`                                                                 | `false` |
| borderless                | Borderless mode.                                                                                                                                                     | `boolean`                                                                                | `false` |
| cols                      | Visible width in average character widths.                                                                                                                           | `number`                                                                                 | `20`    |
| composition               | Whether to enable IME mode; onChange does not fire during a composition and fires once after it ends.                                                                | `boolean`                                                                                | `false` |
| defaultValue              | Initial content when the text area is uncontrolled.                                                                                                                  | `string`                                                                                 | -       |
| disabled                  | Whether the text area is disabled.                                                                                                                                   | `boolean`                                                                                | -       |
| disabledEnterStartNewLine | Internal property used by Chat: Enter does not start a new line, Shift + Enter does.                                                                                 | `boolean`                                                                                | -       |
| getValueLength            | Custom string length calculation, used with maxLength, minLength and maxCount.                                                                                       | <ApiType detail={"(value: string) => number"}>Type</ApiType>                             | -       |
| lineNumberClassName       | Class name of the line number area.                                                                                                                                  | `string`                                                                                 | -       |
| lineNumberStart           | Starting line number.                                                                                                                                                | `number`                                                                                 | `1`     |
| lineNumberStyle           | Style of the line number area; object keys are kebab-case and numbers need a unit.                                                                                   | `JSX.CSSProperties`                                                                      | -       |
| maxCount                  | Maximum number of characters; setting it shows a counter, which turns red when exceeded. The input is not truncated.                                                 | `number`                                                                                 | -       |
| maxLength                 | Maximum length. With getValueLength the length is measured and truncated by it; otherwise the native attribute applies.                                              | `number`                                                                                 | -       |
| minLength                 | Minimum length. With getValueLength it is converted accordingly; otherwise the native attribute applies.                                                             | `number`                                                                                 | -       |
| onBlur                    | Called when focus leaves the text area. It listens on the wrapper, so focus moving between the text area and the clear button does not fire it.                      | <ApiType detail={"(e: FocusEvent) => void"}>Type</ApiType>                               | -       |
| onChange                  | Called when the text changes, with the new value and the event.                                                                                                      | <ApiType detail={"(value: string, e: TextAreaChangeEvent) => void"}>Type</ApiType>       | -       |
| onClear                   | Called when the clear button is clicked.                                                                                                                             | <ApiType detail={"(e: MouseEvent) => void"}>Type</ApiType>                               | -       |
| onEnterPress              | Called on Enter keydown; the Enter that commits an IME composition does not fire it.                                                                                 | <ApiType detail={"(e: TextAreaElementEvent\u003cKeyboardEvent>) => void"}>Type</ApiType> | -       |
| onFocus                   | Called when the text area gets focus. It listens on the wrapper, so focus moving inside the component does not fire it again.                                        | <ApiType detail={"(e: FocusEvent) => void"}>Type</ApiType>                               | -       |
| onResize                  | Called when the size changes: when autosize changes the height, and when the user drags the resize handle, which also provides width.                                | <ApiType detail={"(data: { height: number; width?: number }) => void"}>Type</ApiType>    | -       |
| readonly                  | Read-only: blocks typing and hides the clear button. The native readOnly only blocks typing.                                                                         | `boolean`                                                                                | -       |
| ref                       | Receives the wrapper div element; its `focus()` and `blur()` act on the textarea.                                                                                    | <ApiType detail={"Ref\u003cHTMLDivElement>"}>Type</ApiType>                              | -       |
| resize                    | Whether and in which direction the user may resize: none, both, horizontal, vertical, block or inline. It only takes effect when given and is ignored with autosize. | `'none' \| 'both' \| 'horizontal' \| 'vertical' \| 'block' \| 'inline'`                  | -       |
| rows                      | Number of visible text lines.                                                                                                                                        | `number`                                                                                 | `4`     |
| showClear                 | Shows a clear button when the text area has content and the pointer or focus is inside the component.                                                                | `boolean`                                                                                | `false` |
| showCounter               | Whether to show the counter; it is always shown when maxCount is set.                                                                                                | `boolean`                                                                                | `false` |
| showLineNumber            | Whether to show line numbers.                                                                                                                                        | `boolean`                                                                                | `false` |
| textareaRef               | Receives the native textarea element; showLineNumber remounts the textarea, so it is called again with the new element.                                              | <ApiType detail={"(element: HTMLTextAreaElement) => void"}>Type</ApiType>                | -       |
| textareaStyle             | Style of the inner textarea element, such as its height; object keys are kebab-case and numbers need a unit.                                                         | `JSX.CSSProperties`                                                                      | -       |
| validateStatus            | Validation status: default, error or warning; error outputs aria-invalid.                                                                                            | `ValidateStatus`                                                                         | -       |
| value                     | Text content, used as a controlled value together with onChange.                                                                                                     | `string`                                                                                 | -       |
