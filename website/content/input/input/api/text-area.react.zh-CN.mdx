import ApiType from '#/components/ApiType.astro';

**Props**

| 属性                      | 说明                                                                                                                                          | 类型                                                                                          | 默认值  |
| ------------------------- | --------------------------------------------------------------------------------------------------------------------------------------------- | --------------------------------------------------------------------------------------------- | ------- |
| autoFocus                 | 挂载后自动获得焦点。                                                                                                                          | `boolean`                                                                                     | -       |
| autosize                  | 高度是否随内容自适应。传入 `{ minRows, maxRows }` 可限制行数，`minRows` 缺省时取 `rows`。                                                     | `boolean \| AutosizeRow`                                                                      | `false` |
| borderless                | 无边框模式。                                                                                                                                  | `boolean`                                                                                     | `false` |
| cols                      | 可见宽度，按平均字符宽度计。                                                                                                                  | `number`                                                                                      | `20`    |
| composition               | 是否开启输入法模式，开启后输入法组合期间不触发 `onChange`，组合结束后触发一次。                                                               | `boolean`                                                                                     | `false` |
| defaultValue              | 非受控时的初始内容。                                                                                                                          | `string`                                                                                      | -       |
| disabled                  | 是否禁用。                                                                                                                                    | `boolean`                                                                                     | -       |
| disabledEnterStartNewLine | 内部属性，供 Chat 使用：Enter 不换行，Shift + Enter 换行。                                                                                    | `boolean`                                                                                     | -       |
| getValueLength            | 自定义计算字符串长度，搭配 `maxLength`、`minLength` 与 `maxCount` 使用。                                                                      | <ApiType detail={"(value: string) => number"}>Type</ApiType>                                  | -       |
| lineNumberClassName       | 行号区域的类名。                                                                                                                              | `string`                                                                                      | -       |
| lineNumberStart           | 行号起始值。                                                                                                                                  | `number`                                                                                      | `1`     |
| lineNumberStyle           | 行号区域的内联样式。                                                                                                                          | `CSSProperties`                                                                               | -       |
| maxCount                  | 字数上限，设置后显示字数统计；超出时计数器标红，但不截断输入，需要限制输入时搭配 `maxLength`。                                                | `number`                                                                                      | -       |
| onBlur                    | 失去焦点时的回调。监听在外层 `div` 上，焦点在文本框与清除按钮之间移动时不触发。                                                               | <ApiType detail={"(e: FocusEvent) => void"}>Type</ApiType>                                    | -       |
| onChange                  | 文本内容变化时的回调，参数为新值与事件。                                                                                                      | <ApiType detail={"(value: string, e: TextAreaChangeEvent) => void"}>Type</ApiType>            | -       |
| onClear                   | 点击清除按钮时的回调。                                                                                                                        | <ApiType detail={"(e: React.MouseEvent\u003cHTMLButtonElement>) => void"}>Type</ApiType>      | -       |
| onEnterPress              | 按下回车时的回调（keydown），输入法组合中确认的回车不触发。                                                                                   | <ApiType detail={"(e: React.KeyboardEvent\u003cHTMLTextAreaElement>) => void"}>Type</ApiType> | -       |
| onFocus                   | 获得焦点时的回调。监听在外层 `div` 上，组件内部的焦点移动不会再次触发。                                                                       | <ApiType detail={"(e: React.FocusEvent\u003cHTMLDivElement>) => void"}>Type</ApiType>         | -       |
| onInput                   | 原生 `input` 事件的回调，直接绑定在 `textarea` 元素上。                                                                                       | <ApiType detail={"(e: React.InputEvent\u003cHTMLTextAreaElement>) => void"}>Type</ApiType>    | -       |
| onKeyDown                 | keydown 回调，参数为原生事件。                                                                                                                | <ApiType detail={"(e: React.KeyboardEvent\u003cHTMLTextAreaElement>) => void"}>Type</ApiType> | -       |
| onKeyUp                   | keyup 回调，参数为原生事件，直接绑定在 `textarea` 元素上。                                                                                    | <ApiType detail={"(e: React.KeyboardEvent\u003cHTMLTextAreaElement>) => void"}>Type</ApiType> | -       |
| onResize                  | 尺寸变化回调：`autosize` 导致高度变化时触发；用户拖拽 `resize` 调整尺寸时也会触发，并额外给出 `width`。                                       | <ApiType detail={"(data: { height: number; width?: number }) => void"}>Type</ApiType>         | -       |
| placeholder               | 文本框为空时显示的提示文字。                                                                                                                  | `string`                                                                                      | -       |
| readonly                  | 是否只读：禁止输入，且不显示清除按钮。原生 `readOnly` 只禁止输入。                                                                            | `boolean`                                                                                     | -       |
| ref                       | 命令式句柄（`TextAreaHandle`），成员见 **Methods** 表。                                                                                       | <ApiType detail={"React.Ref\u003cTextAreaHandle>"}>Type</ApiType>                             | -       |
| resize                    | 是否允许用户拖拽调整尺寸及方向，可选 `none`、`both`、`horizontal`、`vertical`、`block`、`inline`。仅在显式传入时生效，`autosize` 开启时忽略。 | `'none' \| 'both' \| 'horizontal' \| 'vertical' \| 'block' \| 'inline'`                       | -       |
| rows                      | 可见行数。                                                                                                                                    | `number`                                                                                      | `4`     |
| showClear                 | 文本框有内容且悬停或焦点位于组件内时展示清除按钮；不可编辑时不显示。                                                                          | `boolean`                                                                                     | `false` |
| showCounter               | 是否显示字数统计；设置 `maxCount` 时总是显示。                                                                                                | `boolean`                                                                                     | `false` |
| showLineNumber            | 是否展示行号。                                                                                                                                | `boolean`                                                                                     | `false` |
| textareaStyle             | 内部 `textarea` 元素的内联样式，如高度、背景色。                                                                                              | `CSSProperties`                                                                               | -       |
| validateStatus            | 校验状态，可选 `default`、`error` 或 `warning`；`error` 时输出 `aria-invalid` 为 `true`。                                                     | `ValidateStatus`                                                                              | -       |
| value                     | 文本内容，受控使用，配合 `onChange` 更新。                                                                                                    | `string`                                                                                      | -       |

**Methods**

| 方法              | 说明                                                                                                    | 类型                                                                                                                                                   |
| ----------------- | ------------------------------------------------------------------------------------------------------- | ------------------------------------------------------------------------------------------------------------------------------------------------------ |
| blur              | 移出焦点。                                                                                              | <ApiType detail={"() => void"}>Type</ApiType>                                                                                                          |
| focus             | 获取焦点。                                                                                              | <ApiType detail={"(options?: FocusOptions) => void"}>Type</ApiType>                                                                                    |
| nativeElement     | 组件根元素，即外层 `div`。卸载后为 `null`。                                                             | `HTMLElement \| null`                                                                                                                                  |
| select            | 选中文本框中的全部文本。                                                                                | <ApiType detail={"() => void"}>Type</ApiType>                                                                                                          |
| setSelectionRange | 设置选区，参数同原生 `setSelectionRange`。                                                              | <ApiType detail={"(start: number \u007c null, end: number \u007c null, direction?: 'forward' \u007c 'backward' \u007c 'none') => void"}>Type</ApiType> |
| textarea          | 原生 `textarea` 元素。每次读取返回当前挂载的节点，卸载后为 `null`；切换 `showLineNumber` 会重新挂载它。 | `HTMLTextAreaElement \| null`                                                                                                                          |
