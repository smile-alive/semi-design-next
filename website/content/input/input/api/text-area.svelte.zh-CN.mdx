import ApiType from '#/components/ApiType.astro';

**Props**

| 属性                      | 说明                                                                                                                        | 类型                                                                                   | 默认值  |
| ------------------------- | --------------------------------------------------------------------------------------------------------------------------- | -------------------------------------------------------------------------------------- | ------- |
| autofocus                 | 挂载时自动获取焦点。                                                                                                        | `boolean`                                                                              | -       |
| autosize                  | 高度是否随内容自适应，可传 `{ minRows, maxRows }` 限制行数。                                                                | `boolean \| AutosizeRow`                                                               | `false` |
| borderless                | 无边框模式。                                                                                                                | `boolean`                                                                              | `false` |
| cols                      | 默认列数。                                                                                                                  | `number`                                                                               | `20`    |
| composition               | 是否开启输入法模式，开启后输入法组合期间不调用 onChange，组合结束后调用一次。                                               | `boolean`                                                                              | `false` |
| disabled                  | 是否禁用。                                                                                                                  | `boolean`                                                                              | -       |
| disabledEnterStartNewLine | 内部属性，供 Chat 使用：Enter 不换行，Shift + Enter 换行。                                                                  | `boolean`                                                                              | -       |
| getValueLength            | 自定义计算字符串长度，搭配 maxlength、minlength、maxCount 使用。                                                            | <ApiType detail={"(value: string) => number"}>Type</ApiType>                           | -       |
| lineNumberClassName       | 行号区域的类名。                                                                                                            | `string`                                                                               | -       |
| lineNumberStart           | 行号起始值。                                                                                                                | `number`                                                                               | `1`     |
| lineNumberStyle           | 行号区域的样式字符串。                                                                                                      | `string`                                                                               | -       |
| maxCount                  | 字数上限，设置后显示字数统计；超出时计数器标红，不截断输入。                                                                | `number`                                                                               | -       |
| maxlength                 | 最大长度。传入 getValueLength 时按它计算长度并截断，否则使用原生属性。                                                      | `number`                                                                               | -       |
| minlength                 | 最小长度。传入 getValueLength 时按它换算，否则使用原生属性。                                                                | `number`                                                                               | -       |
| onBlur                    | 焦点离开组件时调用。                                                                                                        | <ApiType detail={"(event: FocusEvent) => void"}>Type</ApiType>                         | -       |
| onChange                  | 内容变化时调用，参数为新值与事件；输入法模式下组合结束后调用一次。这是 TextArea 自己的回调，不是原生 change。               | <ApiType detail={"(value: string, event: TextAreaChangeEvent) => void"}>Type</ApiType> | -       |
| onClear                   | 点击清除按钮时调用。                                                                                                        | <ApiType detail={"(event: MouseEvent) => void"}>Type</ApiType>                         | -       |
| onCompositionEnd          | 输入法组合结束时调用。                                                                                                      | <ApiType detail={"(event: CompositionEvent) => void"}>Type</ApiType>                   | -       |
| onCompositionStart        | 输入法组合开始时调用。                                                                                                      | <ApiType detail={"(event: CompositionEvent) => void"}>Type</ApiType>                   | -       |
| onCompositionUpdate       | 输入法组合更新时调用。                                                                                                      | <ApiType detail={"(event: CompositionEvent) => void"}>Type</ApiType>                   | -       |
| onEnterPress              | 按下回车时调用（keydown），输入法组合中的回车不调用。                                                                       | <ApiType detail={"(event: KeyboardEvent) => void"}>Type</ApiType>                      | -       |
| onFocus                   | 焦点进入组件时调用。                                                                                                        | <ApiType detail={"(event: FocusEvent) => void"}>Type</ApiType>                         | -       |
| onInput                   | 原生 input 事件。                                                                                                           | <ApiType detail={"(event: Event) => void"}>Type</ApiType>                              | -       |
| onKeyDown                 | 原生 keydown 事件。                                                                                                         | <ApiType detail={"(event: KeyboardEvent) => void"}>Type</ApiType>                      | -       |
| onKeyUp                   | 原生 keyup 事件。                                                                                                           | <ApiType detail={"(event: KeyboardEvent) => void"}>Type</ApiType>                      | -       |
| onResize                  | 尺寸变化时调用：autosize 导致高度变化时参数为 border-box 高度；拖拽调整尺寸时为内容区的 height 与 width。                   | <ApiType detail={"(size: { height: number; width?: number }) => void"}>Type</ApiType>  | -       |
| placeholder               | 输入框为空时显示的提示文字。                                                                                                | `string`                                                                               | -       |
| readonly                  | 是否只读。                                                                                                                  | `boolean`                                                                              | -       |
| resize                    | 是否允许拖拽调整尺寸及方向，可选 none、both、horizontal、vertical、block、inline；仅在显式传入时生效，autosize 开启时忽略。 | `'none' \| 'both' \| 'horizontal' \| 'vertical' \| 'block' \| 'inline'`                | -       |
| rows                      | 默认行数。                                                                                                                  | `number`                                                                               | `4`     |
| showClear                 | 有内容且悬停或焦点位于组件内时展示清除按钮。                                                                                | `boolean`                                                                              | `false` |
| showCounter               | 是否显示字数统计；设置 maxCount 时总是显示。                                                                                | `boolean`                                                                              | `false` |
| showLineNumber            | 是否展示行号。                                                                                                              | `boolean`                                                                              | `false` |
| textareaStyle             | 内部 textarea 元素的样式字符串，可设置高度等。                                                                              | `string`                                                                               | -       |
| validateStatus            | 校验状态，可选 default、error、warning；error 时输出 aria-invalid。                                                         | `ValidateStatus`                                                                       | -       |
| value                     | 文本内容，可用 `bind:value` 绑定。只传值时作为初始内容，之后由组件自己维护；`bind:value={get, set}` 使其完全受控。          | `string`                                                                               | -       |

**Methods**

| 方法              | 说明                                      | 类型                                                                                                                                                   |
| ----------------- | ----------------------------------------- | ------------------------------------------------------------------------------------------------------------------------------------------------------ |
| blur              | 移出焦点。                                | <ApiType detail={"() => void"}>Type</ApiType>                                                                                                          |
| focus             | 聚焦原生 textarea。                       | <ApiType detail={"(options?: FocusOptions) => void"}>Type</ApiType>                                                                                    |
| getNativeElement  | 返回组件根元素，未渲染时为 null。         | <ApiType detail={"() => HTMLDivElement \u007c undefined"}>Type</ApiType>                                                                               |
| getTextarea       | 返回原生 textarea 元素，未渲染时为 null。 | <ApiType detail={"() => HTMLTextAreaElement \u007c undefined"}>Type</ApiType>                                                                          |
| select            | 选中输入框中的全部文本。                  | <ApiType detail={"() => void"}>Type</ApiType>                                                                                                          |
| setSelectionRange | 设置选区，参数同原生 setSelectionRange。  | <ApiType detail={"(start: number \u007c null, end: number \u007c null, direction?: 'forward' \u007c 'backward' \u007c 'none') => void"}>Type</ApiType> |
