import ApiType from '#/components/ApiType.astro';

**Props**

| Property                  | Description                                                                                                                                                                              | Type                                                                                   | Default |
| ------------------------- | ---------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------- | -------------------------------------------------------------------------------------- | ------- |
| autofocus                 | Focuses the text area on mount.                                                                                                                                                          | `boolean`                                                                              | -       |
| autosize                  | Whether the height follows the content; pass `{ minRows, maxRows }` to limit the rows.                                                                                                   | `boolean \| AutosizeRow`                                                               | `false` |
| borderless                | Borderless mode.                                                                                                                                                                         | `boolean`                                                                              | `false` |
| cols                      | Visible width in average character widths.                                                                                                                                               | `number`                                                                               | `20`    |
| composition               | Whether to enable IME mode; onChange is not called during a composition and is called once after it ends.                                                                                | `boolean`                                                                              | `false` |
| disabled                  | Whether the text area is disabled.                                                                                                                                                       | `boolean`                                                                              | -       |
| disabledEnterStartNewLine | Internal property used by Chat: Enter does not start a new line, Shift + Enter does.                                                                                                     | `boolean`                                                                              | -       |
| getValueLength            | Custom string length calculation, used with maxlength, minlength and maxCount.                                                                                                           | <ApiType detail={"(value: string) => number"}>Type</ApiType>                           | -       |
| lineNumberClassName       | Class name of the line number area.                                                                                                                                                      | `string`                                                                               | -       |
| lineNumberStart           | Starting line number.                                                                                                                                                                    | `number`                                                                               | `1`     |
| lineNumberStyle           | Style string of the line number area.                                                                                                                                                    | `string`                                                                               | -       |
| maxCount                  | Maximum number of characters; setting it shows a counter, which turns red when exceeded. The input is not truncated.                                                                     | `number`                                                                               | -       |
| maxlength                 | Maximum length. With getValueLength the length is measured and truncated by it; otherwise the native attribute applies.                                                                  | `number`                                                                               | -       |
| minlength                 | Minimum length. With getValueLength it is converted accordingly; otherwise the native attribute applies.                                                                                 | `number`                                                                               | -       |
| onBlur                    | Called when focus leaves the component.                                                                                                                                                  | <ApiType detail={"(event: FocusEvent) => void"}>Type</ApiType>                         | -       |
| onChange                  | Called when the content changes, with the new value and the event; in IME mode it is called once after the composition ends. This is the callback of TextArea, not the native change.    | <ApiType detail={"(value: string, event: TextAreaChangeEvent) => void"}>Type</ApiType> | -       |
| onClear                   | Called when the clear button is activated.                                                                                                                                               | <ApiType detail={"(event: MouseEvent) => void"}>Type</ApiType>                         | -       |
| onCompositionEnd          | Called when an IME composition ends.                                                                                                                                                     | <ApiType detail={"(event: CompositionEvent) => void"}>Type</ApiType>                   | -       |
| onCompositionStart        | Called when an IME composition starts.                                                                                                                                                   | <ApiType detail={"(event: CompositionEvent) => void"}>Type</ApiType>                   | -       |
| onCompositionUpdate       | Called when an IME composition updates.                                                                                                                                                  | <ApiType detail={"(event: CompositionEvent) => void"}>Type</ApiType>                   | -       |
| onEnterPress              | Called when Enter is pressed (keydown); the Enter that commits an IME composition does not call it.                                                                                      | <ApiType detail={"(event: KeyboardEvent) => void"}>Type</ApiType>                      | -       |
| onFocus                   | Called when focus enters the component.                                                                                                                                                  | <ApiType detail={"(event: FocusEvent) => void"}>Type</ApiType>                         | -       |
| onInput                   | The native input event.                                                                                                                                                                  | <ApiType detail={"(event: Event) => void"}>Type</ApiType>                              | -       |
| onKeyDown                 | The native keydown event.                                                                                                                                                                | <ApiType detail={"(event: KeyboardEvent) => void"}>Type</ApiType>                      | -       |
| onKeyUp                   | The native keyup event.                                                                                                                                                                  | <ApiType detail={"(event: KeyboardEvent) => void"}>Type</ApiType>                      | -       |
| onResize                  | Called when the size changes: with autosize the payload is the border-box height; when the user drags to resize it is the content-box height and width.                                  | <ApiType detail={"(size: { height: number; width?: number }) => void"}>Type</ApiType>  | -       |
| placeholder               | Hint text shown while the text area is empty.                                                                                                                                            | `string`                                                                               | -       |
| readonly                  | Whether the text area is read-only.                                                                                                                                                      | `boolean`                                                                              | -       |
| resize                    | Whether and in which direction the user may resize: none, both, horizontal, vertical, block or inline. It only takes effect when given and is ignored with autosize.                     | `'none' \| 'both' \| 'horizontal' \| 'vertical' \| 'block' \| 'inline'`                | -       |
| rows                      | Number of visible text lines.                                                                                                                                                            | `number`                                                                               | `4`     |
| showClear                 | Shows a clear button when the text area has content and the pointer or focus is inside the component.                                                                                    | `boolean`                                                                              | `false` |
| showCounter               | Whether to show the counter; it is always shown when maxCount is set.                                                                                                                    | `boolean`                                                                              | `false` |
| showLineNumber            | Whether to show line numbers.                                                                                                                                                            | `boolean`                                                                              | `false` |
| textareaStyle             | Style string of the inner textarea element, such as its height.                                                                                                                          | `string`                                                                               | -       |
| validateStatus            | Validation status: default, error or warning; error outputs aria-invalid.                                                                                                                | `ValidateStatus`                                                                       | -       |
| value                     | Content of the text area; bind it with `bind:value`. A plain value is the initial content, which the text area then maintains itself; `bind:value={get, set}` makes it fully controlled. | `string`                                                                               | -       |

**Methods**

| Method            | Description                                                                  | Type                                                                                                                                                   |
| ----------------- | ---------------------------------------------------------------------------- | ------------------------------------------------------------------------------------------------------------------------------------------------------ |
| blur              | Removes focus.                                                               | <ApiType detail={"() => void"}>Type</ApiType>                                                                                                          |
| focus             | Focuses the native textarea.                                                 | <ApiType detail={"(options?: FocusOptions) => void"}>Type</ApiType>                                                                                    |
| getNativeElement  | Returns the root element of the component, or null while it is not rendered. | <ApiType detail={"() => HTMLDivElement \u007c undefined"}>Type</ApiType>                                                                               |
| getTextarea       | Returns the native textarea element, or null while it is not rendered.       | <ApiType detail={"() => HTMLTextAreaElement \u007c undefined"}>Type</ApiType>                                                                          |
| select            | Selects all the text in the text area.                                       | <ApiType detail={"() => void"}>Type</ApiType>                                                                                                          |
| setSelectionRange | Sets the selection, with the parameters of the native setSelectionRange.     | <ApiType detail={"(start: number \u007c null, end: number \u007c null, direction?: 'forward' \u007c 'backward' \u007c 'none') => void"}>Type</ApiType> |
