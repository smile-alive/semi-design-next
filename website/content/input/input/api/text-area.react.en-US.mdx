import ApiType from '#/components/ApiType.astro';

**Props**

| Property                  | Description                                                                                                                                                                         | Type                                                                                          | Default |
| ------------------------- | ----------------------------------------------------------------------------------------------------------------------------------------------------------------------------------- | --------------------------------------------------------------------------------------------- | ------- |
| autoFocus                 | Focuses the textarea on mount.                                                                                                                                                      | `boolean`                                                                                     | -       |
| autosize                  | Whether the height follows the content. Pass `{ minRows, maxRows }` to limit the rows; `minRows` defaults to `rows`.                                                                | `boolean \| AutosizeRow`                                                                      | `false` |
| borderless                | Borderless mode.                                                                                                                                                                    | `boolean`                                                                                     | `false` |
| cols                      | Visible width in average character widths.                                                                                                                                          | `number`                                                                                      | `20`    |
| composition               | Whether to enable IME mode; `onChange` does not fire during a composition and fires once after it ends.                                                                             | `boolean`                                                                                     | `false` |
| defaultValue              | Initial content when the textarea is uncontrolled.                                                                                                                                  | `string`                                                                                      | -       |
| disabled                  | Whether the textarea is disabled.                                                                                                                                                   | `boolean`                                                                                     | -       |
| disabledEnterStartNewLine | Internal prop used by Chat: Enter does not start a new line, Shift + Enter does.                                                                                                    | `boolean`                                                                                     | -       |
| getValueLength            | Custom string length calculation, used with `maxLength`, `minLength` and `maxCount`.                                                                                                | <ApiType detail={"(value: string) => number"}>Type</ApiType>                                  | -       |
| lineNumberClassName       | Class name of the line number area.                                                                                                                                                 | `string`                                                                                      | -       |
| lineNumberStart           | Starting line number.                                                                                                                                                               | `number`                                                                                      | `1`     |
| lineNumberStyle           | Inline style of the line number area.                                                                                                                                               | `CSSProperties`                                                                               | -       |
| maxCount                  | Maximum number of characters; setting it shows a counter, which turns red when exceeded. The input is not truncated; use `maxLength` to limit input.                                | `number`                                                                                      | -       |
| onBlur                    | Called when focus leaves the textarea. It listens on the wrapper `div`, so focus moving between the textarea and the clear button does not fire it.                                 | <ApiType detail={"(e: FocusEvent) => void"}>Type</ApiType>                                    | -       |
| onChange                  | Called when the text changes, with the new value and the event.                                                                                                                     | <ApiType detail={"(value: string, e: TextAreaChangeEvent) => void"}>Type</ApiType>            | -       |
| onClear                   | Called when the clear button is clicked.                                                                                                                                            | <ApiType detail={"(e: React.MouseEvent\u003cHTMLButtonElement>) => void"}>Type</ApiType>      | -       |
| onEnterPress              | Called on Enter keydown; the Enter that commits an IME composition does not fire it.                                                                                                | <ApiType detail={"(e: React.KeyboardEvent\u003cHTMLTextAreaElement>) => void"}>Type</ApiType> | -       |
| onFocus                   | Called when the textarea gets focus. It listens on the wrapper `div`, so focus moving inside the component does not fire it again.                                                  | <ApiType detail={"(e: React.FocusEvent\u003cHTMLDivElement>) => void"}>Type</ApiType>         | -       |
| onInput                   | Called on the native `input` event, bound to the `textarea` element.                                                                                                                | <ApiType detail={"(e: React.InputEvent\u003cHTMLTextAreaElement>) => void"}>Type</ApiType>    | -       |
| onKeyDown                 | Called on keydown with the native event.                                                                                                                                            | <ApiType detail={"(e: React.KeyboardEvent\u003cHTMLTextAreaElement>) => void"}>Type</ApiType> | -       |
| onKeyUp                   | Called on keyup with the native event, bound to the `textarea` element.                                                                                                             | <ApiType detail={"(e: React.KeyboardEvent\u003cHTMLTextAreaElement>) => void"}>Type</ApiType> | -       |
| onResize                  | Called when the size changes: when `autosize` changes the height, and when the user drags the resize handle, which also provides `width`.                                           | <ApiType detail={"(data: { height: number; width?: number }) => void"}>Type</ApiType>         | -       |
| placeholder               | Hint text shown while the textarea is empty.                                                                                                                                        | `string`                                                                                      | -       |
| readonly                  | Read-only: blocks typing and hides the clear button. The native `readOnly` only blocks typing.                                                                                      | `boolean`                                                                                     | -       |
| ref                       | Imperative handle (`TextAreaHandle`); its members are in the **Methods** table.                                                                                                     | <ApiType detail={"React.Ref\u003cTextAreaHandle>"}>Type</ApiType>                             | -       |
| resize                    | Whether and in which direction the user may resize: `none`, `both`, `horizontal`, `vertical`, `block` or `inline`. It takes effect only when given, and is ignored with `autosize`. | `'none' \| 'both' \| 'horizontal' \| 'vertical' \| 'block' \| 'inline'`                       | -       |
| rows                      | Number of visible text lines.                                                                                                                                                       | `number`                                                                                      | `4`     |
| showClear                 | Shows a clear button when the textarea has content and the pointer or focus is inside the component; hidden when it is not editable.                                                | `boolean`                                                                                     | `false` |
| showCounter               | Whether to show the character counter; it is always shown when `maxCount` is set.                                                                                                   | `boolean`                                                                                     | `false` |
| showLineNumber            | Whether to show line numbers.                                                                                                                                                       | `boolean`                                                                                     | `false` |
| textareaStyle             | Inline style of the inner `textarea` element, such as its height or background color.                                                                                               | `CSSProperties`                                                                               | -       |
| validateStatus            | Validation status: `default`, `error` or `warning`; `error` sets `aria-invalid` to `true`.                                                                                          | `ValidateStatus`                                                                              | -       |
| value                     | Text content, used as a controlled value together with `onChange`.                                                                                                                  | `string`                                                                                      | -       |

**Methods**

| Method            | Description                                                                                                                             | Type                                                                                                                                                   |
| ----------------- | --------------------------------------------------------------------------------------------------------------------------------------- | ------------------------------------------------------------------------------------------------------------------------------------------------------ |
| blur              | Removes focus from the textarea.                                                                                                        | <ApiType detail={"() => void"}>Type</ApiType>                                                                                                          |
| focus             | Focuses the textarea.                                                                                                                   | <ApiType detail={"(options?: FocusOptions) => void"}>Type</ApiType>                                                                                    |
| nativeElement     | The root element of the component, the wrapper `div`. `null` after unmount.                                                             | `HTMLElement \| null`                                                                                                                                  |
| select            | Selects all the text in the textarea.                                                                                                   | <ApiType detail={"() => void"}>Type</ApiType>                                                                                                          |
| setSelectionRange | Sets the selection, with the parameters of the native `setSelectionRange`.                                                              | <ApiType detail={"(start: number \u007c null, end: number \u007c null, direction?: 'forward' \u007c 'backward' \u007c 'none') => void"}>Type</ApiType> |
| textarea          | The native `textarea` element. Each read returns the node mounted now, and `null` after unmount; toggling `showLineNumber` remounts it. | `HTMLTextAreaElement \| null`                                                                                                                          |
