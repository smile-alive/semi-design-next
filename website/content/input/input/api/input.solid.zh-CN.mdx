import ApiType from '#/components/ApiType.astro';

**Props**

| 属性                    | 说明                                                                                                                        | 类型                                                                                  | 默认值      |
| ----------------------- | --------------------------------------------------------------------------------------------------------------------------- | ------------------------------------------------------------------------------------- | ----------- |
| addonAfter              | 后置标签，可以是文本、Semi 图标或任意 JSX 元素。                                                                            | `JSX.Element`                                                                         | -           |
| addonBefore             | 前置标签，可以是文本、Semi 图标或任意 JSX 元素。                                                                            | `JSX.Element`                                                                         | -           |
| borderless              | 无边框模式。                                                                                                                | `boolean`                                                                             | `false`     |
| clearIcon               | 自定义清除按钮的图标，showClear 为 true 时有效。                                                                            | `JSX.Element`                                                                         | -           |
| composition             | 是否开启输入法模式，开启后输入法组合期间不触发 onChange，组合结束后触发一次。                                               | `boolean`                                                                             | `false`     |
| defaultValue            | 非受控时的初始内容。                                                                                                        | `string \| number`                                                                    | -           |
| disabled                | 是否禁用。未设置时使用所在 InputGroup 的 disabled。                                                                         | `boolean`                                                                             | -           |
| getValueLength          | 自定义计算字符串长度，搭配 maxLength、minLength 使用。                                                                      | <ApiType detail={"(value: string) => number"}>Type</ApiType>                          | -           |
| hideSuffix              | 清除按钮显示时隐藏后缀，默认两者并列。                                                                                      | `boolean`                                                                             | `false`     |
| inputRef                | 接收原生 input 元素。                                                                                                       | <ApiType detail={"(element: HTMLInputElement) => void"}>Type</ApiType>                | -           |
| inputStyle              | 原生 input 元素的样式，对象键使用 kebab-case，数值需带单位。                                                                | `JSX.CSSProperties \| string`                                                         | -           |
| insetLabel              | 内嵌标签，未设置 prefix 时渲染在前缀位置；此时若未设置 aria-label 与 aria-labelledby，输入框通过 aria-labelledby 由它命名。 | `JSX.Element`                                                                         | -           |
| insetLabelId            | insetLabel 所在元素的 id，供 aria-labelledby 引用。                                                                         | `string`                                                                              | -           |
| maxLength               | 最大长度。传入 getValueLength 时按它计算长度并截断，否则使用原生属性。                                                      | `number`                                                                              | -           |
| minLength               | 最小长度。传入 getValueLength 时按它换算，否则使用原生属性。                                                                | `number`                                                                              | -           |
| mode                    | 输入框模式，可选 password，显示密码显隐按钮。                                                                               | `InputMode`                                                                           | -           |
| onBlur                  | 输入框失去焦点时的回调。监听在外层容器上，焦点在输入框与组件内按钮之间移动时不触发。                                        | <ApiType detail={"(e: FocusEvent) => void"}>Type</ApiType>                            | -           |
| onChange                | 输入内容变化时的回调，参数为新值与事件。                                                                                    | <ApiType detail={"(value: string, e: InputChangeEvent) => void"}>Type</ApiType>       | -           |
| onClear                 | 点击清除按钮时的回调。                                                                                                      | <ApiType detail={"(e: MouseEvent) => void"}>Type</ApiType>                            | -           |
| onEnterPress            | 按下回车时的回调（keydown），输入法组合中的回车不触发。                                                                     | <ApiType detail={"(e: InputElementEvent\u003cKeyboardEvent>) => void"}>Type</ApiType> | -           |
| onFocus                 | 输入框获得焦点时的回调。监听在外层容器上，组件内部的焦点移动不会再次触发。                                                  | <ApiType detail={"(e: FocusEvent) => void"}>Type</ApiType>                            | -           |
| onlyBorder              | 外层容器的边框宽度，单位 px。                                                                                               | `number`                                                                              | -           |
| placeholder             | 输入框为空时显示的提示文字。                                                                                                | `string \| number`                                                                    | -           |
| prefix                  | 前缀标签，可以是文本、Semi 图标或任意 JSX 元素。                                                                            | `JSX.Element`                                                                         | -           |
| preventScroll           | 聚焦时是否阻止浏览器滚动文档以显示新聚焦的元素，作用于组件内的 focus 调用。                                                 | `boolean`                                                                             | -           |
| readonly                | 是否只读：禁止输入，且不显示清除按钮。原生 readOnly 只禁止输入。                                                            | `boolean`                                                                             | `false`     |
| ref                     | 接收外层容器 div 元素，其 `focus()` 与 `blur()` 作用于输入框。                                                              | <ApiType detail={"Ref\u003cHTMLDivElement>"}>Type</ApiType>                           | -           |
| showClear               | 有内容且悬停或焦点位于组件内时展示清除按钮。                                                                                | `boolean`                                                                             | `false`     |
| showClearIgnoreDisabled | 内部属性，供 DatePicker 使用：禁用时仍可清除。                                                                              | `boolean`                                                                             | -           |
| size                    | 输入框大小，可选 large、default、small。在 InputGroup 中未设置时取组的 size。                                               | `InputSize`                                                                           | -           |
| suffix                  | 后缀标签，可以是文本、Semi 图标或任意 JSX 元素。                                                                            | `JSX.Element`                                                                         | -           |
| type                    | 原生 input 的 type。                                                                                                        | `string`                                                                              | `"text"`    |
| validateStatus          | 校验状态，可选 default、error、warning；error 时输出 aria-invalid。                                                         | `ValidateStatus`                                                                      | `"default"` |
| value                   | 输入框内容，受控使用，配合 onChange 更新。                                                                                  | `string \| number`                                                                    | -           |
