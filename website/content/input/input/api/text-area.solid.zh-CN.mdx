import ApiType from '#/components/ApiType.astro';

**Props**

| 属性                      | 说明                                                                                                                        | 类型                                                                                     | 默认值  |
| ------------------------- | --------------------------------------------------------------------------------------------------------------------------- | ---------------------------------------------------------------------------------------- | ------- |
| autosize                  | 高度是否随内容自适应，可传 `{ minRows, maxRows }` 限制行数。                                                                | `boolean \| AutosizeRow`                                                                 | `false` |
| borderless                | 无边框模式。                                                                                                                | `boolean`                                                                                | `false` |
| cols                      | 默认列数，按平均字符宽度计。                                                                                                | `number`                                                                                 | `20`    |
| composition               | 是否开启输入法模式，开启后输入法组合期间不触发 onChange，组合结束后触发一次。                                               | `boolean`                                                                                | `false` |
| defaultValue              | 非受控时的初始内容。                                                                                                        | `string`                                                                                 | -       |
| disabled                  | 是否禁用。                                                                                                                  | `boolean`                                                                                | -       |
| disabledEnterStartNewLine | 内部属性，供 Chat 使用：Enter 不换行，Shift + Enter 换行。                                                                  | `boolean`                                                                                | -       |
| getValueLength            | 自定义计算字符串长度，搭配 maxLength、minLength、maxCount 使用。                                                            | <ApiType detail={"(value: string) => number"}>Type</ApiType>                             | -       |
| lineNumberClassName       | 行号区域的类名。                                                                                                            | `string`                                                                                 | -       |
| lineNumberStart           | 行号起始值。                                                                                                                | `number`                                                                                 | `1`     |
| lineNumberStyle           | 行号区域的样式，对象键使用 kebab-case，数值需带单位。                                                                       | `JSX.CSSProperties`                                                                      | -       |
| maxCount                  | 字数上限，设置后显示字数统计；超出时计数器标红，不截断输入。                                                                | `number`                                                                                 | -       |
| maxLength                 | 最大长度。传入 getValueLength 时按它计算长度并截断，否则使用原生属性。                                                      | `number`                                                                                 | -       |
| minLength                 | 最小长度。传入 getValueLength 时按它换算，否则使用原生属性。                                                                | `number`                                                                                 | -       |
| onBlur                    | 失去焦点时的回调。监听在外层容器上，焦点在文本框与清除按钮之间移动时不触发。                                                | <ApiType detail={"(e: FocusEvent) => void"}>Type</ApiType>                               | -       |
| onChange                  | 文本内容变化时的回调，参数为新值与事件。                                                                                    | <ApiType detail={"(value: string, e: TextAreaChangeEvent) => void"}>Type</ApiType>       | -       |
| onClear                   | 点击清除按钮时的回调。                                                                                                      | <ApiType detail={"(e: MouseEvent) => void"}>Type</ApiType>                               | -       |
| onEnterPress              | 按下回车时的回调（keydown），输入法组合中的回车不触发。                                                                     | <ApiType detail={"(e: TextAreaElementEvent\u003cKeyboardEvent>) => void"}>Type</ApiType> | -       |
| onFocus                   | 获得焦点时的回调。监听在外层容器上，组件内部的焦点移动不会再次触发。                                                        | <ApiType detail={"(e: FocusEvent) => void"}>Type</ApiType>                               | -       |
| onResize                  | 尺寸变化回调：autosize 导致高度变化时触发；用户拖拽 resize 调整尺寸时也会触发，并额外给出 width。                           | <ApiType detail={"(data: { height: number; width?: number }) => void"}>Type</ApiType>    | -       |
| readonly                  | 是否只读：禁止输入，且不显示清除按钮。原生 readOnly 只禁止输入。                                                            | `boolean`                                                                                | -       |
| ref                       | 接收外层容器 div 元素，其 `focus()` 与 `blur()` 作用于文本框。                                                              | <ApiType detail={"Ref\u003cHTMLDivElement>"}>Type</ApiType>                              | -       |
| resize                    | 是否允许拖拽调整尺寸及方向，可选 none、both、horizontal、vertical、block、inline；仅在显式传入时生效，autosize 开启时忽略。 | `'none' \| 'both' \| 'horizontal' \| 'vertical' \| 'block' \| 'inline'`                  | -       |
| rows                      | 默认行数。                                                                                                                  | `number`                                                                                 | `4`     |
| showClear                 | 有内容且悬停或焦点位于组件内时展示清除按钮。                                                                                | `boolean`                                                                                | `false` |
| showCounter               | 是否显示字数统计；设置 maxCount 时总是显示。                                                                                | `boolean`                                                                                | `false` |
| showLineNumber            | 是否展示行号。                                                                                                              | `boolean`                                                                                | `false` |
| textareaRef               | 接收原生 textarea 元素；showLineNumber 切换时 textarea 会重新挂载，并再次调用。                                             | <ApiType detail={"(element: HTMLTextAreaElement) => void"}>Type</ApiType>                | -       |
| textareaStyle             | 内部 textarea 元素的样式，如高度；对象键使用 kebab-case，数值需带单位。                                                     | `JSX.CSSProperties`                                                                      | -       |
| validateStatus            | 校验状态，可选 default、error、warning；error 时输出 aria-invalid。                                                         | `ValidateStatus`                                                                         | -       |
| value                     | 文本内容，受控使用，配合 onChange 更新。                                                                                    | `string`                                                                                 | -       |
