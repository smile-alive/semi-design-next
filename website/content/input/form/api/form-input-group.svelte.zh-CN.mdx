import ApiType from '#/components/ApiType.astro';

**Props**

| 属性              | 说明                                                                                                            | 类型                                                           | 默认值     |
| ----------------- | --------------------------------------------------------------------------------------------------------------- | -------------------------------------------------------------- | ---------- |
| aria-describedby  | Group 的 aria-describedby，会与 extraText 的 id 合并后写在 group 元素上。                                       | `string \| null`                                               | -          |
| aria-errormessage | 不生效，错误信息的 id 由 Group 自动生成并写入。                                                                 | `string \| null`                                               | -          |
| aria-labelledby   | Group 的 aria-labelledby，会与 Label 的 id 合并后写在 group 元素上。                                            | `string \| null`                                               | -          |
| children          | 组内的字段，只渲染设置了 field 的直接子项。                                                                     | `Snippet`                                                      | -          |
| disabled          | 组内字段统一的 disabled，不传时沿用 Form 的 disabled。                                                          | `boolean`                                                      | -          |
| extraText         | 额外的提示信息，常显，由组内每个控件的 aria-describedby 引用。                                                  | `Snippet`                                                      | -          |
| extraTextPosition | extraText 的位置，middle 在 Label 与 Group 之间，bottom 在 Group 之后。                                         | `'bottom' \| 'middle'`                                         | `"bottom"` |
| fieldLabel        | Group 的 Label 内容，优先于 label 属性中的文本。                                                                | `Snippet`                                                      | -          |
| label             | 整个 Group 的 Label：字符串作为文本，对象作为 Label 配置，其中 id 用作 Label 的 id；`fieldLabel` snippet 优先。 | `string \| number \| (FieldLabel & { id?: string })`           | -          |
| labelExtra        | 跟随在 Group Label 文本后的内容。                                                                               | `Snippet`                                                      | -          |
| labelPosition     | Label 的位置，默认取自 Form，可选 top 或 left。                                                                 | `'top' \| 'left'`                                              | -          |
| onBlur            | 焦点离开整个 Group 时的回调。                                                                                   | <ApiType detail={"(event: FocusEvent) => void"}>Type</ApiType> | -          |
| onFocus           | 焦点进入整个 Group 时的回调。                                                                                   | <ApiType detail={"(event: FocusEvent) => void"}>Type</ApiType> | -          |
| size              | 组内控件的尺寸。                                                                                                | `InputSize`                                                    | -          |
