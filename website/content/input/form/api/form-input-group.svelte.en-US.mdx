import ApiType from '#/components/ApiType.astro';

**Props**

| Property          | Description                                                                                                                                                        | Type                                                           | Default    |
| ----------------- | ------------------------------------------------------------------------------------------------------------------------------------------------------------------ | -------------------------------------------------------------- | ---------- |
| aria-describedby  | The aria-describedby of the Group, merged with the id of extraText on the group element.                                                                           | `string \| null`                                               | -          |
| aria-errormessage | Has no effect; the Group generates the id of the error message itself.                                                                                             | `string \| null`                                               | -          |
| aria-labelledby   | The aria-labelledby of the Group, merged with the id of the Label on the group element.                                                                            | `string \| null`                                               | -          |
| children          | The fields in the group; only direct children with a field are rendered.                                                                                           | `Snippet`                                                      | -          |
| disabled          | The disabled of the fields in the group; the disabled of the Form applies when omitted.                                                                            | `boolean`                                                      | -          |
| extraText         | Extra hint, always shown, referred to by the aria-describedby of every control in the group.                                                                       | `Snippet`                                                      | -          |
| extraTextPosition | Position of extraText: middle puts it between the Label and the Group, bottom after the Group.                                                                     | `'bottom' \| 'middle'`                                         | `"bottom"` |
| fieldLabel        | Label content of the Group, which takes precedence over the text in the label prop.                                                                                | `Snippet`                                                      | -          |
| label             | The Label of the whole Group: a string is the text, an object is a Label configuration whose id is the id of the Label; the `fieldLabel` snippet takes precedence. | `string \| number \| (FieldLabel & { id?: string })`           | -          |
| labelExtra        | Content that follows the Group Label text.                                                                                                                         | `Snippet`                                                      | -          |
| labelPosition     | Position of the Label, taken from the Form by default: top or left.                                                                                                | `'top' \| 'left'`                                              | -          |
| onBlur            | Called when focus leaves the whole Group.                                                                                                                          | <ApiType detail={"(event: FocusEvent) => void"}>Type</ApiType> | -          |
| onFocus           | Called when focus enters the whole Group.                                                                                                                          | <ApiType detail={"(event: FocusEvent) => void"}>Type</ApiType> | -          |
| size              | Size of the controls in the group.                                                                                                                                 | `InputSize`                                                    | -          |
