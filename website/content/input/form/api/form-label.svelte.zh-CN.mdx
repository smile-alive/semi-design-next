import ApiType from '#/components/ApiType.astro';

**Props**

| 属性       | 说明                                                                                                                                   | 类型                                                                                    | 默认值   |
| ---------- | -------------------------------------------------------------------------------------------------------------------------------------- | --------------------------------------------------------------------------------------- | -------- |
| align      | Label 文本的对齐方式。                                                                                                                 | `string`                                                                                | `"left"` |
| children   | Label 内容，优先于 text。                                                                                                              | `Snippet`                                                                               | -        |
| classNames | 内部各部位的类名，键为 `text`、`extra`、`tooltip`，与组件自带的类名合并。                                                              | <ApiType detail={"Partial\u003cRecord\u003cFormLabelPart, ClassValue>>"}>Type</ApiType> | -        |
| disabled   | 是否为禁用样式。                                                                                                                       | `boolean`                                                                               | -        |
| extra      | 跟随在 label 文本后的内容。                                                                                                            | `Snippet`                                                                               | -        |
| name       | 所标注控件的 id，写入 label 的 for 属性。                                                                                              | `string`                                                                                | -        |
| optional   | 是否在文本后追加语言包中的可选标识。为 true 时不再显示 required 的 \*。                                                                | `boolean`                                                                               | `false`  |
| required   | 是否显示必填的 \* 号。                                                                                                                 | `boolean`                                                                               | `false`  |
| styles     | 内部各部位的内联样式（CSS 字符串），键同 `classNames`。                                                                                | <ApiType detail={"Partial\u003cRecord\u003cFormLabelPart, string>>"}>Type</ApiType>     | -        |
| text       | Label 文本，`children` snippet 优先。                                                                                                  | `string \| number`                                                                      | -        |
| tooltip    | label 旁问号按钮的 Tooltip 内容，可以是 snippet、字符串或数字；设置后按钮在 label 元素之外，不计入字段的可访问名称，按钮名称取语言包。 | `Snippet \| string \| number`                                                           | -        |
| width      | Label 宽度，数字单位为 px。                                                                                                            | `number \| string`                                                                      | -        |
