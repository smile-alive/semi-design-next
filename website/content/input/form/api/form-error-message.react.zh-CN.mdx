import ApiType from '#/components/ApiType.astro';

**Props**

| 属性             | 说明                                                                                                                           | 类型                                                              | 默认值 |
| ---------------- | ------------------------------------------------------------------------------------------------------------------------------ | ----------------------------------------------------------------- | ------ |
| className        | 错误信息元素的类名。                                                                                                           | `string`                                                          | -      |
| error            | 错误信息。字符串与 ReactNode 直接显示，数组会用 `, ` 合并其中的错误；`true` 只显示错误态容器，没有文本；`false` 与空值不渲染。 | `ReactFieldError`                                                 | -      |
| errorMessageId   | 错误文本元素的 id，供 `aria-errormessage` 引用。                                                                               | `string`                                                          | -      |
| helpText         | 提示信息，与错误信息共用同一区块，两者都有时显示错误信息。                                                                     | `React.ReactNode`                                                 | -      |
| helpTextId       | 提示文本元素的 id，供 `aria-describedby` 引用。                                                                                | `string`                                                          | -      |
| isInInputGroup   | 是否位于 InputGroup 中；为 `true` 且 `showValidateIcon` 为 `true` 时，不论 `validateStatus` 都显示错误图标。                   | `boolean`                                                         | -      |
| ref              | 错误信息元素的引用。                                                                                                           | <ApiType detail={"React.Ref\u003cHTMLDivElement>"}>Type</ApiType> | -      |
| showValidateIcon | 是否显示 `validateStatus` 对应的图标。                                                                                         | `boolean`                                                         | -      |
| style            | 错误信息元素的内联样式。                                                                                                       | `React.CSSProperties`                                             | -      |
| validateStatus   | 信息所属的校验状态，可选 `default`、`error`、`warning`、`success`。                                                            | `string`                                                          | -      |
