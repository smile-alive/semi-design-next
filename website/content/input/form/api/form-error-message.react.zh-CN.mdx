import ApiType from '#/components/ApiType.astro';

**Props**

| 属性             | 说明                                                                                                                                       | 类型                                                              | 默认值 |
| ---------------- | ------------------------------------------------------------------------------------------------------------------------------------------ | ----------------------------------------------------------------- | ------ |
| className        | 错误信息元素的类名。                                                                                                                       | `string`                                                          | -      |
| error            | 错误信息，字符串数组，多条用 `, ` 连接显示。有错误时不显示 helpText；`['']` 表示校验失败但没有文本，不渲染内容；空数组与空值视为没有错误。 | `FieldError`                                                      | -      |
| helpText         | 提示信息，与错误信息共用同一区块，两者都有时显示错误信息。                                                                                 | `React.ReactNode`                                                 | -      |
| ref              | 错误信息元素的引用。                                                                                                                       | <ApiType detail={"React.Ref\u003cHTMLDivElement>"}>Type</ApiType> | -      |
| showValidateIcon | 是否显示 `validateStatus` 对应的图标。                                                                                                     | `boolean`                                                         | -      |
| style            | 错误信息元素的内联样式。                                                                                                                   | `React.CSSProperties`                                             | -      |
| validateStatus   | 信息所属的校验状态，可选 `default`、`error`、`warning`、`success`。                                                                        | `string`                                                          | -      |
