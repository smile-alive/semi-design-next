import ApiType from '#/components/ApiType.astro';

**Props**

| 属性 | 说明 | 类型 | 默认值 |
| --- | --- | --- | --- |
| aria-label | 滑块的可访问名称；range 模式下两个手柄在其后分别追加语言包中的起点、终点。 | `string` | - |
| aria-labelledby | 引用标签元素的 id，作为手柄的可访问名称；range 模式下每个手柄再加上起点或终点。 | `string` | - |
| aria-valuetext | 为当前值提供用户友好的名称。 | `string` | - |
| autofocus | 挂载后自动聚焦第一个滑块。 | `boolean` | - |
| defaultValue | 非受控时的初始取值，未设置时为 `min`。 | `number \| number[]` | - |
| disabled | 滑块是否禁用。 | `boolean` | `false` |
| getAriaValueText | 为滑块的当前值提供用户友好的名称，参数为该滑块的值与顺序。 | <ApiType detail={"(value: number, index?: number) => string"}>Type</ApiType> | - |
| handleDot | 滑块上的圆点，单个滑块传一个对象，range 时按手柄传数组。 | `HandleDot \| HandleDot[]` | - |
| included | marks 不为空对象时有效，true 表示刻度为包含关系，false 表示并列。 | `boolean` | `true` |
| marks | 刻度，key 为 number 且取值在闭区间 [min, max] 内，值为字符串或数字；自定义节点使用 renderMark。 | <ApiType detail={"Record\u003cnumber, string \u007c number>"}>Type</ApiType> | - |
| max | 最大值。 | `number` | `100` |
| min | 最小值。 | `number` | `0` |
| name | 原生表单提交时的字段名，落在每个手柄的 input 上；range 模式下两个 input 同名，依次提交两个值。 | `string` | - |
| onAfterChange | 值变化完成后触发，参数为当前取值；受控值被改回 `undefined` 时不触发。 | <ApiType detail={"(value: SliderProps['value']) => void"}>Type</ApiType> | - |
| onBlur | 焦点离开两个手柄时触发，挂在 `.semi-slider-wrapper` 上；在两个手柄之间移动焦点不触发。 | <ApiType detail={"JSX.EventHandlerUnion\u003cHTMLDivElement, FocusEvent>"}>Type</ApiType> | - |
| onChange | 值变化时触发，参数为当前取值。 | <ApiType detail={"(value: SliderProps['value']) => void"}>Type</ApiType> | - |
| onFocus | 任一手柄获得焦点时触发，挂在 `.semi-slider-wrapper` 上。 | <ApiType detail={"JSX.EventHandlerUnion\u003cHTMLDivElement, FocusEvent>"}>Type</ApiType> | - |
| onKeyDown | 手柄 input 上的按键回调，先于滑块处理；在回调中调用 `preventDefault` 可拦截取值键。 | <ApiType detail={"JSX.EventHandlerUnion\u003cHTMLInputElement, KeyboardEvent>"}>Type</ApiType> | - |
| onMouseUp | 拖动滑块的指针（鼠标、触摸、触控笔）松开或被取消时，以及取值键松开时触发。 | <ApiType detail={"(event: PointerEvent \u007c KeyboardEvent) => void"}>Type</ApiType> | - |
| railStyle | 滑块轨道的样式，可传对象（键为 CSS 属性名）或字符串。 | `JSX.CSSProperties \| string` | - |
| range | 是否支持两端同时可滑动。 | `boolean` | `false` |
| renderMark | 渲染刻度标签，参数为 (value, label)；开启 tooltipOnMark 时标签与气泡各调用一次。不传时显示 label。 | <ApiType detail={"(value: number, label: string \u007c number) => JSX.Element"}>Type</ApiType> | - |
| showArrow | 提示是否带箭头。 | `boolean` | `true` |
| showBoundary | 是否在悬停时展示最大值与最小值。 | `boolean` | `false` |
| showMarkLabel | 是否显示刻度标签。 | `boolean` | `true` |
| step | 步长，从 `min` 起计算。 | `number` | `1` |
| tipFormatter | 设置提示的展示格式，每个手柄单独调用，参数为该手柄的值；默认显示当前取值，`null` 隐藏提示。 | <ApiType detail={"((value: number) => any) \u007c null"}>Type</ApiType> | <ApiType detail={"(value: number) => value"}>Type</ApiType> |
| tooltipOnMark | 刻度上是否带有提示。 | `boolean` | `false` |
| tooltipVisible | 是否始终显示提示。 | `boolean` | - |
| value | 当前取值，受控使用；只传 value 而不在 onChange 中回写时，拖动、点击与方向键仅触发 onChange，滑块不移动。变回 `undefined` 时回到 defaultValue，没有则回到 min，且不触发 onAfterChange。 | `number \| number[]` | - |
| vertical | 是否为垂直方向。 | `boolean` | `false` |
| verticalReverse | 反转垂直方向，即上大下小。 | `boolean` | `false` |
