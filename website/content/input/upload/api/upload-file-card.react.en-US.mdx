import ApiType from '#/components/ApiType.astro';

**Props**

| Property             | Description                                                                                                                                      | Type                                                                                           | Default  |
| -------------------- | ------------------------------------------------------------------------------------------------------------------------------------------------ | ---------------------------------------------------------------------------------------------- | -------- |
| \_sizeInvalid        | Internal flag set by the component when the file size is out of range. Upload uses it to show the invalid size message; do not pass it manually. | `boolean`                                                                                      | -        |
| className            | Class name of the card. It is not applied to the root element.                                                                                   | `string`                                                                                       | -        |
| disabled             | Whether the card is disabled. A disabled card shows no remove, retry or replace button.                                                          | `boolean`                                                                                      | -        |
| event                | xhr event.                                                                                                                                       | `Event`                                                                                        | -        |
| fileInstance         | The File object the browser provides, which extends Blob.                                                                                        | `File`                                                                                         | -        |
| index                | Position of the file in the list.                                                                                                                | `number`                                                                                       | -        |
| listType             | Display type of the card: `list`, `picture` or `none`.                                                                                           | `UploadListType`                                                                               | `"list"` |
| name                 | File name.                                                                                                                                       | `string`                                                                                       | `""`     |
| onPreviewClick       | Called when the card is clicked. While present the card is clickable and shows a preview button; its presence changes the rendering.             | <ApiType detail={"() => void"}>Type</ApiType>                                                  | -        |
| onRemove             | Called when remove is clicked.                                                                                                                   | <ApiType detail={"() => void"}>Type</ApiType>                                                  | -        |
| onReplace            | Called when replace is clicked.                                                                                                                  | <ApiType detail={"() => void"}>Type</ApiType>                                                  | -        |
| onRetry              | Called when retry is clicked.                                                                                                                    | <ApiType detail={"() => void"}>Type</ApiType>                                                  | -        |
| percent              | Upload progress in percent.                                                                                                                      | `number`                                                                                       | -        |
| picHeight            | Height of the picture in picture mode.                                                                                                           | `string \| number`                                                                             | -        |
| picWidth             | Width of the picture in picture mode.                                                                                                            | `string \| number`                                                                             | -        |
| preview              | Whether to preview from the url.                                                                                                                 | `boolean`                                                                                      | `false`  |
| previewFile          | Custom preview that replaces the thumbnail. Only applies in list mode.                                                                           | <ApiType detail={"(fileItem: RenderFileItemProps) => ReactNode"}>Type</ApiType>                | -        |
| renderFileOperation  | Custom operation area of a list item. Only applies in list mode.                                                                                 | <ApiType detail={"(fileItem: RenderFileItemProps) => ReactNode"}>Type</ApiType>                | -        |
| renderPicClose       | Custom close button of the picture wall, with `className` and `remove`. Only applies in picture mode.                                            | <ApiType detail={"(renderPicCloseProps: RenderPictureCloseProps) => ReactNode"}>Type</ApiType> | -        |
| renderPicInfo        | Custom picture wall information. Only applies in picture mode.                                                                                   | <ApiType detail={"(renderFileItemProps: RenderFileItemProps) => ReactNode"}>Type</ApiType>     | -        |
| renderPicPreviewIcon | Custom preview icon shown on hover in the picture wall. Only applies in picture mode.                                                            | <ApiType detail={"(renderFileItemProps: RenderFileItemProps) => ReactNode"}>Type</ApiType>     | -        |
| renderThumbnail      | Custom picture wall thumbnail. Only applies in picture mode.                                                                                     | <ApiType detail={"(renderFileItemProps: RenderFileItemProps) => ReactNode"}>Type</ApiType>     | -        |
| response             | Response of the xhr. The response body on success, or the matching error on failure.                                                             | `any`                                                                                          | -        |
| shouldUpload         | Whether the file should still be uploaded.                                                                                                       | `boolean`                                                                                      | -        |
| showPicInfo          | Whether to show picture information. Only applies in picture mode.                                                                               | `boolean`                                                                                      | -        |
| showReplace          | Whether to show a replace button after success. The showReplace of the file wins.                                                                | `boolean`                                                                                      | -        |
| showRetry            | Whether to show a retry button on failure. The showRetry of the file wins.                                                                       | `boolean`                                                                                      | -        |
| showTooltip          | Whether a truncated file name shows a tooltip, or the tooltip type and the props passed to it.                                                   | `boolean \| ShowTooltip`                                                                       | `true`   |
| size                 | File size in KB.                                                                                                                                 | `string`                                                                                       | `""`     |
| status               | File status: `success`, `uploadFail`, `validateFail`, `validating`, `uploading` or `wait`.                                                       | `FileItemStatus`                                                                               | -        |
| style                | Inline style of the card.                                                                                                                        | `CSSProperties`                                                                                | -        |
| uid                  | Unique identifier of the file. Must be unique in the list.                                                                                       | `string`                                                                                       | -        |
| url                  | File URL. The picture wall shows the image from it, and previews use it.                                                                         | `string`                                                                                       | -        |
| validateMessage      | Validate message as a ReactNode. Wins over the validateMessage of the file.                                                                      | `ReactNode`                                                                                    | -        |
