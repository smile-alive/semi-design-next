import ApiType from '#/components/ApiType.astro';

**Props**

| 属性                 | 说明                                                                                        | 类型                                                                                           | 默认值   |
| -------------------- | ------------------------------------------------------------------------------------------- | ---------------------------------------------------------------------------------------------- | -------- |
| \_sizeInvalid        | 内部标记，文件大小超出限制时由组件设置，Upload 据此显示大小不合法的提示，无需手动传入。     | `boolean`                                                                                      | -        |
| className            | 文件卡片的类名，不会应用到根元素。                                                          | `string`                                                                                       | -        |
| disabled             | 是否禁用，禁用后不显示移除、重试和替换按钮。                                                | `boolean`                                                                                      | -        |
| event                | xhr 事件。                                                                                  | `Event`                                                                                        | -        |
| fileInstance         | 浏览器实际获取到的文件对象，继承自 Blob。                                                   | `File`                                                                                         | -        |
| index                | 文件在列表中的位置。                                                                        | `number`                                                                                       | -        |
| listType             | 卡片的展示类型，可选 `list`、`picture`、`none`。                                            | `UploadListType`                                                                               | `"list"` |
| name                 | 文件名。                                                                                    | `string`                                                                                       | `""`     |
| onPreviewClick       | 点击卡片时的回调。传入后卡片可点击并显示预览按钮，是否传入会改变渲染。                      | <ApiType detail={"() => void"}>Type</ApiType>                                                  | -        |
| onRemove             | 点击移除时的回调。                                                                          | <ApiType detail={"() => void"}>Type</ApiType>                                                  | -        |
| onReplace            | 点击替换时的回调。                                                                          | <ApiType detail={"() => void"}>Type</ApiType>                                                  | -        |
| onRetry              | 点击重试时的回调。                                                                          | <ApiType detail={"() => void"}>Type</ApiType>                                                  | -        |
| percent              | 上传进度百分比。                                                                            | `number`                                                                                       | -        |
| picHeight            | 照片墙模式下图片的高度。                                                                    | `string \| number`                                                                             | -        |
| picWidth             | 照片墙模式下图片的宽度。                                                                    | `string \| number`                                                                             | -        |
| preview              | 是否根据 `url` 进行预览。                                                                   | `boolean`                                                                                      | `false`  |
| previewFile          | 自定义预览逻辑，返回的内容替换原缩略图，只在 `list` 模式下有效。                            | <ApiType detail={"(fileItem: RenderFileItemProps) => ReactNode"}>Type</ApiType>                | -        |
| renderFileOperation  | 自定义列表项的操作区，只在 `list` 模式下有效。                                              | <ApiType detail={"(fileItem: RenderFileItemProps) => ReactNode"}>Type</ApiType>                | -        |
| renderPicClose       | 自定义照片墙的关闭按钮，参数为 `className` 和 `remove`，只在照片墙模式下有效。              | <ApiType detail={"(renderPicCloseProps: RenderPictureCloseProps) => ReactNode"}>Type</ApiType> | -        |
| renderPicInfo        | 自定义照片墙信息，只在照片墙模式下有效。                                                    | <ApiType detail={"(renderFileItemProps: RenderFileItemProps) => ReactNode"}>Type</ApiType>     | -        |
| renderPicPreviewIcon | 自定义照片墙悬停时的预览图标，只在照片墙模式下有效。                                        | <ApiType detail={"(renderFileItemProps: RenderFileItemProps) => ReactNode"}>Type</ApiType>     | -        |
| renderThumbnail      | 自定义照片墙缩略图，只在照片墙模式下有效。                                                  | <ApiType detail={"(renderFileItemProps: RenderFileItemProps) => ReactNode"}>Type</ApiType>     | -        |
| response             | xhr 的响应。请求成功时为响应体，失败时为对应的错误。                                        | `any`                                                                                          | -        |
| shouldUpload         | 是否应该继续上传。                                                                          | `boolean`                                                                                      | -        |
| showPicInfo          | 是否显示图片信息，只在照片墙模式下有效。                                                    | `boolean`                                                                                      | -        |
| showReplace          | 上传成功后是否展示替换按钮，文件自身的 `showReplace` 优先。                                 | `boolean`                                                                                      | -        |
| showRetry            | 上传失败时是否展示重试按钮，文件自身的 `showRetry` 优先。                                   | `boolean`                                                                                      | -        |
| showTooltip          | 文件名过长时是否展示提示，或配置提示的类型和透传属性。                                      | `boolean \| ShowTooltip`                                                                       | `true`   |
| size                 | 文件大小，单位 KB。                                                                         | `string`                                                                                       | `""`     |
| status               | 文件状态，可选 `success`、`uploadFail`、`validateFail`、`validating`、`uploading`、`wait`。 | `FileItemStatus`                                                                               | -        |
| style                | 文件卡片的内联样式。                                                                        | `CSSProperties`                                                                                | -        |
| uid                  | 文件唯一标识符，列表内必须唯一。                                                            | `string`                                                                                       | -        |
| url                  | 文件地址，图片墙用它展示图片，预览时使用。                                                  | `string`                                                                                       | -        |
| validateMessage      | 校验信息，ReactNode，优先于文件自身的 `validateMessage`。                                   | `ReactNode`                                                                                    | -        |
