import ApiType from '#/components/ApiType.astro';

**Props**

| Property     | Description                                                                                                                                                     | Type                                                                            | Default      |
| ------------ | --------------------------------------------------------------------------------------------------------------------------------------------------------------- | ------------------------------------------------------------------------------- | ------------ |
| children     | Checkboxes inside the group. Used when `options` is not set.                                                                                                    | `React.ReactNode`                                                               | -            |
| className    | Class name of the root element.                                                                                                                                 | `string`                                                                        | -            |
| defaultValue | Values selected by default when uncontrolled, matched against each Checkbox's `value`.                                                                          | `CheckboxValue[]`                                                               | `[]`         |
| direction    | Layout direction inside the group: `horizontal` or `vertical`.                                                                                                  | `CheckboxDirection`                                                             | `"vertical"` |
| disabled     | Whether the whole group is disabled.                                                                                                                            | `boolean`                                                                       | `false`      |
| id           | Id of the root element.                                                                                                                                         | `string`                                                                        | -            |
| name         | The `name` of every native input in the group. A unique value is generated per group when omitted.                                                              | `string`                                                                        | -            |
| onChange     | Called when the selected values change, with the new array of values.                                                                                           | <ApiType detail={"(value: CheckboxValue[]) => void"}>Type</ApiType>             | -            |
| options      | Declares the options as data. Each item is a string or `{ label, value, disabled, extra, style, className, onChange }`; `label` and `extra` accept `ReactNode`. | <ApiType detail={"Array\u003cstring \u007c CheckboxOptionItem>"}>Type</ApiType> | -            |
| prefixCls    | CSS class name prefix.                                                                                                                                          | `string`                                                                        | -            |
| ref          | Reference to the root element (`role="group"`).                                                                                                                 | <ApiType detail={"React.Ref\u003cHTMLDivElement>"}>Type</ApiType>               | -            |
| style        | Style of the root element.                                                                                                                                      | `React.CSSProperties`                                                           | -            |
| type         | Style type of every Checkbox in the group: `default`, `card` or `pureCard`.                                                                                     | `CheckboxType`                                                                  | `"default"`  |
| value        | Selected values. Passing it makes the group controlled.                                                                                                         | `CheckboxValue[]`                                                               | -            |
