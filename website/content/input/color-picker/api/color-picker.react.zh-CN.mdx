import ApiType from '#/components/ApiType.astro';

**Props**

| 属性 | 说明 | 类型 | 默认值 |
| --- | --- | --- | --- |
| alpha | 是否开启透明度选择。 | `boolean` | `true` |
| bottomSlot | 底部渲染额外元素。 | `React.ReactNode` | - |
| children | 使用 usePopover 时的触发器，默认是一个显示当前颜色的色块按钮；不使用 usePopover 时不渲染。 | `React.ReactNode` | - |
| className | picker 根元素的类名；使用 usePopover 时位于弹层内。 | `string` | - |
| defaultFormat | 手动输入时默认使用的颜色格式，可选 `rgba`、`hex`、`hsva`。 | `ColorFormat` | `"hex"` |
| defaultValue | 非受控的初始值，是同时包含 hsva、hex、rgba 三种格式的 ColorValue 对象。 | `ColorValue` | <ApiType detail={"{\n hsva: { h: 176, s: 71, v: 77, a: 1 },\n rgba: { r: 57, g: 197, b: 187, a: 1 },\n hex: '#39c5bb',\n }"}>Type</ApiType> |
| eyeDropper | 是否开启滴管拾色器，需要 HTTPS 或 localhost 等安全 context，且浏览器为 Chromium 95 以上，否则无效果。 | `boolean` | `true` |
| height | 高度，单位为像素，未设置时为 280。 | `number` | - |
| onChange | 用户选中颜色时的回调，参数为 ColorValue，包含 hsva、hex、rgba 三种格式。 | <ApiType detail={"(value: ColorValue) => void"}>Type</ApiType> | - |
| popoverProps | 放入 Popover 时传给 Popover 的属性，如 `trigger`、`aria-label`、`aria-labelledby`。弹层默认以点击触发、role 为 `dialog`，名称默认取语言包；`aria-label` 或 `aria-labelledby` 优先。 | `PopoverProps` | - |
| ref | picker 根元素（`div`）的引用；使用 usePopover 时指向弹层内的 picker 元素。 | <ApiType detail={"React.Ref\u003cHTMLDivElement>"}>Type</ApiType> | - |
| style | picker 根元素的样式；使用 usePopover 时位于弹层内。 | `React.CSSProperties` | - |
| topSlot | 顶部渲染额外元素。 | `React.ReactNode` | - |
| usePopover | 是否放入 Popover 渲染，触发器使用 `children`，默认 `false`。 | `boolean` | - |
| value | 受控值，形态同 defaultValue。传 `undefined` 视为非受控，从受控变回 `undefined` 时显示回到 defaultValue；传 `null` 是受控的空值，显示 defaultValue 且只报告变化。ColorValue 按不可变值对待，更新时整体替换。 | `ColorValue \| null` | - |
| width | 宽度，单位为像素，未设置时为 280。 | `number` | - |
