import ApiType from '#/components/ApiType.astro';

**Props**

| Property     | Description                                                                                                                                                                                            | Type                                                                     | Default      |
| ------------ | ------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------ | ------------------------------------------------------------------------ | ------------ |
| bordered     | Whether to show a border.                                                                                                                                                                              | `boolean`                                                                | `false`      |
| dataSource   | Data source of the list; each entry is rendered by renderItem.                                                                                                                                         | `T[]`                                                                    | -            |
| emptyContent | Content shown when there is no data; the locale's empty text by default.                                                                                                                               | `JSX.Element`                                                            | -            |
| footer       | List footer.                                                                                                                                                                                           | `JSX.Element`                                                            | -            |
| grid         | Grid configuration: align, gutter, justify and type of Row, and span, offset, order, push, pull, the responsive breakpoints, class and style of Col. class and style land on the Col around each item. | `Grid`                                                                   | -            |
| header       | List header.                                                                                                                                                                                           | `JSX.Element`                                                            | -            |
| layout       | List layout: vertical or horizontal.                                                                                                                                                                   | `'vertical' \| 'horizontal'`                                             | `"vertical"` |
| loading      | Whether the list is loading; a Spin is shown when true.                                                                                                                                                | `boolean`                                                                | `false`      |
| loadMore     | Load-more button.                                                                                                                                                                                      | `JSX.Element`                                                            | -            |
| onClick      | Click handler for the items that have no onClick of their own; an item with a handler renders a native button.                                                                                         | `ListItemEventHandler`                                                   | -            |
| onRightClick | Context-menu handler for the items that have no onRightClick of their own; those items also render a native button.                                                                                    | `ListItemEventHandler`                                                   | -            |
| renderItem   | Renders each entry of dataSource, receiving the entry and its index and returning JSX, usually a List.Item.                                                                                            | <ApiType detail={"(item: T, ind: number) => JSX.Element"}>Type</ApiType> | -            |
| size         | List size: small, default or large.                                                                                                                                                                    | `Size`                                                                   | `"default"`  |
| split        | Whether to show dividers between items.                                                                                                                                                                | `boolean`                                                                | `true`       |
