import ApiType from '#/components/ApiType.astro';

**Props**

| 属性         | 说明                                                                                                                                                      | 类型                                                                     | 默认值       |
| ------------ | --------------------------------------------------------------------------------------------------------------------------------------------------------- | ------------------------------------------------------------------------ | ------------ |
| bordered     | 是否显示边框。                                                                                                                                            | `boolean`                                                                | `false`      |
| dataSource   | 列表数据源，每一项通过 renderItem 渲染。                                                                                                                  | `T[]`                                                                    | -            |
| emptyContent | 没有数据时的展示内容，缺省为语言包中的空文案。                                                                                                            | `JSX.Element`                                                            | -            |
| footer       | 列表底部。                                                                                                                                                | `JSX.Element`                                                            | -            |
| grid         | 栅格配置：Row 的 align、gutter、justify、type 与 Col 的 span、offset、order、push、pull、响应式断点、class、style，class 与 style 落在每项外层的 Col 上。 | `Grid`                                                                   | -            |
| header       | 列表头部。                                                                                                                                                | `JSX.Element`                                                            | -            |
| layout       | 列表布局，可选 vertical 或 horizontal。                                                                                                                   | `'vertical' \| 'horizontal'`                                             | `"vertical"` |
| loading      | 是否处于加载中，为 true 时显示 Spin。                                                                                                                     | `boolean`                                                                | `false`      |
| loadMore     | 加载更多的按钮。                                                                                                                                          | `JSX.Element`                                                            | -            |
| onClick      | 列表项的点击回调，作用于每个没有自己 onClick 的项；有处理函数的项渲染为原生按钮。                                                                         | `ListItemEventHandler`                                                   | -            |
| onRightClick | 列表项的右键（上下文菜单）回调，作用于每个没有自己 onRightClick 的项，这些项同样渲染为原生按钮。                                                          | `ListItemEventHandler`                                                   | -            |
| renderItem   | 渲染 dataSource 中的每一项，接收数据项与下标，返回 JSX 元素，通常是 List.Item。                                                                           | <ApiType detail={"(item: T, ind: number) => JSX.Element"}>Type</ApiType> | -            |
| size         | 列表尺寸，可选 small、default 或 large。                                                                                                                  | `Size`                                                                   | `"default"`  |
| split        | 是否展示分割线。                                                                                                                                          | `boolean`                                                                | `true`       |
