import ApiType from '#/components/ApiType.astro';

**Props**

| 属性              | 说明                                                                                                                                                                          | 类型                                                                               | 默认值     |
| ----------------- | ----------------------------------------------------------------------------------------------------------------------------------------------------------------------------- | ---------------------------------------------------------------------------------- | ---------- |
| activeIndex       | 当前展示的索引，可用 `bind:activeIndex` 绑定；传 undefined 时交还控制权，从最后一次受控的索引继续。                                                                           | `number`                                                                           | -          |
| animation         | 切换动画，可选 slide 或 fade。                                                                                                                                                | `'slide' \| 'fade'`                                                                | `"slide"`  |
| arrowProps        | 上一项、下一项箭头按钮的数据，`leftArrow.props`、`rightArrow.props` 放按钮属性；`onclick` 与 `class` 会和内置处理、内置类名合并。图标使用 `leftArrow`、`rightArrow` snippet。 | `ArrowProps`                                                                       | -          |
| arrowType         | 箭头展示时机，可选 always 或 hover。                                                                                                                                          | `'always' \| 'hover'`                                                              | `"always"` |
| autoPlay          | 是否自动循环播放，也可传入 interval（切换间隔，默认 2000 毫秒）和 hoverToPause（悬浮时是否暂停，默认 true）。                                                                 | `CarouselAutoPlay`                                                                 | `true`     |
| indicatorPosition | 指示器位置，可选 left、center 或 right。                                                                                                                                      | `'left' \| 'center' \| 'right'`                                                    | `"center"` |
| indicatorSize     | 指示器尺寸，可选 small 或 medium。                                                                                                                                            | `'small' \| 'medium'`                                                              | `"small"`  |
| indicatorType     | 指示器类型，可选 dot、line 或 columnar。                                                                                                                                      | `'dot' \| 'line' \| 'columnar'`                                                    | `"dot"`    |
| items             | 幻灯片的数据，每项一张，由 `slide` snippet 渲染；数据量决定幻灯片数，首次渲染前就已知。                                                                                       | `readonly T[]`                                                                     | -          |
| leftArrow         | 上一项箭头图标的 snippet；传入空 snippet 时按钮内容为空。                                                                                                                     | `Snippet`                                                                          | -          |
| onChange          | 切换到新的一项时调用，参数为新索引和上一个索引。                                                                                                                              | <ApiType detail={"(activeIndex: number, preIndex: number) => void"}>Type</ApiType> | -          |
| onfocusin         | 焦点进入轮播时调用；同时暂停自动播放。                                                                                                                                        | <ApiType detail={"FocusEventHandler\u003cT> \u007c null"}>Type</ApiType>           | -          |
| onfocusout        | 焦点离开轮播时调用；同时恢复自动播放。                                                                                                                                        | <ApiType detail={"FocusEventHandler\u003cT> \u007c null"}>Type</ApiType>           | -          |
| onmouseenter      | 指针进入轮播时调用；`hoverToPause` 为 true 时同时暂停自动播放。                                                                                                               | <ApiType detail={"MouseEventHandler\u003cT> \u007c null"}>Type</ApiType>           | -          |
| onmouseleave      | 指针离开轮播时调用；同时恢复自动播放。                                                                                                                                        | <ApiType detail={"MouseEventHandler\u003cT> \u007c null"}>Type</ApiType>           | -          |
| rightArrow        | 下一项箭头图标的 snippet；传入空 snippet 时按钮内容为空。                                                                                                                     | `Snippet`                                                                          | -          |
| showArrow         | 是否展示箭头。                                                                                                                                                                | `boolean`                                                                          | `true`     |
| showIndicator     | 是否展示指示器。                                                                                                                                                              | `boolean`                                                                          | `true`     |
| slide             | 一张幻灯片内容的 snippet，参数为 `{ item, index }`；Carousel 给每张包一层带样式类、role 与 `inert` 的元素。                                                                   | <ApiType detail={"Snippet\u003c[CarouselSlideContext\u003cT>]>"}>Type</ApiType>    | -          |
| slideDirection    | 动画为 slide 时的滑动方向，可选 left 或 right。                                                                                                                               | `'left' \| 'right'`                                                                | `"left"`   |
| speed             | 切换速度，单位为毫秒。                                                                                                                                                        | `number`                                                                           | `300`      |
| theme             | 指示器和箭头的主题，可选 primary、light 或 dark。                                                                                                                             | `'dark' \| 'primary' \| 'light'`                                                   | `"light"`  |
| trigger           | 指示器触发切换的时机，可选 click 或 hover。                                                                                                                                   | `'click' \| 'hover'`                                                               | `"click"`  |

**Methods**

| 方法 | 说明                                         | 类型                                                             |
| ---- | -------------------------------------------- | ---------------------------------------------------------------- |
| goTo | 切换到指定索引，超出范围时循环。             | <ApiType detail={"(targetIndex: number) => void"}>Type</ApiType> |
| next | 切换到下一项。                               | <ApiType detail={"() => void"}>Type</ApiType>                    |
| play | 开始自动播放，autoPlay 为 false 时同样生效。 | <ApiType detail={"() => void"}>Type</ApiType>                    |
| prev | 切换到上一项。                               | <ApiType detail={"() => void"}>Type</ApiType>                    |
| stop | 停止自动播放。                               | <ApiType detail={"() => void"}>Type</ApiType>                    |
