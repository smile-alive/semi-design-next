import ApiType from '#/components/ApiType.astro';

**Props**

| Property        | Description                                                                                                                                                                                                   | Type                                                                | Default |
| --------------- | ------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------- | ------------------------------------------------------------------- | ------- |
| alt             | Image description, also the accessible name of a previewable thumbnail.                                                                                                                                       | `string`                                                            | -       |
| children        | Not rendered by the component, so setting it has no effect.                                                                                                                                                   | `ReactNode`                                                         | -       |
| className       | Class name of the thumbnail root element.                                                                                                                                                                     | `string`                                                            | -       |
| crossOrigin     | The crossOrigin passed to the native img element, `anonymous` or `use-credentials`. When set, the preview image uses the same value; otherwise it uses `preview.crossOrigin`.                                 | `'anonymous' \| 'use-credentials'`                                  | -       |
| fallback        | Content shown when loading fails: a string is the fallback image URL, and a React element is shown as it is. Without it, the default failure icon is shown.                                                   | `string \| ReactNode`                                               | -       |
| height          | Display height of the image; a number is in px.                                                                                                                                                               | `string \| number`                                                  | -       |
| imgCls          | Custom class name passed to the img element.                                                                                                                                                                  | `string`                                                            | -       |
| imgStyle        | Custom style passed to the img element.                                                                                                                                                                       | `React.CSSProperties`                                               | -       |
| onClick         | Called when the image is clicked; bound to the inner img element.                                                                                                                                             | <ApiType detail={"(event: any) => void"}>Type</ApiType>             | -       |
| placeholder     | Placeholder content shown until the image has loaded; without it, a skeleton is shown.                                                                                                                        | `ReactNode`                                                         | -       |
| preview         | Preview options; false disables the preview. The object accepts the ImagePreview props except lazyLoad, lazyLoadMargin, children, className and style (use previewCls and previewStyle to style the preview). | `boolean \| ImagePreviewProps`                                      | `true`  |
| ref             | Reference to the inner img element.                                                                                                                                                                           | <ApiType detail={"React.Ref\u003cHTMLImageElement>"}>Type</ApiType> | -       |
| setDownloadName | Sets the download file name: receives the image URL and returns the file name.                                                                                                                                | <ApiType detail={"(src: string) => string"}>Type</ApiType>          | -       |
| src             | Image source URL.                                                                                                                                                                                             | `string`                                                            | -       |
| style           | Style of the thumbnail root element.                                                                                                                                                                          | `CSSProperties`                                                     | -       |
| width           | Display width of the image; a number is in px.                                                                                                                                                                | `string \| number`                                                  | -       |
