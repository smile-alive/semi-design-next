import ApiType from '#/components/ApiType.astro';

**Props**

| 属性                | 说明                                                                                                                                        | 类型                                                                                   | 默认值                  |
| ------------------- | ------------------------------------------------------------------------------------------------------------------------------------------- | -------------------------------------------------------------------------------------- | ----------------------- |
| adaptiveTip         | 适应页面操作按钮的提示。                                                                                                                    | `string`                                                                               | -                       |
| children            | 要预览的 Image 组件。                                                                                                                       | `JSX.Element`                                                                          | -                       |
| closable            | 是否显示关闭按钮。                                                                                                                          | `boolean`                                                                              | `true`                  |
| closeOnEsc          | 按下 Esc 键时是否关闭预览。                                                                                                                 | `boolean`                                                                              | -                       |
| crossOrigin         | 透传给预览图片的原生 img 标签的 crossorigin。                                                                                               | `'anonymous' \| 'use-credentials'`                                                     | -                       |
| currentIndex        | 受控属性，当前预览图片下标。                                                                                                                | `number`                                                                               | -                       |
| defaultCurrentIndex | 首次展示图片下标。                                                                                                                          | `number`                                                                               | -                       |
| defaultVisible      | 首次是否开启预览。                                                                                                                          | `boolean`                                                                              | -                       |
| disableDownload     | 禁用下载；没有当前图片（src 为空列表）时下载按钮同样禁用。                                                                                  | `boolean`                                                                              | -                       |
| downloadTip         | 下载操作按钮的提示。                                                                                                                        | `string`                                                                               | -                       |
| getPopupContainer   | 指定预览渲染的父级 DOM，返回 `document.body` 与不传相同。容器不是 `document.body` 时预览为非模态，自定义容器需要设置 `position: relative`。 | <ApiType detail={"() => HTMLElement"}>Type</ApiType>                                   | -                       |
| infinite            | 是否无限循环切换。                                                                                                                          | `boolean`                                                                              | -                       |
| initialZoom         | 预览图片初始缩放倍数，仅在首次打开或切换到该图片时生效一次，会被 minZoom 与 maxZoom 限制。                                                  | `number`                                                                               | -                       |
| lazyLoad            | 是否开启懒加载，相对视口判断；设了 overflow 的滚动祖先仍会裁剪。                                                                            | `boolean`                                                                              | `true`                  |
| lazyLoadMargin      | 传给 IntersectionObserver 的 rootMargin，相对视口计算。                                                                                     | `string`                                                                               | `"0px 100px 100px 0px"` |
| maskClosable        | 点击遮罩是否可关闭预览。                                                                                                                    | `boolean`                                                                              | -                       |
| maxZoom             | 预览图片最大缩放倍数，如 `5` 表示 500%。                                                                                                    | `number`                                                                               | -                       |
| minZoom             | 预览图片最小缩放倍数，如 `0.1` 表示 10%。                                                                                                   | `number`                                                                               | -                       |
| nextTip             | 下一张操作按钮的提示，同时作为读屏名称；未设置时取语言包的文案。                                                                            | `string`                                                                               | -                       |
| onChange            | 切换图片时触发，参数为新的图片下标。                                                                                                        | <ApiType detail={"(index: number) => void"}>Type</ApiType>                             | -                       |
| onClose             | 点击关闭按钮或按 Esc 关闭预览时触发；点击遮罩关闭只触发 onVisibleChange。                                                                   | <ApiType detail={"() => void"}>Type</ApiType>                                          | -                       |
| onDownload          | 下载图片时触发，参数为图片地址与下标。                                                                                                      | <ApiType detail={"(src: string, index: number) => void"}>Type</ApiType>                | -                       |
| onDownloadError     | 图片下载失败时触发，参数为图片地址。                                                                                                        | <ApiType detail={"(src: string) => void"}>Type</ApiType>                               | -                       |
| onNext              | 向后切换图片时触发，参数为切换后的下标。                                                                                                    | <ApiType detail={"(index: number) => void"}>Type</ApiType>                             | -                       |
| onPrev              | 向前切换图片时触发，参数为切换后的下标。                                                                                                    | <ApiType detail={"(index: number) => void"}>Type</ApiType>                             | -                       |
| onRatioChange       | 切换原始尺寸与适应页面时触发，参数为 `'adaptation'` 或 `'realSize'`。                                                                       | <ApiType detail={"(type: RatioType) => void"}>Type</ApiType>                           | -                       |
| onRotateLeft        | 逆时针旋转图片时触发，参数为当前旋转角度。                                                                                                  | <ApiType detail={"(angle: number) => void"}>Type</ApiType>                             | -                       |
| onVisibleChange     | 可见状态变化时触发，参数为是否可见。                                                                                                        | <ApiType detail={"(visible: boolean) => void"}>Type</ApiType>                          | -                       |
| onZoomIn            | 图片放大时触发，参数为放大后的缩放倍数。                                                                                                    | <ApiType detail={"(zoom: number) => void"}>Type</ApiType>                              | -                       |
| onZoomOut           | 图片缩小时触发，参数为缩小后的缩放倍数。                                                                                                    | <ApiType detail={"(zoom: number) => void"}>Type</ApiType>                              | -                       |
| originTip           | 原始尺寸操作按钮的提示。                                                                                                                    | `string`                                                                               | -                       |
| preLoad             | 是否开启预加载。                                                                                                                            | `boolean`                                                                              | -                       |
| preLoadGap          | 预加载的步长。                                                                                                                              | `number`                                                                               | -                       |
| previewCls          | 自定义预览层的类名。                                                                                                                        | `string`                                                                               | -                       |
| previewStyle        | 自定义预览层的样式，可以是样式对象（键用 kebab-case）或字符串。                                                                             | `JSX.CSSProperties \| string`                                                          | -                       |
| previewTitle        | 预览顶部标题，同时作为预览对话框的读屏名称；组内图片的 preview.previewTitle 优先。                                                          | `JSX.Element`                                                                          | -                       |
| prevTip             | 上一张操作按钮的提示，同时作为读屏名称；未设置时取语言包的文案。                                                                            | `string`                                                                               | -                       |
| renderCloseIcon     | 自定义关闭图标，可以是 JSX 元素，或返回 JSX 元素的函数。                                                                                    | <ApiType detail={"JSX.Element \u007c (() => JSX.Element)"}>Type</ApiType>              | -                       |
| renderHeader        | 自定义渲染预览顶部信息，接收当前标题，返回 JSX 元素。                                                                                       | <ApiType detail={"(title: JSX.Element) => JSX.Element"}>Type</ApiType>                 | -                       |
| renderLeftIcon      | 自定义向左切换图标，可以是 JSX 元素，或接收当前下标并返回 JSX 元素的函数。                                                                  | <ApiType detail={"JSX.Element \u007c ((index: number) => JSX.Element)"}>Type</ApiType> | -                       |
| renderPreviewMenu   | 自定义渲染预览底部操作区，接收 MenuProps，返回 JSX 元素。                                                                                   | <ApiType detail={"(props: MenuProps) => JSX.Element"}>Type</ApiType>                   | -                       |
| renderRightIcon     | 自定义向右切换图标，可以是 JSX 元素，或接收当前下标并返回 JSX 元素的函数。                                                                  | <ApiType detail={"JSX.Element \u007c ((index: number) => JSX.Element)"}>Type</ApiType> | -                       |
| rotateTip           | 旋转操作按钮的提示。                                                                                                                        | `string`                                                                               | -                       |
| setDownloadName     | 设置图片下载文件名，接收图片地址并返回文件名。                                                                                              | <ApiType detail={"(src: string) => string"}>Type</ApiType>                             | -                       |
| showTooltip         | 是否展示底部操作区的提示。                                                                                                                  | `boolean`                                                                              | -                       |
| src                 | 要预览的图片地址列表。包裹 Image 时与各 Image 登记的预览图合并，src 中的图片排在前面。                                                      | `string \| string[]`                                                                   | `[]`                    |
| viewerVisibleDelay  | 隐藏预览操作按钮前的无操作时长，单位为毫秒。                                                                                                | `number`                                                                               | -                       |
| visible             | 受控属性，是否预览。                                                                                                                        | `boolean`                                                                              | -                       |
| zIndex              | 预览层层级。                                                                                                                                | `number`                                                                               | -                       |
| zoomInTip           | 放大操作按钮的提示。                                                                                                                        | `string`                                                                               | -                       |
| zoomOutTip          | 缩小操作按钮的提示。                                                                                                                        | `string`                                                                               | -                       |
| zoomStep            | 每次缩小或放大的倍数，如 `0.1` 表示每次 10%。                                                                                               | `number`                                                                               | -                       |
