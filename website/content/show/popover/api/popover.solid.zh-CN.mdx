import ApiType from '#/components/ApiType.astro';

**Props**

| 属性 | 说明 | 类型 | 默认值 |
| --- | --- | --- | --- |
| afterClose | 弹出层关闭且离场动画结束后触发。 | <ApiType detail={"() => void"}>Type</ApiType> | - |
| afterOpen | 弹出层打开且进场动画结束后触发，进场结束前被关闭的打开不触发。 | `TooltipProps['afterOpen']` | - |
| aria-label | 弹出层 role 为 dialog 时的名称。 | `string` | - |
| aria-labelledby | 弹出层 role 为 dialog 时，指向名称元素的 id。 | `string` | - |
| arrowPointAtCenter | 小三角是否指向触发元素中心，需要同时开启 `showArrow`。未设置时为 true。 | `boolean` | - |
| arrowStyle | 小三角的 `borderColor`、`backgroundColor` 与 `borderOpacity`（驼峰键名的对象，不是 Solid 的 style 对象）；缺省跟随对象形式 `style` 中的 `background-color` 与 `border-color`。 | `ArrowStyle` | - |
| autoAdjustOverflow | 弹出层被遮挡时是否自动调整方向。 | `boolean` | `true` |
| children | 触发元素。传入以 `getTriggerProps` 为参数的函数时，把 `trigger()` 展开到接收焦点的元素上；其他内容会被包在 Tooltip 的 span 里作为触发元素。 | `TooltipProps['children']` | - |
| class | 弹出层的类名。 | `string` | - |
| clickToHide | 点击弹出层及其内部任一元素时是否关闭弹出层。 | `TooltipProps['clickToHide']` | - |
| closeOnEsc | 弹出层打开期间按 Esc 是否关闭弹出层；trigger 为 custom 时只调用 `onEscKeyDown`，`visible` 受控时只调用 `onVisibleChange(false)`，使用方回写后才关闭。 | `TooltipProps['closeOnEsc']` | `true` |
| condition | 是否允许触发显示。仅显式设为 false 时，hover、click、focus 等触发行为不生效，trigger 为 custom 时仍按 `visible` 显示；由 true 变为 false 时立即关闭已显示的弹出层，`visible` 受控时只调用 `onVisibleChange(false)`，使用方回写后才关闭。 | `boolean` | - |
| content | 弹出层内容。可传 JSX，也可传函数；函数的入参是 `{ initialFocusRef }`，把 `initialFocusRef` 作为 `ref` 绑定到可聚焦元素上，打开时会自动聚焦在该位置。 | `TooltipProps['content']` | - |
| contentClassName | 弹出层内容卡片的类名。 | `ClassValue` | - |
| disableArrowKeyDown | 是否禁用在触发元素上按方向键把焦点移入弹出层。 | `TooltipProps['disableArrowKeyDown']` | - |
| disableFocusListener | trigger 为 hover 时，是否不响应键盘聚焦打开弹出层。 | `boolean` | - |
| guardFocus | 焦点位于弹出层内时，Tab 是否在弹出层内循环。 | `TooltipProps['guardFocus']` | `true` |
| keepDOM | 关闭时是否保留弹出层内容而不销毁。 | `boolean` | - |
| motion | 是否展示弹出层动画。 | `boolean` | `true` |
| mouseEnterDelay | 指针移入后延迟显示的毫秒数，仅 trigger 为 hover 或 focus 时生效。未设置时为 50。 | `number` | - |
| mouseLeaveDelay | 指针移出后延迟隐藏的毫秒数，仅 trigger 为 hover 或 focus 时生效，不小于 mouseEnterDelay。未设置时为 50。 | `number` | - |
| onClickOutSide | 弹出层处于展示状态时，点击触发元素和弹出层之外区域的回调，参数是原生 `MouseEvent`；仅 trigger 为 custom 或 click 时有效。 | `TooltipProps['onClickOutSide']` | - |
| onEscKeyDown | 弹出层打开期间按 Esc 时调用，参数是原生 `KeyboardEvent`。 | `TooltipProps['onEscKeyDown']` | - |
| onOpen | 弹出层显示、进场动画开始时调用，每次打开一次，早于 `afterOpen`。 | `TooltipProps['onOpen']` | - |
| onVisibleChange | 弹出层显示或隐藏时的回调；`visible` 受控时只报告用户交互发起的请求，使用方自己改变 `visible` 不触发。 | <ApiType detail={"(visible: boolean) => void"}>Type</ApiType> | - |
| position | 弹出层相对触发元素的位置，共十二个方向。 | `Position` | `"bottom"` |
| prefixCls | 弹出层类名前缀，设置后弹出层不再带 Popover 的样式。 | `string` | `"semi-popover"` |
| ref | 接收 Popover 实例的方法，见下方 Methods。 | <ApiType detail={"Ref\u003cPopoverHandle>"}>Type</ApiType> | - |
| returnFocusOnClose | 按 Esc 关闭弹出层后焦点是否回到触发元素。 | `TooltipProps['returnFocusOnClose']` | `true` |
| role | 弹出层的 role。不传时 trigger 为 click、contextMenu、custom 是 dialog，其余是 tooltip；弹出层只是另一个控件的附属（如 combobox 的下拉）时传 presentation。 | `string` | - |
| showArrow | 是否显示小三角。 | `boolean` | `false` |
| spacing | 弹出层与触发元素的距离，单位为像素；对象形式分别设置水平与垂直方向。不传时无小三角为 4，有小三角为 10。 | `number \| { x: number; y: number }` | - |
| stopPropagation | 是否阻止弹出层上的点击事件冒泡。 | `boolean \| string` | - |
| style | 弹出层的内联样式，可传 style 对象或字符串。小三角只从对象形式中读取 `background-color` 与 `border-color`。 | `JSX.CSSProperties \| string` | - |
| trigger | 触发显示的时机，可选 hover、focus、click、custom 或 contextMenu。custom 时显示与否完全由 `visible` 决定。 | `Trigger` | `"hover"` |
| visible | 是否显示弹出层（受控）。传入后触发器、点击外部与 Esc 只通过 `onVisibleChange` 请求打开或关闭，使用方回写后才生效；不需要触发器事件时配合 trigger 为 custom 使用。 | `boolean` | - |

**Methods**

| 方法         | 说明                                                                                                                                                                                     | 类型                                          |
| ------------ | ---------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------- | --------------------------------------------- |
| focusTrigger | 让触发元素获得焦点，仅在 `returnFocusOnClose` 为 true 且 trigger 不是 custom 时生效。                                                                                                    | <ApiType detail={"() => void"}>Type</ApiType> |
| rePosition   | 按触发器当前位置立即重新计算弹出层位置。弹出层已随滚动、窗口尺寸变化、触发器移动以及触发器或弹出层的尺寸变化自动重新定位，一般无需调用；跟随触发器移动会晚一帧，需要同一帧内对齐时调用。 | <ApiType detail={"() => void"}>Type</ApiType> |
