import ApiType from '#/components/ApiType.astro';

**Props**

| 属性           | 说明                                                                                                                        | 类型                                                           | 默认值   |
| -------------- | --------------------------------------------------------------------------------------------------------------------------- | -------------------------------------------------------------- | -------- |
| alt            | 头像图片或文字的替代文本。                                                                                                  | `string`                                                       | -        |
| border         | 显示额外边框，可配置颜色和动画。                                                                                            | `boolean \| { color?: string; motion?: boolean }`              | -        |
| bottomSlot     | 底部标记的形状、背景色、文字颜色、类名和样式字符串；内容使用 bottomSlotText snippet，`render` 为 snippet 时完全自定义标记。 | `AvatarBottomSlot`                                             | -        |
| bottomSlotText | 预设底部标记中的文字或节点的 snippet。                                                                                      | `Snippet`                                                      | -        |
| children       | 头像的文字或自定义内容的 snippet。                                                                                          | `Snippet`                                                      | -        |
| color          | 头像的预设背景颜色。                                                                                                        | `AvatarColor`                                                  | `"grey"` |
| contentMotion  | 启用头像内容区域的动画。                                                                                                    | `boolean`                                                      | -        |
| gap            | 字符内容与头像左右两侧的间距，单位为像素。                                                                                  | `number`                                                       | `3`      |
| hoverMask      | 悬停时显示的覆盖层的 snippet，自行设置样式。                                                                                | `Snippet`                                                      | -        |
| imgAttr        | 传给内部 img 元素的原生属性。                                                                                               | `HTMLImgAttributes`                                            | -        |
| onError        | 图片加载失败时调用，返回 false 阻止默认回退。                                                                               | <ApiType detail={"() => boolean \u007c void"}>Type</ApiType>   | -        |
| onMouseEnter   | 指针进入头像时调用。                                                                                                        | <ApiType detail={"(event: MouseEvent) => void"}>Type</ApiType> | -        |
| onMouseLeave   | 指针离开头像时调用。                                                                                                        | <ApiType detail={"(event: MouseEvent) => void"}>Type</ApiType> | -        |
| shape          | 头像形状，可选 circle 或 square。                                                                                           | `AvatarShape`                                                  | -        |
| size           | 头像尺寸，支持预设名称或合法 CSS 尺寸。                                                                                     | `string`                                                       | -        |
| src            | 图片头像的资源地址。                                                                                                        | `string`                                                       | -        |
| srcSet         | 图片头像的响应式资源集合。                                                                                                  | `string`                                                       | -        |
| topSlot        | 顶部标记的渐变色、文字颜色、类名和样式字符串；内容使用 topSlotText snippet，`render` 为 snippet 时完全自定义标记。          | `AvatarTopSlot`                                                | -        |
| topSlotText    | 预设顶部标记中的文字或节点的 snippet。                                                                                      | `Snippet`                                                      | -        |
