import ApiType from '#/components/ApiType.astro';

**Props**

| 属性          | 说明                                                                         | 类型                                                                                                                      | 默认值                                         |
| ------------- | ---------------------------------------------------------------------------- | ------------------------------------------------------------------------------------------------------------------------- | ---------------------------------------------- |
| alt           | 头像图片或文字的替代文本。                                                   | `string`                                                                                                                  | `semiGlobalDefault("Avatar", "alt")`           |
| border        | 显示额外边框，可配置颜色和动画。                                             | <ApiType detail={"boolean \u007c { color?: string \u007c undefined; motion?: boolean \u007c undefined; }"}>Type</ApiType> | `semiGlobalDefault("Avatar", "border")`        |
| bottomSlot    | 底部标记的形状、背景色、文字颜色、类名和样式；内容使用 bottomSlotText 插槽。 | `AvatarBottomSlot`                                                                                                        | `semiGlobalDefault("Avatar", "bottomSlot")`    |
| color         | 头像的预设背景颜色。                                                         | `AvatarColor`                                                                                                             | `"grey"`                                       |
| contentMotion | 启用头像内容区域的动画。                                                     | `boolean`                                                                                                                 | `semiGlobalDefault("Avatar", "contentMotion")` |
| gap           | 字符内容与头像左右两侧的间距，单位为像素。                                   | `number`                                                                                                                  | `3`                                            |
| imgAttr       | 传给内部 img 元素的原生属性。                                                | `ImgHTMLAttributes`                                                                                                       | `semiGlobalDefault("Avatar", "imgAttr")`       |
| onError       | 图片加载失败时调用，返回 false 阻止默认回退；保留为函数 prop 以读取返回值。  | <ApiType detail={"((event: Event) => void \u007c boolean)"}>Type</ApiType>                                                | `undefined`                                    |
| shape         | 头像形状，可选 circle 或 square。                                            | `AvatarShape`                                                                                                             | `undefined`                                    |
| size          | 头像尺寸，支持预设名称或合法 CSS 尺寸。                                      | `string`                                                                                                                  | `undefined`                                    |
| src           | 图片头像的资源地址。                                                         | `string`                                                                                                                  | `semiGlobalDefault("Avatar", "src")`           |
| srcSet        | 图片头像的响应式资源集合。                                                   | `string`                                                                                                                  | `semiGlobalDefault("Avatar", "srcSet")`        |
| topSlot       | 顶部标记的渐变色、文字颜色、类名和样式；内容使用 topSlotText 插槽。          | `AvatarTopSlot`                                                                                                           | `semiGlobalDefault("Avatar", "topSlot")`       |

**Events**

| 事件       | 说明                 | 回调参数              |
| ---------- | -------------------- | --------------------- |
| mouseEnter | 指针进入头像时触发。 | `[event: MouseEvent]` |
| mouseLeave | 指针离开头像时触发。 | `[event: MouseEvent]` |

**Slots**

| 插槽           | 说明                               | 作用域参数 |
| -------------- | ---------------------------------- | ---------- |
| bottomSlot     | 完全自定义底部标记。               | `any`      |
| bottomSlotText | 预设底部标记中的文字或节点。       | `any`      |
| default        | 头像的文字或自定义内容。           | `any`      |
| hoverMask      | 悬停时显示的覆盖层，自行设置样式。 | `any`      |
| topSlot        | 完全自定义顶部标记。               | `any`      |
| topSlotText    | 预设顶部标记中的文字或节点。       | `any`      |
