import ApiType from '#/components/ApiType.astro';

**Props**

| 属性 | 说明 | 类型 | 默认值 |
| --- | --- | --- | --- |
| aspectRatio | 固定的裁切框宽高比；加载后改变时，裁切框按新比例重排。 | `number` | - |
| classNames | 内部各部位的类名，键为 `cropperBox`（裁切框），与组件自带的类名合并。 | <ApiType detail={"Partial\u003cRecord\u003cCropperPart, string>>"}>Type</ApiType> | - |
| defaultAspectRatio | 裁切框的初始宽高比，拖动时会随之变化。 | `number` | `1` |
| fill | 裁切结果中非图片部分的填充色。 | `string` | `"rgba(0, 0, 0, 0)"` |
| imgProps | 传给被裁切图片 img 标签的属性。src 与布局样式由组件决定，class 与组件自身的类名合并，style 位于组件布局之下（宽高与位移以组件为准），onLoad 在组件处理完成后调用；alt 默认为空字符串，crossorigin 默认为 anonymous，均可覆盖。 | `CropperImgProps` | - |
| maxZoom | 最大缩放倍数。 | `number` | `3` |
| minZoom | 最小缩放倍数。 | `number` | `0.1` |
| onZoomChange | 滚轮缩放时触发，参数为新的缩放比例。 | <ApiType detail={"(zoom: number) => void"}>Type</ApiType> | - |
| preview | 返回预览容器 DOM 的函数，裁切效果会实时镜像到该容器；可返回 null，表示容器尚未挂载。 | <ApiType detail={"() => HTMLElement \u007c null"}>Type</ApiType> | - |
| ref | 根 div 元素，其上附加 `getCropperCanvas` 方法。 | <ApiType detail={"Ref\u003cCropperElement>"}>Type</ApiType> | - |
| rotate | 图片旋转角度，单向绑定，组件不会改变它。 | `number` | - |
| shape | 裁切框形状。 | `'rect' \| 'round' \| 'roundRect'` | `"rect"` |
| showResizeBox | 是否展示裁切框边角的调整块。 | `boolean` | `true` |
| src | 被裁切图片的地址。 | `string` | - |
| styles | 内部各部位的样式对象（键用 kebab-case），键同 `classNames`；`cropperBox` 与组件计算的宽高和位移合并，宽高与位移以组件为准。 | <ApiType detail={"Partial\u003cRecord\u003cCropperPart, JSX.CSSProperties>>"}>Type</ApiType> | - |
| zoom | 受控的缩放比例，叠加在图片适应裁切区域的初始缩放之上。 | `number` | - |
| zoomStep | 滚轮缩放的步长。 | `number` | `0.1` |

**Methods**

| 方法             | 说明                                                     | 类型                                                       |
| ---------------- | -------------------------------------------------------- | ---------------------------------------------------------- |
| getCropperCanvas | 将裁切框内的区域按图片自身的分辨率绘制到新的 canvas 上。 | <ApiType detail={"() => HTMLCanvasElement"}>Type</ApiType> |
