import ApiType from '#/components/ApiType.astro';

**Props**

| Property      | Description                                                                                                                                                                  | Type                                                                           | Default |
| ------------- | ---------------------------------------------------------------------------------------------------------------------------------------------------------------------------- | ------------------------------------------------------------------------------ | ------- |
| active        | Whether the item is active: bolder text and a deeper color, plus a tick on its leading side when the tick is shown.                                                          | `boolean`                                                                      | -       |
| children      | Content of the item, rendered after the icon.                                                                                                                                | `ReactNode`                                                                    | -       |
| className     | Class name of the root `li` element.                                                                                                                                         | `string`                                                                       | -       |
| disabled      | Whether the item is disabled. A disabled item does not respond to clicks or mouse events and has `aria-disabled` `true`.                                                     | `boolean`                                                                      | `false` |
| forwardRef    | Called with the root `li` element when it mounts, for a component that wraps `Dropdown.Item` and needs the element.                                                          | <ApiType detail={"(ele: HTMLLIElement) => void"}>Type</ApiType>                | -       |
| icon          | Icon of the item, rendered before the content.                                                                                                                               | `React.ReactNode`                                                              | -       |
| onClick       | Called on click with the `MouseEvent`. For an item inside a nested submenu it is called on the left mouse press, and on keyboard activation. Not called for a disabled item. | <ApiType detail={"React.MouseEventHandler\u003cHTMLLIElement>"}>Type</ApiType> | -       |
| onContextMenu | Called on a right click. Not called for a disabled item.                                                                                                                     | <ApiType detail={"React.MouseEventHandler\u003cHTMLLIElement>"}>Type</ApiType> | -       |
| onMouseEnter  | Called when the pointer enters the item. Not called for a disabled item.                                                                                                     | <ApiType detail={"React.MouseEventHandler\u003cHTMLLIElement>"}>Type</ApiType> | -       |
| onMouseLeave  | Called when the pointer leaves the item. Not called for a disabled item.                                                                                                     | <ApiType detail={"React.MouseEventHandler\u003cHTMLLIElement>"}>Type</ApiType> | -       |
| ref           | Reference to the root `li` element.                                                                                                                                          | <ApiType detail={"React.Ref\u003cHTMLLIElement>"}>Type</ApiType>               | -       |
| showTick      | Takes effect only outside a Dropdown. Inside one, the `showTick` of the Dropdown applies.                                                                                    | `boolean`                                                                      | -       |
| style         | Inline style of the root `li` element.                                                                                                                                       | `CSSProperties`                                                                | -       |
| type          | Text color type: `primary`, `secondary`, `tertiary`, `warning` or `danger`.                                                                                                  | `Type`                                                                         | -       |
