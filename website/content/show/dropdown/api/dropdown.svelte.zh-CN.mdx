import ApiType from '#/components/ApiType.astro';

**Props**

| 属性 | 说明 | 类型 | 默认值 |
| --- | --- | --- | --- |
| afterClose | 菜单关闭且离场动画结束后触发。 | <ApiType detail={"() => void"}>Type</ApiType> | - |
| afterOpen | 菜单打开且进场动画结束后触发；没有进场动画时打开后即触发，受控与非受控相同；进场结束前被关闭的打开不触发。 | <ApiType detail={"() => void"}>Type</ApiType> | - |
| aria-describedby | 弹出层 role 为 dialog 时的 aria-describedby。 | `string` | - |
| aria-label | 弹出层 role 为 dialog 时的 aria-label。 | `string` | - |
| aria-labelledby | 弹出层 role 为 dialog 时的 aria-labelledby。 | `string` | - |
| autoAdjustOverflow | 弹出层被遮挡时是否自动调整方向。 | `boolean` | - |
| child | 自己渲染触发元素的 snippet，参数 `{ props }` 要展开到可聚焦的元素上，`class` 需与自己的类合并（`class={[props.class, 'own']}`）；适用于按钮、链接，以及嵌套菜单里作为父项的 DropdownItem。优先于 children。 | <ApiType detail={"Snippet\u003c[{ props: DropdownTriggerProps }]>"}>Type</ApiType> | - |
| children | 触发元素，被一个 span 包裹，span 带触发属性并响应键盘。 | `Snippet` | - |
| class | 弹出层 `.semi-dropdown-wrapper` 的类名。 | `string` | - |
| clickToHide | 在弹出层内点击时是否自动关闭弹出层。 | `boolean` | - |
| clickTriggerToHide | 点击触发元素时是否关闭已打开的弹出层。 | `boolean` | - |
| closeOnEsc | 在触发元素或弹出层按 Esc 是否关闭菜单；trigger 为 custom 时只触发 onEscKeyDown，由使用方决定是否关闭。 | `boolean` | `true` |
| condition | 是否允许触发显示；仅显式设为 false 时，hover、click、focus 等触发行为不生效，trigger 为 custom 时仍按 visible 显示。由 true 变为 false 时立即关闭已显示的弹出层；visible 受控时只调用 onVisibleChange(false)，使用方回写后才关闭。 | `boolean` | - |
| content | 菜单内容，由 DropdownMenu 及 DropdownItem、DropdownTitle、DropdownDivider 构成；提供后（即使为空）优先于 menu。 | `Snippet` | - |
| contentClassName | 下拉菜单根元素 `.semi-dropdown` 的类名。 | `string` | - |
| disableFocusListener | trigger 为 hover 时，是否不响应键盘聚焦打开菜单。 | `boolean` | - |
| guardFocus | 焦点位于弹出层内时，Tab 是否在弹出层内循环。 | `boolean` | - |
| keepDOM | 关闭时是否保留弹出层内容而不销毁。 | `boolean` | - |
| menu | 通过数据快速配置菜单，数据项见下方 DropdownMenuItem；提供了 content 时以 content 为准。 | `DropDownMenuItem[]` | - |
| menuIcon | menu 中每个 item 项的图标，参数为 `{ item, index }`，即该数据项与序号；图标容器总会渲染，snippet 无输出时为空。 | <ApiType detail={"Snippet\u003c[DropdownMenuIconArgs]>"}>Type</ApiType> | - |
| motion | 是否展示弹出层动画。 | `boolean` | `true` |
| mouseEnterDelay | 指针移入触发元素后延迟显示的毫秒数，仅 trigger 为 hover 或 focus 时生效。 | `number` | - |
| mouseLeaveDelay | 指针移出弹出层后延迟隐藏的毫秒数，仅 trigger 为 hover 或 focus 时生效，不小于 mouseEnterDelay。 | `number` | `100` |
| onClickOutSide | 菜单打开时在触发元素与弹出层以外的区域按下鼠标时触发，仅 trigger 为 click、contextMenu、custom 或开启 clickTriggerToHide 时有效。 | <ApiType detail={"(event: MouseEvent) => void"}>Type</ApiType> | - |
| onEscKeyDown | closeOnEsc 为 true 时，菜单打开期间按 Esc 触发。 | <ApiType detail={"(event: KeyboardEvent) => void"}>Type</ApiType> | - |
| onOpen | 菜单显示、进场动画开始时触发，每次打开一次，早于 afterOpen，受控与非受控相同；悬停打开后指针已离开而撤回的打开不触发。 | <ApiType detail={"() => void"}>Type</ApiType> | - |
| onVisibleChange | 菜单显示状态变化时触发；visible 受控时只报告用户交互发起的请求，父组件自己改变 visible 不触发。 | <ApiType detail={"(visible: boolean) => void"}>Type</ApiType> | - |
| position | 弹出菜单相对触发元素的位置，常用 bottom、bottomLeft、bottomRight，更多见 Tooltip 的位置。 | `Position` | `"bottom"` |
| prefixCls | 弹出层类名前缀，设置后弹出层不再带 Dropdown 的样式。 | `string` | `"semi-dropdown"` |
| preventScroll | 聚焦时是否阻止浏览器滚动到被聚焦元素。 | `boolean` | - |
| returnFocusOnClose | 菜单关闭后焦点是否回到触发元素。 | `boolean` | `true` |
| role | 包裹菜单的弹层容器的 ARIA role，默认 presentation，菜单本身的 role 为 menu。 | `string` | `"presentation"` |
| selectionMode | 本层菜单项的选择方式，已选项由 active 标记：single 时菜单项为带 aria-checked 的 menuitemradio，multiple 时为带 aria-checked 的 menuitemcheckbox；未设置时 showTick 开启则为带 aria-checked 的 menuitemcheckbox，否则为 menuitem。只影响角色与状态，勾的显示仍由 showTick 决定；嵌套的子菜单不继承。 | `DropdownSelectionMode` | - |
| showTick | 是否在 active 的 DropdownItem 左侧展示表示选中的勾；只作用于本 Dropdown 的菜单，嵌套的子菜单不继承。 | `boolean` | `false` |
| spacing | 弹出层与触发元素的距离，单位为像素；缺省时顶层菜单为 4，嵌套的子菜单为 2。 | `number \| { x: number; y: number }` | - |
| stopPropagation | 是否阻止弹出层上的 click、mousedown 与焦点进出事件冒泡。 | `boolean` | - |
| style | 弹出层 `.semi-dropdown-wrapper` 的内联样式字符串。 | `string` | - |
| trigger | 触发菜单显示的行为，可选 hover、focus、click、custom、contextMenu；custom 时显示与否完全由 visible 决定。 | `'hover' \| 'focus' \| 'click' \| 'custom' \| 'contextMenu'` | `"hover"` |
| visible | 菜单是否显示，可用 `bind:visible` 绑定，配合 trigger 为 custom 使用；传入后由该值决定显示，绑定 undefined 则回到非受控。 | `boolean` | - |
| wrapperClassName | 触发元素被 span 包裹时，该 span 的类名。 | `string` | - |
| wrapperId | 弹出层节点的 id，触发元素的 aria 属性指向它；缺省时自动生成。 | `string` | - |

**Methods**

| 方法         | 说明                                                                                                                                                                                     | 类型                                          |
| ------------ | ---------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------- | --------------------------------------------- |
| focusTrigger | 让触发元素获得焦点，仅在 returnFocusOnClose 为 true 且 trigger 不是 custom 时生效，且只作用于非 child 触发元素。                                                                         | <ApiType detail={"() => void"}>Type</ApiType> |
| rePosition   | 按触发器当前位置立即重新计算弹出层位置。弹出层已随滚动、窗口尺寸变化、触发器移动以及触发器或弹出层的尺寸变化自动重新定位，一般无需调用；跟随触发器移动会晚一帧，需要同一帧内对齐时调用。 | <ApiType detail={"() => void"}>Type</ApiType> |
