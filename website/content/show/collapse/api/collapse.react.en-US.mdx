import ApiType from '#/components/ApiType.astro';

**Props**

| Property            | Description                                                                                                                                               | Type                                                                                  | Default   |
| ------------------- | --------------------------------------------------------------------------------------------------------------------------------------------------------- | ------------------------------------------------------------------------------------- | --------- |
| accordion           | Accordion mode, only one panel is expanded at a time.                                                                                                     | `boolean`                                                                             | -         |
| activeKey           | Controlled property, key of the currently expanded panel; `null` and the empty string mean all panels are collapsed.                                      | `string \| string[] \| null`                                                          | -         |
| children            | `Collapse.Panel` panels.                                                                                                                                  | `React.ReactNode`                                                                     | -         |
| className           | Class name of the root element.                                                                                                                           | `string`                                                                              | -         |
| clickHeaderToExpand | Clicking anywhere in the header row toggles the panel; when `false` only the toggle button and the arrow respond.                                         | `boolean`                                                                             | `true`    |
| collapseIcon        | Icon shown while a panel is expanded, given as a ReactNode; `IconChevronUp` when omitted, and `null` shows no icon.                                       | `React.ReactNode`                                                                     | -         |
| defaultActiveKey    | Key of the panel expanded initially.                                                                                                                      | `string \| string[]`                                                                  | `""`      |
| expandIcon          | Icon shown while a panel is collapsed, given as a ReactNode; `IconChevronDown` when omitted, and `null` shows no icon.                                    | `React.ReactNode`                                                                     | -         |
| expandIconPosition  | Position of the expand icon: `left` or `right`.                                                                                                           | `'left' \| 'right'`                                                                   | `"right"` |
| keepDOM             | Whether to keep the DOM tree of hidden panels; destroyed by default. When omitted it reads the global default of Collapse, then that of Collapsible.      | `boolean`                                                                             | -         |
| lazyRender          | Used with `keepDOM`; when `true`, panel content is not rendered until the panel is first expanded.                                                        | `boolean`                                                                             | `false`   |
| motion              | Whether to enable the animation; enabled by default. When omitted it reads the global default of Collapse, then that of Collapsible.                      | `boolean`                                                                             | -         |
| onChange            | Called when a panel is toggled, with the array of expanded keys and the native MouseEvent (keyboard use also produces a click through the native button). | <ApiType detail={"(activeKey: string[], e: React.MouseEvent) => void"}>Type</ApiType> | -         |
| ref                 | Reference to the root `div` element.                                                                                                                      | <ApiType detail={"React.Ref\u003cHTMLDivElement>"}>Type</ApiType>                     | -         |
| style               | Style of the root element.                                                                                                                                | `React.CSSProperties`                                                                 | -         |
