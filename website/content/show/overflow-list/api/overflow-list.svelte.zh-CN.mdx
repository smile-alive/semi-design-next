import ApiType from '#/components/ApiType.astro';

**Props**

| 属性 | 说明 | 类型 | 默认值 |
| --- | --- | --- | --- |
| collapseFrom | collapse 模式的折叠方向，可选 start 或 end。 | `'start' \| 'end'` | `"end"` |
| item | 单个展示项内容的 snippet，参数 `{ item, index }`：数据项及其在 items 中的位置。scroll 模式要求 snippet 渲染出元素，否则该项被跳过。 | <ApiType detail={"Snippet\u003c[OverflowListItemSnippetProps\u003cT>]>"}>Type</ApiType> | - |
| itemKey | 每项 key 的字段名，或取 key 的函数；默认读取 key 字段，没有时取该项在 items 中的位置。 | <ApiType detail={"OverflowListKey \u007c ((item: T) => OverflowListKey)"}>Type</ApiType> | - |
| items | 渲染项目，只放数据，不放节点；scroll 模式要求每项含 key。 | `T[]` | - |
| minVisibleItems | collapse 模式下最少展示的项数。 | `number` | `0` |
| onIntersect | scroll 模式下各项的相交状态变化时的回调。 | <ApiType detail={"(res: Record\u003cstring, IntersectionObserverEntry>) => void"}>Type</ApiType> | - |
| onOverflow | collapse 模式下溢出项变化时的回调，全部放得下时参数为空数组。 | <ApiType detail={"(overflowItems: T[]) => void"}>Type</ApiType> | - |
| onVisibleStateChange | scroll 模式下各项的显示或隐藏状态变化时的回调。 | <ApiType detail={"(visibleState: Map\u003cstring, boolean>) => void"}>Type</ApiType> | - |
| overflow | 溢出项内容的 snippet，参数 `{ items, position }`。collapse 模式只在有溢出项时渲染一次，items 为放不下的项；scroll 模式在两侧各渲染一次，position 为 start 或 end，items 为该侧滚出可见区域的项。 | <ApiType detail={"Snippet\u003c[OverflowListOverflowSnippetProps\u003cT>]>"}>Type</ApiType> | - |
| overflowRenderDirection | scroll 模式下溢出节点的位置，可选 both、start 或 end；供 Tabs 使用，不面向使用方。 | `'both' \| 'start' \| 'end'` | `"both"` |
| renderMode | 渲染模式，可选 collapse 或 scroll。 | `'collapse' \| 'scroll'` | `"collapse"` |
| threshold | scroll 模式下触发溢出回调的可见比例阈值。 | `number` | `0.75` |
| wrapperClassName | scroll 模式下滚动 wrapper 的类名。 | `string` | - |
| wrapperProps | scroll 模式下滚动 wrapper 的其他属性，例如承载可见项的 role。 | <ApiType detail={"HTMLAttributes\u003cHTMLDivElement>"}>Type</ApiType> | - |
| wrapperStyle | scroll 模式下滚动 wrapper 的样式字符串。 | `string` | - |
