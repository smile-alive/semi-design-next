import ApiType from '#/components/ApiType.astro';

**Props**

| Property       | Description                                                                                                                                                                             | Type                                                               | Default     |
| -------------- | --------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------- | ------------------------------------------------------------------ | ----------- |
| children       | The element the badge is attached to; the badge is positioned relative to it. Without children, the badge is shown as a standalone element.                                             | `React.ReactNode`                                                  | -           |
| className      | Class name of the root `span` element.                                                                                                                                                  | `string`                                                           | -           |
| count          | Badge content. Numbers and strings are shown as text; other nodes are rendered as is, without the `type` and `theme` classes.                                                           | `React.ReactNode`                                                  | -           |
| countClassName | Class name of the count element.                                                                                                                                                        | `string`                                                           | -           |
| countStyle     | Style of the count element, used only when `style` is not set.                                                                                                                          | `React.CSSProperties`                                              | -           |
| dot            | Shows a small dot instead of `count`.                                                                                                                                                   | `boolean`                                                          | `false`     |
| onClick        | Called when the root element is clicked.                                                                                                                                                | <ApiType detail={"(e: React.MouseEvent) => any"}>Type</ApiType>    | -           |
| onMouseEnter   | Called when the pointer enters the root element.                                                                                                                                        | <ApiType detail={"(e: React.MouseEvent) => any"}>Type</ApiType>    | -           |
| onMouseLeave   | Called when the pointer leaves the root element.                                                                                                                                        | <ApiType detail={"(e: React.MouseEvent) => any"}>Type</ApiType>    | -           |
| overflowCount  | Maximum number shown; a larger numeric `count` is displayed as `${overflowCount}+`. Applies to numbers only, and `0` is treated as unset.                                               | `number`                                                           | -           |
| position       | Badge position relative to the wrapped element: `leftTop`, `leftBottom`, `rightTop` or `rightBottom`. When unset, it is `leftTop` under an RTL ConfigProvider and `rightTop` otherwise. | `BadgePosition`                                                    | -           |
| ref            | Reference to the root `span` element.                                                                                                                                                   | <ApiType detail={"React.Ref\u003cHTMLSpanElement>"}>Type</ApiType> | -           |
| style          | Style of the count element. When set, it replaces `countStyle` rather than merging with it; it is not applied to the root element.                                                      | `React.CSSProperties`                                              | -           |
| theme          | Badge theme: `solid`, `light` or `inverted`.                                                                                                                                            | `BadgeTheme`                                                       | `"solid"`   |
| type           | Badge type: `primary`, `secondary`, `tertiary`, `danger`, `warning` or `success`.                                                                                                       | `BadgeType`                                                        | `"primary"` |
