import ApiType from '#/components/ApiType.astro';

**Props**

| 属性           | 说明                                                                                                                                                    | 类型                                                               | 默认值      |
| -------------- | ------------------------------------------------------------------------------------------------------------------------------------------------------- | ------------------------------------------------------------------ | ----------- |
| children       | 徽章所附着的元素，徽章相对它定位；不传时徽章作为独立元素显示。                                                                                          | `React.ReactNode`                                                  | -           |
| className      | 根元素（`span`）的类名。                                                                                                                                | `string`                                                           | -           |
| count          | 徽章内容，数字和字符串按文字显示；其他 ReactNode 直接渲染，此时不带 `type` 与 `theme` 对应的样式类。                                                    | `React.ReactNode`                                                  | -           |
| countClassName | 内容区域的类名。                                                                                                                                        | `string`                                                           | -           |
| countStyle     | 内容区域的样式，仅在未设置 `style` 时生效。                                                                                                             | `React.CSSProperties`                                              | -           |
| dot            | 显示为小圆点，不展示 `count`。                                                                                                                          | `boolean`                                                          | `false`     |
| onClick        | 点击根元素时调用。                                                                                                                                      | <ApiType detail={"(e: React.MouseEvent) => any"}>Type</ApiType>    | -           |
| onMouseEnter   | 鼠标移入根元素时调用。                                                                                                                                  | <ApiType detail={"(e: React.MouseEvent) => any"}>Type</ApiType>    | -           |
| onMouseLeave   | 鼠标移出根元素时调用。                                                                                                                                  | <ApiType detail={"(e: React.MouseEvent) => any"}>Type</ApiType>    | -           |
| overflowCount  | 数字 `count` 的最大展示值，超出后显示为 `${overflowCount}+`；只对数字生效，`0` 视为未设置。                                                             | `number`                                                           | -           |
| position       | 徽章相对被包裹元素的位置，可选 `leftTop`、`leftBottom`、`rightTop`、`rightBottom`；未设置时在 RTL 的 ConfigProvider 下为 `leftTop`，否则为 `rightTop`。 | `BadgePosition`                                                    | -           |
| ref            | 根元素（`span`）的引用。                                                                                                                                | <ApiType detail={"React.Ref\u003cHTMLSpanElement>"}>Type</ApiType> | -           |
| style          | 内容区域的样式，设置后替代 `countStyle` 而不是与其合并；不作用于根元素。                                                                                | `React.CSSProperties`                                              | -           |
| theme          | 徽章主题，可选 `solid`、`light`、`inverted`。                                                                                                           | `BadgeTheme`                                                       | `"solid"`   |
| type           | 徽章类型，可选 `primary`、`secondary`、`tertiary`、`danger`、`warning`、`success`。                                                                     | `BadgeType`                                                        | `"primary"` |
