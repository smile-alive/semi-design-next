import ApiType from '#/components/ApiType.astro';

**Props**

| 属性        | 说明                                                                    | 类型                                                              | 默认值 |
| ----------- | ----------------------------------------------------------------------- | ----------------------------------------------------------------- | ------ |
| active      | 是否展示动画效果。                                                      | `boolean`                                                         | -      |
| children    | 加载结束后显示的内容，`loading` 为 `false` 时原样渲染，不包外层元素。   | `ReactNode`                                                       | -      |
| className   | 占位根元素的类名，`loading` 为 `false` 时不渲染占位根元素，因此不生效。 | `string`                                                          | -      |
| loading     | 为 `true` 时显示 `placeholder`，反之显示子内容。                        | `boolean`                                                         | `true` |
| placeholder | 加载等待时的占位元素。                                                  | `ReactNode`                                                       | -      |
| ref         | 占位根 `div` 的引用，`loading` 为 `false` 时不挂载。                    | <ApiType detail={"React.Ref\u003cHTMLDivElement>"}>Type</ApiType> | -      |
| style       | 占位根元素的样式，`loading` 为 `false` 时不生效。                       | `CSSProperties`                                                   | -      |
