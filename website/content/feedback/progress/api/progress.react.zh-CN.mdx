import ApiType from '#/components/ApiType.astro';

**Props**

| 属性 | 说明 | 类型 | 默认值 |
| --- | --- | --- | --- |
| className | 根元素的类名。 | `string` | `""` |
| direction | 条状进度条的方向，可选 `horizontal` 或 `vertical`。 | `'horizontal' \| 'vertical'` | `"horizontal"` |
| format | 格式化函数，入参为当前显示的百分比（截断到 0 到 100，动画过程中为中间值），返回值渲染在环形进度条中心或条状进度条右侧。每次渲染都会调用，即使 `showInfo` 为 `false`。 | <ApiType detail={"(percent: number) => React.ReactNode"}>Type</ApiType> | <ApiType detail={"(percent: number) => `${percent}%`"}>Type</ApiType> |
| indeterminate | 是否为不确定状态。开启后 `percent` 不再控制进度长度，不显示文字，且不设置 `aria-valuenow`。 | `boolean` | `false` |
| motion | 文字中的数字是否以动画过渡到新的 `percent`；关闭后直接跳到目标值。 | `Motion` | `true` |
| orbitStroke | 进度条轨道的填充色；未设置时由样式决定。 | `string` | - |
| percent | 进度百分比；超出 0 到 100 的值会被截断。 | `number` | `0` |
| ref | 根元素（`role="progressbar"`）的引用。 | <ApiType detail={"React.Ref\u003cHTMLDivElement>"}>Type</ApiType> | - |
| showInfo | 是否显示文字：环形显示在中间，条状显示在右侧。`size` 为 `small` 的环形进度条不显示文字。 | `boolean` | `false` |
| size | 尺寸，可选 `default`、`small`（仅环形生效）或 `large`（仅条状生效）。 | `Size` | `"default"` |
| stroke | 填充色。数组形式为 `{ percent, color }` 的区间列表，`color` 可为任意 CSS 颜色或 Semi 色板名，如 `blue`、`orange-9`，未写色阶时取 5。 | `string \| StrokeArr` | - |
| strokeGradient | 是否在相邻颜色区间之间生成渐变色补齐颜色，仅在 `stroke` 为区间数组时生效。 | `boolean` | `false` |
| strokeLinecap | 环形进度条端点形状，可选 `round` 或 `square`；条状进度条不生效。 | `'round' \| 'square'` | `"round"` |
| strokeWidth | 环形进度条的线宽。 | `number` | `4` |
| style | 根元素的样式。 | `React.CSSProperties` | `{}` |
| type | 类型，可选 `line` 或 `circle`。 | `'line' \| 'circle'` | `"line"` |
| width | 环形进度条的直径，仅 `type` 为 `circle` 时生效；未设置时，`size` 为 `small` 取 24，否则取 72。 | `number` | - |
