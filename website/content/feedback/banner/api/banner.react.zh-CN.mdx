import ApiType from '#/components/ApiType.astro';

**Props**

| 属性        | 说明                                                                           | 类型                                                              | 默认值   |
| ----------- | ------------------------------------------------------------------------------ | ----------------------------------------------------------------- | -------- |
| bordered    | 是否展示边框，仅在非全屏模式下有效，默认不展示。                               | `boolean`                                                         | -        |
| children    | 自定义内容，渲染在横幅底部的额外区域。                                         | `React.ReactNode`                                                 | -        |
| className   | 根元素的类名。                                                                 | `string`                                                          | -        |
| closeIcon   | 替换关闭按钮内的图标；传 `null` 时不显示关闭按钮，未设置时显示默认关闭图标。   | `React.ReactNode`                                                 | -        |
| description | 描述内容。                                                                     | `React.ReactNode`                                                 | -        |
| fullMode    | 是否为全屏模式；设为 false 时使用非全屏样式。                                  | `boolean`                                                         | `true`   |
| icon        | 替换 `type` 对应的图标；传 `null` 时不显示图标，未设置时显示该类型的默认图标。 | `React.ReactNode`                                                 | -        |
| onClose     | 点击关闭按钮时调用，参数为按钮的原生 click 事件；回调返回后横幅隐藏。          | <ApiType detail={"(e: React.MouseEvent) => void"}>Type</ApiType>  | -        |
| ref         | 根元素（`role="alert"` 的 `div`）的引用。                                      | <ApiType detail={"React.Ref\u003cHTMLDivElement>"}>Type</ApiType> | -        |
| style       | 根元素的样式。                                                                 | `React.CSSProperties`                                             | -        |
| title       | 标题内容。                                                                     | `React.ReactNode`                                                 | -        |
| type        | 类型，可选 `info`、`success`、`danger` 或 `warning`。                          | `Type`                                                            | `"info"` |
