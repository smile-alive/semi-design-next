import ApiType from '#/components/ApiType.astro';

**Props**

| 属性        | 说明                                                                  | 类型                                                       | 默认值   |
| ----------- | --------------------------------------------------------------------- | ---------------------------------------------------------- | -------- |
| bordered    | 是否展示边框，仅在非全屏模式下有效。                                  | `boolean`                                                  | -        |
| closeIcon   | 替换关闭按钮内的图标；传 null 时不显示关闭按钮。                      | `JSX.Element`                                              | -        |
| description | 描述内容。                                                            | `JSX.Element`                                              | -        |
| fullMode    | 是否为全屏模式。                                                      | `boolean`                                                  | `true`   |
| icon        | 替换 type 对应的图标；传 null 时不显示图标。                          | `JSX.Element`                                              | -        |
| onClose     | 点击关闭按钮时调用，参数为按钮的原生 click 事件；回调返回后横幅隐藏。 | <ApiType detail={"(e: MouseEvent) => void"}>Type</ApiType> | -        |
| title       | 标题内容。                                                            | `JSX.Element`                                              | -        |
| type        | 类型，可选 info、success、danger、warning。                           | `BannerType`                                               | `"info"` |
