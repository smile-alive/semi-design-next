import ApiType from '#/components/ApiType.astro';

**Props**

| 属性 | 说明 | 类型 | 默认值 |
| --- | --- | --- | --- |
| height | 高度，数字按 px 处理，字符串原样使用。 | `number \| string` | `400` |
| limitSearchButtonBounds | 是否限制搜索按钮的拖动范围在组件容器内。 | `boolean` | - |
| onChange | 编辑内容变化时触发，参数为编辑器当前文本。 | <ApiType detail={"(value: string) => void"}>Type</ApiType> | - |
| options | 编辑器配置，字段见 JsonViewerOptions。ariaLabel、readOnly、autoWrap 的默认值按键合并，只传部分字段时其余仍取默认值。options 按深比较变化时重建编辑器；就地修改 formatOptions、customRenderRule 等嵌套对象不会被察觉，需替换为新对象。 | `JsonViewerOptions` | - |
| ref | 根元素，类型为 `JsonViewerElement`，上面挂载着 JsonViewerHandle 的方法。 | <ApiType detail={"Ref\u003cJsonViewerElement>"}>Type</ApiType> | - |
| renderSearchButton | 自定义渲染搜索按钮。第一个参数是创建默认按钮的函数，在插入处调用；第二个参数 `controls` 为搜索状态与方法；`controls.showSearchBar` 是 getter，需在返回的 JSX 内读取。 | <ApiType detail={"(defaultSearchButton: () => JSX.Element, searchControls: JsonViewerSearchControls) => JSX.Element"}>Type</ApiType> | - |
| showSearch | 是否显示搜索按钮。 | `boolean` | `true` |
| value | 展示与编辑的 JSON 文本，未传时按空文本创建。只有 value 变为与编辑器当前内容不同的值时才会重建编辑器，把 onChange 的值写回 value 不会丢失光标。 | `string` | `""` |
| width | 宽度，数字按 px 处理，字符串原样使用。 | `number \| string` | `400` |

**Methods**

| 方法             | 说明                                                                                           | 类型                                                                                                                           |
| ---------------- | ---------------------------------------------------------------------------------------------- | ------------------------------------------------------------------------------------------------------------------------------ |
| format           | 按 formatOptions 格式化当前内容。                                                              | <ApiType detail={"() => void"}>Type</ApiType>                                                                                  |
| getSearchResults | 获取当前搜索结果。                                                                             | <ApiType detail={"() => ReturnType\u003cJsonViewerFoundation['getSearchResults']>"}>Type</ApiType>                             |
| getStyle         | 返回当前的 width 与 height。                                                                   | <ApiType detail={"() => { width: number \u007c string; height: number \u007c string }"}>Type</ApiType>                         |
| getValue         | 获取编辑器当前文本。                                                                           | <ApiType detail={"() => string"}>Type</ApiType>                                                                                |
| nextSearch       | 跳转到下一个搜索结果，可选步长。                                                               | <ApiType detail={"(step?: number) => void"}>Type</ApiType>                                                                     |
| prevSearch       | 跳转到上一个搜索结果，可选步长。                                                               | <ApiType detail={"(step?: number) => void"}>Type</ApiType>                                                                     |
| replace          | 替换当前搜索匹配项；替换文本为空串时删除匹配，没有匹配时不做任何事，只读模式下同样不做任何事。 | <ApiType detail={"(replaceText: string) => void"}>Type</ApiType>                                                               |
| replaceAll       | 替换所有搜索匹配项；替换文本为空串时删除全部匹配，只读模式下不做任何事。                       | <ApiType detail={"(replaceText: string) => void"}>Type</ApiType>                                                               |
| search           | 搜索文本，可选参数控制大小写敏感、全词匹配和正则表达式。                                       | <ApiType detail={"(searchText: string, caseSensitive?: boolean, wholeWord?: boolean, regex?: boolean) => void"}>Type</ApiType> |
