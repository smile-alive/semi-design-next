import ApiType from '#/components/ApiType.astro';

**Props**

| 属性                     | 说明                                                                                                                                                          | 类型                                                                                               | 默认值        |
| ------------------------ | ------------------------------------------------------------------------------------------------------------------------------------------------------------- | -------------------------------------------------------------------------------------------------- | ------------- |
| align                    | 对话布局方式，可选 `leftRight`（左右分布）或 `leftAlign`（左对齐）。                                                                                          | `ChatAlign`                                                                                        | `"leftRight"` |
| bottomSlot               | 输入框下方的节点。                                                                                                                                            | `JSX.Element`                                                                                      | -             |
| canSend                  | 发送按钮是否可用。通常由内部逻辑决定，设置后以此值为准。                                                                                                      | `boolean`                                                                                          | -             |
| chatBoxRenderConfig      | 消息框各部分的自定义渲染函数，类型见 `ChatBoxRenderConfig`。                                                                                                  | `ChatBoxRenderConfig`                                                                              | -             |
| chats                    | 对话列表，受控使用，配合 `onChatsChange` 回写。只有整体替换数组才视为变化，就地修改同一数组不会更新界面。                                                     | `Message[]`                                                                                        | -             |
| customMarkDownComponents | 自定义 Markdown 渲染组件，键为元素名，透传给消息内容的 MarkdownRender。                                                                                       | `MarkdownRenderComponents`                                                                         | -             |
| enableUpload             | 是否启用上传。布尔值同时控制粘贴、点击和拖拽上传；传对象时分别设置 `pasteUpload`、`clickUpload`、`dragUpload`，未设置的项为 `true`。                          | `boolean \| EnableUploadProps`                                                                     | -             |
| escapeHtml               | 是否转义用户消息中的 HTML 标签，防止被 Markdown 解析器当作 HTML 处理导致内容丢失。                                                                            | `boolean`                                                                                          | `true`        |
| hintCls                  | 提示区最外层的类名。                                                                                                                                          | `string`                                                                                           | -             |
| hints                    | 提示信息列表。点击某项后，该项作为新的用户消息加入对话，并触发 `onHintClick`。                                                                                | `string[]`                                                                                         | -             |
| hintStyle                | 提示区最外层的样式。                                                                                                                                          | `JSX.CSSProperties \| string`                                                                      | -             |
| inputBoxCls              | 输入框的类名。                                                                                                                                                | `string`                                                                                           | -             |
| inputBoxStyle            | 输入框的样式。                                                                                                                                                | `JSX.CSSProperties \| string`                                                                      | -             |
| markdownRenderProps      | 传给消息内容所用 MarkdownRender 的属性，默认 `format` 为 `md`。`raw` 与 `components` 由 Chat 设置，自定义渲染组件使用 `customMarkDownComponents`。            | `MarkdownRenderProps`                                                                              | -             |
| mode                     | 对话模式，可选 `bubble`、`noBubble` 或 `userBubble`。                                                                                                         | `ChatMode`                                                                                         | `"bubble"`    |
| onChatsChange            | 对话列表变化时触发，参数为新的列表。                                                                                                                          | <ApiType detail={"(chats: Message[]) => void"}>Type</ApiType>                                      | -             |
| onClear                  | 清除上下文时触发。                                                                                                                                            | <ApiType detail={"() => void"}>Type</ApiType>                                                      | -             |
| onHintClick              | 点击提示信息时触发，参数为提示文本。                                                                                                                          | <ApiType detail={"(hint: string) => void"}>Type</ApiType>                                          | -             |
| onInputChange            | 输入内容或附件变化时触发，参数为 `{ inputValue, attachment }`。                                                                                               | <ApiType detail={"(props: { inputValue: string; attachment: FileItem[] }) => void"}>Type</ApiType> | -             |
| onMessageBadFeedback     | 消息负向反馈时触发，参数为消息。                                                                                                                              | <ApiType detail={"(message?: Message) => void"}>Type</ApiType>                                     | -             |
| onMessageCopy            | 复制消息时触发，参数为消息。                                                                                                                                  | <ApiType detail={"(message?: Message) => void"}>Type</ApiType>                                     | -             |
| onMessageDelete          | 删除消息时触发，参数为消息。                                                                                                                                  | <ApiType detail={"(message?: Message) => void"}>Type</ApiType>                                     | -             |
| onMessageGoodFeedback    | 消息正向反馈时触发，参数为消息。                                                                                                                              | <ApiType detail={"(message?: Message) => void"}>Type</ApiType>                                     | -             |
| onMessageReset           | 重置消息时触发；由 `resetMessage` 方法触发时参数为 `null`。                                                                                                   | <ApiType detail={"(message: Message \u007c null) => void"}>Type</ApiType>                          | -             |
| onMessageSend            | 发送消息时触发，参数为文本内容和附件。                                                                                                                        | <ApiType detail={"(content: string, attachment?: FileItem[]) => void"}>Type</ApiType>              | -             |
| onStopGenerator          | 点击停止生成按钮时触发，参数为鼠标事件。                                                                                                                      | <ApiType detail={"(e: MouseEvent) => void"}>Type</ApiType>                                         | -             |
| placeholder              | 输入框占位符。                                                                                                                                                | `string`                                                                                           | -             |
| ref                      | 根元素，带有下方 Methods 中的方法。                                                                                                                           | <ApiType detail={"Ref\u003cChatElement>"}>Type</ApiType>                                           | -             |
| renderDivider            | 自定义清除上下文后的分割线，参数为消息，返回节点。每条分割线调用一次，替换函数时重新调用。                                                                    | <ApiType detail={"(message?: Message) => JSX.Element"}>Type</ApiType>                              | -             |
| renderHintBox            | 自定义提示项的渲染，参数为 `content`、`index` 和 `onHintClick`，返回节点。                                                                                    | <ApiType detail={"(props: RenderHintBoxProps) => JSX.Element"}>Type</ApiType>                      | -             |
| renderInputArea          | 自定义输入区域，参数为 `defaultNode`、`onSend`、`onClear` 和 `detailProps`，返回节点。`defaultNode` 与 `detailProps` 里的节点是渲染函数，在要放置的位置调用。 | <ApiType detail={"(props: RenderInputAreaProps) => JSX.Element"}>Type</ApiType>                    | -             |
| roleConfig               | 角色信息，键为消息的 role。`avatar` 是图片 URL，其他头像内容使用 `chatBoxRenderConfig.renderRoleAvatar` 渲染。`color` 传给 Avatar。                           | `RoleConfig`                                                                                       | -             |
| sendHotKey               | 发送快捷键，可选 `enter` 或 `shift+enter`。`enter` 时单独按 Enter 发送，Shift + Enter 换行；`shift+enter` 相反。                                              | `SendHotKey`                                                                                       | `"enter"`     |
| showClearContext         | 是否在输入框中显示清除上下文按钮。                                                                                                                            | `boolean`                                                                                          | `false`       |
| showStopGenerate         | 是否在最后一条消息生成中时显示停止生成按钮。生成中发送按钮不可用，点击停止按钮触发 `onStopGenerator`。                                                        | `boolean`                                                                                          | `false`       |
| topSlot                  | 对话区域顶部的节点。                                                                                                                                          | `JSX.Element`                                                                                      | -             |
| uploadProps              | 传给 Upload 的属性。`fileList` 与 `defaultFileList` 由 Chat 管理，`onChange` 由 Chat 处理，不会传给 Upload；`children` 作为上传按钮的自定义触发元素。         | <ApiType detail={"Omit\u003cUploadProps, 'fileList' \u007c 'defaultFileList'>"}>Type</ApiType>     | -             |
| uploadTipProps           | 上传按钮提示 Tooltip 的属性，`content` 为提示文字。                                                                                                           | `TooltipProps`                                                                                     | -             |

**Methods**

| 方法           | 说明                                                                                | 类型                                                                                  |
| -------------- | ----------------------------------------------------------------------------------- | ------------------------------------------------------------------------------------- |
| clearContext   | 在对话末尾追加分割线以清除上下文，并触发 `onClear`；末尾已是分割线时无效果。        | <ApiType detail={"() => void"}>Type</ApiType>                                         |
| resetMessage   | 将最后一条消息重置为空内容的 loading 状态，并触发 `onMessageReset`，参数为 `null`。 | <ApiType detail={"() => void"}>Type</ApiType>                                         |
| scrollToBottom | 滚动到最底部，`animation` 为 `true` 时平滑滚动，否则直接跳转。                      | <ApiType detail={"(animation?: boolean) => void"}>Type</ApiType>                      |
| sendMessage    | 发送消息，不传 `attachment` 时消息的内容为字符串。                                  | <ApiType detail={"(content: string, attachment?: FileItem[]) => void"}>Type</ApiType> |
