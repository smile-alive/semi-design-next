import ApiType from '#/components/ApiType.astro';

**Props**

| Property | Description | Type | Default |
| --- | --- | --- | --- |
| action | Snippet that customizes the message action bar, with `message`, the default buttons `defaultActions` (an array of snippets), `defaultActionsObj` indexed by name, and the class name `class`. | <ApiType detail={"Snippet\u003c[ChatActionSnippetProps]>"}>Type</ApiType> | - |
| align | Conversation layout: `leftRight` or `leftAlign`. | `ChatAlign` | `"leftRight"` |
| avatar | Snippet that customizes the message avatar, with `message`, `role` and the built-in avatar `defaultAvatar` (a snippet). Provided and rendering nothing, it hides the avatar. | <ApiType detail={"Snippet\u003c[ChatAvatarSnippetProps]>"}>Type</ApiType> | - |
| bottomSlot | Snippet for the content below the input box. | `Snippet` | - |
| canSend | Whether the send button is enabled. It is decided internally by default; when set, this value wins. | `boolean` | - |
| chats | The conversation list, bindable with `bind:chats`. When a message is sent, liked, reset, cleared or deleted, Chat writes the new list back first and then calls `onChatsChange`. Replacing the list, a `push` and in-place growth of the last message's text are followed; other in-place edits (a deep content edit, a lone status flip) render but are not followed, so replace the array as a whole. | `ChatMessage[]` | - |
| content | Snippet that customizes the message content area, with `message`, `role`, the built-in content `defaultContent` (a snippet) and the class name `class`. | <ApiType detail={"Snippet\u003c[ChatContentSnippetProps]>"}>Type</ApiType> | - |
| customMarkDownComponents | Custom Markdown render components, keyed by element name, passed to the MarkdownRender of the message content. | `MarkdownRenderComponents` | - |
| divider | Snippet that customizes the divider after the context is cleared, with `message`. | <ApiType detail={"Snippet\u003c[ChatDividerSnippetProps]>"}>Type</ApiType> | - |
| enableUpload | Whether upload is enabled. A boolean controls paste, click and drag upload together; an object sets `pasteUpload`, `clickUpload` and `dragUpload` separately, and an unset item is `true`. | `boolean \| ChatEnableUploadProps` | `true` |
| escapeHtml | Whether to escape HTML tags in the user's messages, so the Markdown parser does not treat them as HTML and drop the content. | `boolean` | `true` |
| fullChatBox | Snippet that customizes the whole message box, with `message`, `role`, the default nodes `defaultNodes` (snippets) and the class name `class`. | <ApiType detail={"Snippet\u003c[ChatFullChatBoxSnippetProps]>"}>Type</ApiType> | - |
| hintBox | Snippet that customizes a hint, with `content`, `index` and the click handler `onHintClick`. | <ApiType detail={"Snippet\u003c[ChatHintBoxSnippetProps]>"}>Type</ApiType> | - |
| hintCls | Class name of the outermost element of the hints area. | `string` | - |
| hints | The list of hints. Clicking one calls `onHintClick` and sends it as a new user message. | `string[]` | - |
| hintStyle | Style of the outermost element of the hints area, a string. | `string` | - |
| inputArea | Snippet that customizes the input area, with `defaultNode`, the part nodes (`clearContextNode`, `uploadNode`, `inputNode`, `sendNode`, all snippets), and `onSend`, `onClear` and `onClick`, which you call yourself. | <ApiType detail={"Snippet\u003c[ChatInputAreaSnippetProps]>"}>Type</ApiType> | - |
| inputBoxCls | Class name of the input box. | `string` | - |
| inputBoxStyle | Style of the input box, a string. | `string` | - |
| markdownRenderProps | Props passed to the MarkdownRender that renders the message content; the default format is md. `raw` and `components` cannot be set; use `customMarkDownComponents` for custom render components. | `ChatMarkdownRenderProps` | - |
| mode | Conversation mode: `bubble`, `noBubble` or `userBubble`. | `ChatMode` | `"bubble"` |
| onChatsChange | Called when the conversation list changes, after `chats` is written back. | <ApiType detail={"(chats: ChatMessage[]) => void"}>Type</ApiType> | - |
| onClear | Called when the context is cleared. | <ApiType detail={"() => void"}>Type</ApiType> | - |
| onHintClick | Called when a hint is clicked, with the hint text. | <ApiType detail={"(hint: string) => void"}>Type</ApiType> | - |
| onInputChange | Called when the input or the attachments change, with `{ inputValue, attachment }`. | <ApiType detail={"(props: ChatInputChangeProps) => void"}>Type</ApiType> | - |
| onMessageBadFeedback | Called when a message gets negative feedback. | <ApiType detail={"(message: ChatMessage) => void"}>Type</ApiType> | - |
| onMessageCopy | Called when a message is copied. | <ApiType detail={"(message: ChatMessage) => void"}>Type</ApiType> | - |
| onMessageDelete | Called when a message is deleted. | <ApiType detail={"(message: ChatMessage) => void"}>Type</ApiType> | - |
| onMessageGoodFeedback | Called when a message gets positive feedback. | <ApiType detail={"(message: ChatMessage) => void"}>Type</ApiType> | - |
| onMessageReset | Called when a message is reset; the argument is `null` when it comes from the `resetMessage` method. | <ApiType detail={"(message: ChatMessage \u007c null) => void"}>Type</ApiType> | - |
| onMessageSend | Called when a message is sent, with the text and the attachments. | <ApiType detail={"(content: string, attachment?: FileItem[]) => void"}>Type</ApiType> | - |
| onStopGenerator | Called when the stop button is clicked. | <ApiType detail={"(event: MouseEvent) => void"}>Type</ApiType> | - |
| placeholder | Placeholder of the input box. | `string` | - |
| roleAvatar | Snippet for the avatar content of a role without an image URL, with `message` and `role`. | <ApiType detail={"Snippet\u003c[ChatRoleAvatarSnippetProps]>"}>Type</ApiType> | - |
| roleConfig | Role information, keyed by the message role. `avatar` takes an image URL only; an avatar node goes through the `roleAvatar` snippet. `color` is passed to Avatar. | `ChatRoleConfig` | - |
| sendHotKey | Send hot key: `enter` or `shift+enter`. With `enter`, Enter sends and Shift+Enter inserts a line break; `shift+enter` is the reverse. | `SendHotKey` | `"enter"` |
| showClearContext | Whether to show the clear context button in the input box. | `boolean` | `false` |
| showStopGenerate | Whether to show the stop button while the last message is generating. Sending is disabled meanwhile, and clicking the stop button calls `onStopGenerator`. | `boolean` | `false` |
| title | Snippet that customizes the message title, with `message`, `role` and the built-in title `defaultTitle` (a snippet). Provided and rendering nothing, it hides the title. | <ApiType detail={"Snippet\u003c[ChatTitleSnippetProps]>"}>Type</ApiType> | - |
| topSlot | Snippet for the content at the top of the conversation area. | `Snippet` | - |
| uploadButton | Snippet that replaces the upload button. | `Snippet` | - |
| uploadProps | Props passed to Upload. Chat owns `fileList` and `children`, and the upload button content is the `uploadButton` snippet; Chat handles `onChange` itself when an attachment is added or removed, and the other callbacks pass through. `action` falls back to an empty string, so nothing is uploaded unless a `customRequest` or an `action` is given. | `ChatUploadProps` | - |
| uploadTip | Snippet for the upload button's tip text. | `Snippet` | - |
| uploadTipProps | Props of the upload button's Tooltip, without `content`; the tip text is the `uploadTip` snippet. | <ApiType detail={"Omit\u003cTooltipProps, 'content' \u007c 'children' \u007c 'child'>"}>Type</ApiType> | - |

**Methods**

| Method         | Description                                                                                                                                            | Type                                                                               |
| -------------- | ------------------------------------------------------------------------------------------------------------------------------------------------------ | ---------------------------------------------------------------------------------- |
| clearContext   | Appends a divider at the end of the conversation to clear the context and calls `onClear`; it has no effect when the list already ends with a divider. | <ApiType detail={"() => void"}>Type</ApiType>                                      |
| resetMessage   | Resets the last message to an empty loading state and calls `onMessageReset` with `null`.                                                              | <ApiType detail={"() => void"}>Type</ApiType>                                      |
| scrollToBottom | Scrolls to the bottom, smoothly when `animation` is `true`, otherwise it jumps.                                                                        | <ApiType detail={"(animation?: boolean) => void"}>Type</ApiType>                   |
| sendMessage    | Sends a message; without `attachment` the message content is a string.                                                                                 | <ApiType detail={"(text: string, attachment?: FileItem[]) => void"}>Type</ApiType> |
