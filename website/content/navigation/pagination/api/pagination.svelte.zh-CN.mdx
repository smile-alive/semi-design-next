import ApiType from '#/components/ApiType.astro';

**Props**

| 属性                              | 说明                                                                                                                                                                     | 类型                                                                               | 默认值              |
| --------------------------------- | ------------------------------------------------------------------------------------------------------------------------------------------------------------------------ | ---------------------------------------------------------------------------------- | ------------------- |
| currentPage                       | 当前页码，可用 `bind:currentPage` 双向绑定；不传时由组件自己管理页码，用 `bind:currentPage={get, set}` 完全受控。                                                        | `number`                                                                           | -                   |
| disabled                          | 是否禁用。                                                                                                                                                               | `boolean`                                                                          | `false`             |
| hideOnSinglePage                  | 总页数小于 2 时是否隐藏分页器；每页条数选择器显示时不生效。                                                                                                              | `boolean`                                                                          | `false`             |
| hoverShowPageSelect               | 仅 size 为 small 时生效：悬停或聚焦页码文字时打开页码列表。                                                                                                              | `boolean`                                                                          | -                   |
| nextText                          | 替换下一页按钮图标的字符串或 snippet，命名规则同 prevText。                                                                                                              | `string \| Snippet`                                                                | -                   |
| onChange                          | 页码或每页条数变化时调用，参数为页码与每页条数。                                                                                                                         | <ApiType detail={"(currentPage: number, pageSize: number) => void"}>Type</ApiType> | -                   |
| onPageChange                      | 页码变化时调用。                                                                                                                                                         | <ApiType detail={"(currentPage: number) => void"}>Type</ApiType>                   | -                   |
| onPageSizeChange                  | 每页条数变化时调用。                                                                                                                                                     | <ApiType detail={"(pageSize: number) => void"}>Type</ApiType>                      | -                   |
| pageSize                          | 每页条数，可用 `bind:pageSize` 绑定，选择器的选择会写回；未传时取 pageSizeOpts 的第一项。                                                                                | `number`                                                                           | -                   |
| pageSizeOpts                      | 每页条数选择器的可选值；当前每页条数不在其中时按大小插入选项。                                                                                                           | `number[]`                                                                         | `[10, 20, 40, 100]` |
| popoverPosition                   | 省略页码浮层与每页条数下拉的位置；size 为 small 时不使用。                                                                                                               | `Position`                                                                         | -                   |
| preventPageChangeOnPageSizeChange | 切换每页条数时保持当前页码，新页码由使用方在 onPageSizeChange 中决定。                                                                                                   | `boolean`                                                                          | `false`             |
| prevText                          | 替换上一页按钮图标的字符串或 snippet。字符串或纯文本 snippet 作为按钮名称，其他内容（图标、元素）仍使用语言包的名称；传入空 snippet 时按钮为空，空字符串则显示默认图标。 | `string \| Snippet`                                                                | -                   |
| showQuickJumper                   | 是否显示快速跳转输入框，总页数小于 2 时禁用。                                                                                                                            | `boolean`                                                                          | `false`             |
| showSizeChanger                   | 是否显示每页条数选择器，size 为 small 时不显示。                                                                                                                         | `boolean`                                                                          | `false`             |
| showTotal                         | 是否显示总页数，size 为 small 时不显示。                                                                                                                                 | `boolean`                                                                          | `false`             |
| size                              | 尺寸，可选 default 或 small。                                                                                                                                            | `'small' \| 'default'`                                                             | `"default"`         |
| total                             | 总条数。                                                                                                                                                                 | `number`                                                                           | `1`                 |
