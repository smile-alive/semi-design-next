import ApiType from '#/components/ApiType.astro';

**Props**

| Property | Description | Type | Default |
| --- | --- | --- | --- |
| className | Class name of the root `ul` element. | `string` | `""` |
| currentPage | Current page. Once set the pagination is controlled and the page is entirely determined by this prop; when undefined the component manages the page itself. | `number` | - |
| defaultCurrentPage | Initial page when not controlled. | `number` | `1` |
| disabled | Whether disabled. When disabled, the navigation buttons, the page numbers, the size changer and the quick jumper are all unavailable. | `boolean` | `false` |
| hideOnSinglePage | Whether to hide the pagination when there are fewer than 2 pages; has no effect while the size changer is shown. | `boolean` | `false` |
| hoverShowPageSelect | Only when `size` is `small`: the page list opens when the page text is hovered or focused. | `boolean` | - |
| nextText | Replaces the content of the next page button, named the same way as `prevText`. | `React.ReactNode` | - |
| onChange | Called when the page or the page size changes, with the page and the page size. | <ApiType detail={"(currentPage: number, pageSize: number) => void"}>Type</ApiType> | - |
| onPageChange | Called when the page changes, with the new page. | <ApiType detail={"(currentPage: number) => void"}>Type</ApiType> | - |
| onPageSizeChange | Called when the page size changes, with the new page size. | <ApiType detail={"(newPageSize: number) => void"}>Type</ApiType> | - |
| pageSize | Number of items per page; falls back to the first item of `pageSizeOpts`. Not strictly controlled: the choice made in the size changer is kept inside, and only a change of this prop to a value other than the current one overrides it and goes back to page 1 (to `currentPage` when controlled). Writing it back from `onChange` does not reset the page. | `number` | `null` |
| pageSizeOpts | Options of the size changer; the current page size is inserted in order when it is not among them. | <ApiType detail={"Array\u003cnumber>"}>Type</ApiType> | `[10, 20, 40, 100]` |
| popoverPosition | Position of the omitted pages popover and the size changer dropdown. When omitted the size changer dropdown uses `bottomLeft` (`bottomRight` in RTL) and the omitted pages popover uses the Popover default `bottom`; not used when `size` is `small`. | `PopoverPosition` | - |
| preventPageChangeOnPageSizeChange | Keeps the current page when the page size changes without calling `onPageChange`; the consumer picks the new page in `onPageSizeChange`. | `boolean` | `false` |
| prevText | Replaces the content of the previous page button. A plain text names the button, and any other content (an icon, an element) keeps the locale name; empty content keeps the default icon and the locale name. | `React.ReactNode` | - |
| ref | Ref of the root `ul` element, in both sizes. | <ApiType detail={"React.Ref\u003cHTMLUListElement>"}>Type</ApiType> | - |
| showQuickJumper | Whether to show the quick jump input; disabled when there are fewer than 2 pages. | `boolean` | `false` |
| showSizeChanger | Whether to show the size changer; has no effect when `size` is `small`. When it is shown, `hideOnSinglePage` no longer takes effect. | `boolean` | `false` |
| showTotal | Whether to show the total number of pages; not shown when `size` is `small`. | `boolean` | `false` |
| size | Size: `default` or `small`. | `'small' \| 'default'` | `"default"` |
| style | Inline style of the root `ul` element. | `React.CSSProperties` | - |
| total | Total number of items. When it changes, an uncontrolled pagination goes back to page 1; when controlled, the page follows `currentPage`. | `number` | `1` |
