import ApiType from '#/components/ApiType.astro';

**Props**

| 属性 | 说明 | 类型 | 默认值 |
| --- | --- | --- | --- |
| currentPage | 当前页码。传入后分页器受控，页码完全由该属性决定；不传（undefined）时由组件自己管理。 | `number` | - |
| defaultCurrentPage | 非受控时的初始页码。 | `number` | `1` |
| disabled | 是否禁用。 | `boolean` | `false` |
| hideOnSinglePage | 总页数小于 2 时是否隐藏分页器；每页条数选择器显示时不生效。 | `boolean` | `false` |
| hoverShowPageSelect | 仅 size 为 small 时生效：悬停或聚焦页码文字时打开页码列表。 | `boolean` | - |
| nextText | 替换下一页按钮的内容，命名规则同 `prevText`。 | `JSX.Element` | - |
| onChange | 页码或每页条数变化时的回调，参数为页码与每页条数。 | <ApiType detail={"(currentPage: number, pageSize: number) => void"}>Type</ApiType> | - |
| onPageChange | 页码变化时的回调，参数为新的页码。 | <ApiType detail={"(currentPage: number) => void"}>Type</ApiType> | - |
| onPageSizeChange | 每页条数变化时的回调，参数为新的每页条数。 | <ApiType detail={"(newPageSize: number) => void"}>Type</ApiType> | - |
| pageSize | 每页条数，未传时取 `pageSizeOpts` 的第一项。非严格受控：每页条数选择器的选择保留在组件内部，只有该属性变为与当前不同的值时才会覆盖它，并回到第 1 页（受控时回到 `currentPage`）；在 `onChange` 中回写该属性不会重置页码。 | `number` | - |
| pageSizeOpts | 每页条数选择器的可选值；当前每页条数不在其中时按大小插入选项。 | <ApiType detail={"Array\u003cnumber>"}>Type</ApiType> | `[10, 20, 40, 100]` |
| popoverPosition | 省略页码浮层与每页条数下拉的位置。未传时每页条数下拉为 `bottomLeft`（RTL 下 `bottomRight`），省略页码浮层使用 Popover 的默认位置 `bottom`；size 为 small 时不使用。 | `PopoverPosition` | - |
| preventPageChangeOnPageSizeChange | 切换每页条数时保持当前页码，新页码由使用方在 `onPageSizeChange` 中决定。 | `boolean` | `false` |
| prevText | 替换上一页按钮的内容。纯文本时按钮以该文字命名，其他内容（图标、元素）仍使用语言包的名称；为空时保留默认图标与语言包名称。 | `JSX.Element` | - |
| showQuickJumper | 是否显示快速跳转输入框，总页数小于 2 时禁用。 | `boolean` | `false` |
| showSizeChanger | 是否显示每页条数选择器，size 为 small 时不生效。开启后 hideOnSinglePage 不再生效。 | `boolean` | `false` |
| showTotal | 是否显示总页数，size 为 small 时不显示。 | `boolean` | `false` |
| size | 尺寸，可选 `default` 或 `small`。 | `'small' \| 'default'` | `"default"` |
| total | 总条数。 | `number` | `1` |
