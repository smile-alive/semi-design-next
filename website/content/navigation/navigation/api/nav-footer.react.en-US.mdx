import ApiType from '#/components/ApiType.astro';

**Props**

| Property       | Description                                                                                                                                                       | Type                                                                        | Default |
| -------------- | ----------------------------------------------------------------------------------------------------------------------------------------------------------------- | --------------------------------------------------------------------------- | ------- |
| children       | Content of the footer. When it has children, `collapseButton` does not apply.                                                                                     | `ReactNode`                                                                 | -       |
| className      | Class name of the outermost `div`.                                                                                                                                | `string`                                                                    | -       |
| collapseButton | Whether to show the collapse button. `true` shows the built-in button; a node replaces it. Applies only when `mode` is `vertical` and the footer has no children. | `React.ReactNode`                                                           | `false` |
| collapseText   | Text of the built-in collapse button, given the collapsed state. A text-only result also names the icon-only button.                                              | <ApiType detail={"(collapsed?: boolean) => React.ReactNode"}>Type</ApiType> | -       |
| onClick        | Called when the footer is clicked.                                                                                                                                | <ApiType detail={"(event: React.MouseEvent) => void"}>Type</ApiType>        | -       |
| ref            | Reference to the outermost `div` of the footer.                                                                                                                   | <ApiType detail={"React.Ref\u003cHTMLDivElement>"}>Type</ApiType>           | -       |
| style          | Style of the outermost `div`.                                                                                                                                     | `CSSProperties`                                                             | -       |
