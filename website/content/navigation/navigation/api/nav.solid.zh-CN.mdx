import ApiType from '#/components/ApiType.astro';

**Props**

| 属性 | 说明 | 类型 | 默认值 |
| --- | --- | --- | --- |
| bodyStyle | 导航项列表的自定义样式。 | `JSX.CSSProperties \| string` | - |
| defaultIsCollapsed | 默认是否处于收起状态，仅 `mode` 为 `vertical` 时有效。 | `boolean` | - |
| defaultOpenKeys | 初始展开的子导航 `itemKey` 数组，仅 `mode` 为 `vertical` 且侧边栏处于展开状态时有效。 | `ItemKey[]` | - |
| defaultSelectedKeys | 初始选中的导航项 `itemKey` 数组。 | `ItemKey[]` | - |
| expandIcon | `items` 生成的子导航的展开收起箭头图标。传了 `renderItemExpandIcon` 时以它为准。 | `JSX.Element` | - |
| footer | 底部区域：配置对象、JSX 或字符串。配置对象只放数据：`collapseButton` 为 `true` 时展示内置按钮，也可以是返回替代按钮的渲染函数；另有 `collapseText`，以及 `class`、`style` 等属性。也可以使用 Nav.Footer 子元素。 | `JSX.Element \| NavFooterConfig` | - |
| header | 头部区域：配置对象、JSX 或字符串。配置对象只放数据：`text` 为字符串或数字，`logo` 为返回 Logo 的渲染函数，只在显示时调用；另有 `link`、`linkOptions`，以及 `class`、`style` 等属性。也可以使用 Nav.Header 子元素。 | `JSX.Element \| NavHeaderConfig` | - |
| isCollapsed | 是否处于收起状态的受控值，仅 `mode` 为 `vertical` 时有效。配合 `onCollapseChange` 使用。 | `boolean` | - |
| items | 导航项数据，带 `items` 的项渲染为子导航。字符串项同时作为 `text` 和 `itemKey`。条目只放数据，图标、显示的文案与箭头由 `renderItemIcon`、`renderItemText`、`renderItemToggleIcon`、`renderItemExpandIcon` 提供。只有 Nav.Item 与 Nav.Sub 声明的数据属性、`class`、`style`、事件回调，以及 `data-*` 与 `aria-*` 会传给导航项，其余字段（如 `path`）会被忽略。 | `NavItems` | - |
| limitIndent | 是否限制缩进。设为 `false` 后多级导航按 `level` 缩进，仅 `mode` 为 `vertical` 时生效，水平模式始终限制缩进。 | `boolean` | `true` |
| mode | 导航方向，可选 `vertical` 或 `horizontal`。 | `Mode` | `"vertical"` |
| onClick | 点击任意导航项时触发。数据含 `itemKey`、`text`，点击子导航标题时还含 `openKeys` 与 `isOpen`；`domEvent` 为原生事件。 | <ApiType detail={"(data: OnClickData) => void"}>Type</ApiType> | - |
| onCollapseChange | 收起状态变化时的回调，参数为变化后的收起状态。 | <ApiType detail={"(isCollapse: boolean) => void"}>Type</ApiType> | - |
| onOpenChange | 切换子导航展开状态时触发。数据含 `itemKey`、`openKeys`、`isOpen`，`domEvent` 为原生事件；悬停打开浮层时没有 `domEvent`。 | <ApiType detail={"(data: OnOpenChangeData) => void"}>Type</ApiType> | - |
| onSelect | 第一次选中某个可选中导航项时触发。数据含 `itemKey`、`selectedKeys`、`selectedItems`，`domEvent` 为原生事件。 | <ApiType detail={"(data: OnSelectedData) => void"}>Type</ApiType> | - |
| openKeys | 展开的子导航 `itemKey` 数组的受控值，配合 `onOpenChange` 使用，仅 `mode` 为 `vertical` 且侧边栏处于展开状态时有效。 | `ItemKey[]` | - |
| renderItemExpandIcon | `items` 生成的子导航的展开收起箭头，入参为 `{ item, level }`，只在箭头显示时调用。传入后替代 `expandIcon`。 | <ApiType detail={"(props: NavItemRenderProps) => JSX.Element"}>Type</ApiType> | - |
| renderItemIcon | `items` 中每一项的图标，入参为 `{ item, level }`（`level` 为嵌套层级，顶层为 0），只在该项显示时调用。 | <ApiType detail={"(props: NavItemRenderProps) => JSX.Element"}>Type</ApiType> | - |
| renderItemText | `items` 中每一项显示的文案，入参为 `{ item, level }`。不传时显示条目的 `text`。 | <ApiType detail={"(props: NavItemRenderProps) => JSX.Element"}>Type</ApiType> | - |
| renderItemToggleIcon | `items` 生成的导航项的切换图标，入参为 `{ item, level }`。 | <ApiType detail={"(props: NavItemRenderProps) => JSX.Element"}>Type</ApiType> | - |
| renderWrapper | 包裹每个导航项的渲染函数，入参为 `{ itemElement, isSubNav, isInSubNav, props }`，返回值放进列表。`itemElement` 只能渲染一次。 | <ApiType detail={"(data: RenderWrapperData) => JSX.Element"}>Type</ApiType> | - |
| selectedKeys | 选中的导航项 `itemKey` 数组的受控值，配合 `onSelect` 使用。 | `ItemKey[]` | - |
| subDropdownProps | `items` 生成的子导航在水平模式或收起状态下弹层 Dropdown 的参数。子导航自己的 `dropdownProps` 或 `subDropdownProps` 优先。 | `DropdownProps` | - |
| subNavCloseDelay | 子导航浮层关闭的延迟，单位为毫秒。收起或水平模式下有效。 | `number` | `100` |
| subNavMotion | 子导航展开收起时是否有动画。 | `boolean` | `true` |
| subNavOpenDelay | 子导航浮层显示的延迟，单位为毫秒。收起或水平模式下有效。 | `number` | `0` |
| toggleIconPosition | 带有子导航项的父级导航项的箭头位置，可选 `left` 或 `right`。 | `'left' \| 'right'` | `"right"` |
| tooltipHideDelay | 收起状态下 Tooltip 隐藏的延迟，单位为毫秒。 | `number` | `100` |
| tooltipShowDelay | 收起状态下 Tooltip 显示的延迟，单位为毫秒。 | `number` | `0` |
