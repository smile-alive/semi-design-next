import ApiType from '#/components/ApiType.astro';

**Props**

| 属性 | 说明 | 类型 | 默认值 |
| --- | --- | --- | --- |
| activeKey | 当前激活的标签页的 itemKey，可用 `bind:activeKey` 双向绑定。传 `null` 时不激活任何标签页，仍是受控；不传或为 `undefined` 时按非受控处理，默认选中第一个启用的标签页。 | `string \| null` | - |
| arrow | 替换折叠模式下每个滚动箭头的 snippet，参数为 `{ items, position, onClick, defaultNode }`：被滚出的标签页、位置、点击处理函数和默认箭头（用 `{@render defaultNode()}` 渲染）。 | <ApiType detail={"Snippet\u003c[TabsArrowSnippetProps]>"}>Type</ApiType> | - |
| arrowPosition | 折叠模式下，左右切换箭头的位置。 | `'both' \| 'start' \| 'end'` | `"both"` |
| children | 没有 `panel` 时的内容区：当前标签页的内容，是以所选标签命名的单个 tabpanel。 | `Snippet` | - |
| collapsible | 是否开启滚动折叠（仅水平模式）。传 `'auto'` 时在标签溢出或换行时自动折叠，放得下时自动退出。 | `boolean \| 'auto'` | `false` |
| contentStyle | 内容区域外层的样式字符串。 | `string` | - |
| dropdownProps | 折叠模式下，分别透传给起始侧与末尾侧箭头下拉菜单的 Dropdown 属性。 | `TabsDropdownProps` | - |
| icon | `tabList` 中每个标签页图标的 snippet，参数为 `{ item, index }`；图标容器始终渲染，即使 snippet 没有输出。 | <ApiType detail={"Snippet\u003c[TabsTabSnippetProps]>"}>Type</ApiType> | - |
| keepDOM | 使用 `panel` 时，是否渲染未激活面板的 DOM。为 false 时只渲染当前面板，没有切换动画。 | `boolean` | `true` |
| lazyRender | 懒渲染，面板被激活过才渲染进 DOM。 | `boolean` | `false` |
| more | 将末尾的一部分标签页收入「更多」下拉菜单。传数字表示收入菜单的标签数；传对象时 `count` 为标签数，`dropdownProps` 透传给下拉菜单。触发器与菜单内容分别用 `moreTrigger` 与 `moreContent` snippet 自定义。 | `number \| TabsMore` | - |
| moreContent | 替换「更多」菜单内容的 snippet，参数 `{ items }` 为收入菜单的标签页。 | <ApiType detail={"Snippet\u003c[TabsMoreContentSnippetProps]>"}>Type</ApiType> | - |
| moreTrigger | 替换「更多」菜单触发器的 snippet。 | `Snippet` | - |
| onChange | 切换标签页时调用，参数为新激活的 itemKey。 | <ApiType detail={"(activeKey: string) => void"}>Type</ApiType> | - |
| onTabClick | 单击，或用 Enter、Space 激活标签页时调用；用键盘激活时事件是 KeyboardEvent。 | <ApiType detail={"(activeKey: string, event: MouseEvent \u007c KeyboardEvent) => void"}>Type</ApiType> | - |
| onTabClose | 关闭标签页时调用，参数为被关闭标签页的 itemKey。标签页的移除由使用方完成。 | <ApiType detail={"(tabKey: string) => void"}>Type</ApiType> | - |
| onVisibleTabsChange | 折叠模式下溢出项的可见状态变化时调用，参数为 itemKey 到是否可见的 Map。 | <ApiType detail={"(visibleState: Map\u003cstring, boolean>) => void"}>Type</ApiType> | - |
| panel | `tabList` 中每个标签页内容的 snippet，参数为 `{ item, index, itemKey }`；传入后每个标签页都是各自的 tabpanel，`keepDOM`、`lazyRender` 与切换动画对它生效。 | <ApiType detail={"Snippet\u003c[TabsPanelSnippetProps]>"}>Type</ApiType> | - |
| preventScroll | 组件内部调用 focus 时，是否阻止浏览器滚动文档以显示新聚焦的元素。 | `boolean` | - |
| showRestInDropdown | 折叠模式下，是否将被收起的标签页展示在下拉菜单中。 | `boolean` | `true` |
| size | 标签栏尺寸，目前仅线条式（line）标签栏支持。 | `TabSize` | `"large"` |
| tab | `tabList` 中每个标签页标签的 snippet，参数为 `{ item, index }`；不提供时显示该项的 `tab` 文本。 | <ApiType detail={"Snippet\u003c[TabsTabSnippetProps]>"}>Type</ApiType> | - |
| tabBar | 替换整个标签栏的 snippet。参数 `{ tabBarProps, tabBar }`：`tabBarProps` 只含数据，`tabBar` 是默认标签栏，用 `{@render tabBar()}` 渲染；自定义标签栏里的 TabItem 会自行控制所在的 Tabs。 | <ApiType detail={"Snippet\u003c[TabsTabBarSnippetProps]>"}>Type</ApiType> | - |
| tabBarClassName | 标签栏的类名。 | `string` | - |
| tabBarExtraContent | 标签栏附加内容的 snippet，位于 tablist 之外。 | `Snippet` | - |
| tabBarStyle | 标签栏的样式字符串。 | `string` | - |
| tabList | 标签页的数据数组，每项含 itemKey、tab、disabled、closable，只放数据；标签内的节点用 `tab` 与 `icon` snippet，内容用 `panel` snippet 或 `children`。 | `PlainTab[]` | `[]` |
| tabPaneMotion | 是否使用动画切换面板。 | `boolean` | `true` |
| tabPosition | 标签栏位置：top 为水平，left 为垂直。 | `TabPosition` | `"top"` |
| type | 标签栏样式：line、card、button、slash。slash 仅支持水平模式。 | `TabType` | `"line"` |
| visibleTabsStyle | 折叠模式下，整体滚动区域的样式字符串。 | `string` | - |
