import ApiType from '#/components/ApiType.astro';

**Props**

| 属性 | 说明 | 类型 | 默认值 |
| --- | --- | --- | --- |
| activeKey | 当前激活的标签页的 itemKey，受控。传 null 时不激活任何标签页，仍是受控；不传或为 undefined 时按非受控处理。 | `string \| null` | - |
| aria-label | 标签栏的可访问名称，落在 `role="tablist"` 元素上。 | `string` | - |
| aria-labelledby | 指向标签栏名称元素的 id，落在 `role="tablist"` 元素上。 | `string` | - |
| arrowPosition | 折叠模式下，左右切换箭头的位置。 | `TabBarProps['arrowPosition']` | `"both"` |
| children | TabPane 子元素，必须是 Tabs 的直接子元素；其余节点按原顺序作为内容渲染。传了 tabList 时，TabPane 被忽略，其余节点是当前标签页的内容。 | `JSX.Element` | - |
| collapsible | 是否开启滚动折叠（仅水平模式）。传 'auto' 时在标签溢出或换行时自动折叠，放得下时自动退出。 | `boolean \| 'auto'` | `false` |
| contentStyle | 内容区域外层的样式，可传对象或字符串。 | `JSX.CSSProperties \| string` | - |
| defaultActiveKey | 非受控时初始选中的 itemKey。不传时取第一个启用且有 itemKey 的标签页。 | `string` | - |
| dropdownProps | 折叠模式下，分别透传给起始侧（start）与末尾侧（end）箭头下拉菜单的 Dropdown 属性。 | `TabsDropdownProps` | - |
| keepDOM | 使用 TabPane 子元素时，是否渲染未激活面板的 DOM。为 false 时只渲染当前面板，没有切换动画。 | `boolean` | `true` |
| lazyRender | 懒渲染，面板被激活过才渲染内容，之后保留。 | `boolean` | `false` |
| more | 将末尾的一部分标签页收入「更多」下拉菜单。传数字表示收入菜单的标签数；传对象时 count 为标签数，render 返回自定义的触发器，dropdownProps 透传给下拉菜单。 | `TabsMore` | - |
| onChange | 切换标签页时触发，参数为新激活的 itemKey。 | <ApiType detail={"(activeKey: string) => void"}>Type</ApiType> | - |
| onTabClick | 单击，或用 Enter、Space 激活标签页时触发；用键盘激活时事件是 KeyboardEvent。 | <ApiType detail={"(activeKey: string, event: MouseEvent \u007c KeyboardEvent) => void"}>Type</ApiType> | - |
| onTabClose | 关闭标签页时触发，参数为被关闭标签页的 itemKey。标签页的移除由使用方完成。 | <ApiType detail={"(tabKey: string) => void"}>Type</ApiType> | - |
| onVisibleTabsChange | 折叠模式下溢出项的可见状态变化时触发，参数为 itemKey 到是否可见的 Map。 | `TabBarProps['onVisibleTabsChange']` | - |
| preventScroll | 组件内部调用 focus 时，是否阻止浏览器滚动文档以显示新聚焦的元素。 | `boolean` | - |
| renderArrow | 替换折叠模式下的每个滚动箭头。参数依次为被滚出的 items、位置 pos、点击处理函数 handleArrowClick 与创建默认箭头的函数 defaultNode，在插入处调用 defaultNode()。items 是普通数据；TabPane 写法下不带 tab，需要标签文本时用 tabList。 | `TabBarProps['renderArrow']` | - |
| renderIcon | tabList 每一项的图标，参数为该项与其下标。在页签、菜单项等每个插入处各调用一次，该项数据或下标变化时重新调用。 | `TabRender` | - |
| renderTab | tabList 每一项的标签内容，参数为该项与其下标；不传时显示该项的 tab。在页签、菜单项与面板名称等每个插入处各调用一次，该项数据或下标变化时重新调用。 | `TabRender` | - |
| renderTabBar | 替换整个标签栏，同一个函数只调用一次。tabBarProps 是活的标签栏属性，在返回的 JSX 中读取；DefaultTabBar 是默认标签栏组件。 | <ApiType detail={"(tabBarProps: TabBarProps, DefaultTabBar: Component\u003cTabBarProps>) => JSX.Element"}>Type</ApiType> | - |
| showRestInDropdown | 折叠模式下，是否将被收起的标签页展示在下拉菜单中。 | `boolean` | `true` |
| size | 标签栏尺寸，目前仅线条式（line）标签栏支持。 | `TabSize` | `"large"` |
| tabBarClassName | 标签栏的类名。 | `string` | - |
| tabBarExtraContent | 标签栏的附加内容，位于 tablist 之外。 | `JSX.Element` | - |
| tabBarStyle | 标签栏的样式，可传对象或字符串。 | `JSX.CSSProperties \| string` | - |
| tabList | 标签页的数据数组，每项只含数据：itemKey、tab（文字）、disabled、closable，节点内容交给 renderTab、renderIcon。只要传了数组（含空数组），children 就是当前标签页的内容，并优先于 TabPane 子元素。 | `PlainTab[]` | - |
| tabPaneMotion | 是否使用动画切换面板。 | `boolean` | `true` |
| tabPosition | 标签栏位置：top 为水平，left 为垂直。 | `TabPosition` | `"top"` |
| type | 标签栏样式：line、card、button、slash。slash 仅支持水平模式。 | `TabType` | `"line"` |
| visibleTabsStyle | 折叠模式下，整体滚动区域的样式，可传对象或字符串。 | `TabBarProps['visibleTabsStyle']` | - |
