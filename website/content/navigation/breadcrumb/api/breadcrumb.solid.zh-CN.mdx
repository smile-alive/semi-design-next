import ApiType from '#/components/ApiType.astro';

**Props**

| 属性 | 说明 | 类型 | 默认值 |
| --- | --- | --- | --- |
| activeIndex | 受控使用，当前项的序号（从 0 起）；缺省时最后一项为当前项。 | `number` | - |
| autoCollapse | 超出 maxItemCount 后是否自动折叠。 | `boolean` | `true` |
| compact | 是否使用紧凑尺寸，影响图标与文字的尺寸。 | `boolean` | `true` |
| maxItemCount | 超出多少个级别后自动折叠。 | `number` | `4` |
| moreType | 内置省略号区域的渲染类型，可选 default、popover；提供 renderMore 时以 renderMore 为准。 | `MoreType` | `"default"` |
| onClick | 点击时调用。routes 写法收到该项的路由；Breadcrumb.Item 写法收到 `{ name, href }`，name 是该项的子节点。第二个参数为原生 MouseEvent。传入后没有 href 的项也可以用 Enter 操作。 | <ApiType detail={"(route: BreadcrumbItemInfo \u007c RouteProps, event: MouseEvent) => void"}>Type</ApiType> | - |
| renderIcon | 自定义 routes 中每一项的图标，参数为该项的路由对象与序号（含被折叠的项）。 | <ApiType detail={"(route: RouteProps, index: number) => JSX.Element"}>Type</ApiType> | - |
| renderItem | 自定义每一项的内容，配合 routes 使用，参数为该项的路由对象（字符串路由为 `{ name }`）。 | <ApiType detail={"(route: RouteProps) => JSX.Element"}>Type</ApiType> | - |
| renderMore | 自定义省略号区域的内容，参数为被折叠的项的数据：routes 写法下是路由对象，children 写法下是各 Breadcrumb.Item 的 props；提供时以它为准。 | <ApiType detail={"(restItems: Array\u003cRouteProps \u007c BreadcrumbItemProps>) => JSX.Element"}>Type</ApiType> | - |
| routes | 路由信息，由路由对象或字符串组成的数组；不为空时按 routes 渲染，不再渲染 children。 | <ApiType detail={"Array\u003cRouteProps \u007c string>"}>Type</ApiType> | `[]` |
| separator | 自定义的分隔符，可以是文字或 JSX 元素。 | `JSX.Element` | `"/"` |
| showTooltip | 是否在标题溢出时展示 Tooltip 及相关配置：width 为截断宽度，数字按 px 计算，传 'auto' 时不截断；ellipsisPos 为截断位置，end 或 middle；opts 为传给 Tooltip 的属性，不含 content。传 false 时不展示 Tooltip。 | `boolean \| showToolTipProps` | `{ width: 150, ellipsisPos: 'end' }` |
