import ApiType from '#/components/ApiType.astro';

**Props**

| Property   | Description                                                                                                                                                                   | Type                                                              | Default     |
| ---------- | ----------------------------------------------------------------------------------------------------------------------------------------------------------------------------- | ----------------------------------------------------------------- | ----------- |
| aria-label | Accessible name of the steps; the root element gets `role="group"` when it is set.                                                                                            | `string`                                                          | -           |
| children   | Steps, usually `Steps.Step`. Each React element takes one step position; text and empty values are not counted. Native elements render as they are and receive no step props. | `React.ReactNode`                                                 | -           |
| className  | Class name of the root element.                                                                                                                                               | `string`                                                          | -           |
| current    | Current step, counting from 0; defaults to `0`. A child Step can override its own state with `status`.                                                                        | `number`                                                          | -           |
| direction  | Direction of the steps: `horizontal` or `vertical`, defaults to `horizontal`. Applies to the `fill` and `basic` types; ignored by `nav`.                                      | `Direction`                                                       | -           |
| hasLine    | Whether to show the line between steps, defaults to `true`. Applies to the `basic` type only.                                                                                 | `boolean`                                                         | -           |
| initial    | Starting index, counting from 0; defaults to `0`. A step's number equals `initial` plus its index among the children.                                                         | `number`                                                          | -           |
| onChange   | Called when a step is selected, with its number (`initial` plus index); not called for the current step. Passing it makes every step clickable.                               | <ApiType detail={"(current: number) => void"}>Type</ApiType>      | -           |
| prefixCls  | Class name prefix, used internally to generate style class names.                                                                                                             | `string`                                                          | -           |
| ref        | Reference to the root element (`div`).                                                                                                                                        | <ApiType detail={"React.Ref\u003cHTMLDivElement>"}>Type</ApiType> | -           |
| size       | Size: `small` or `default`, defaults to `default`. Applies to the `basic` and `nav` types.                                                                                    | `Size`                                                            | `"default"` |
| status     | Status of the current step: `wait`, `process`, `finish`, `error` or `warning`, defaults to `process`. Applies to the `fill` and `basic` types.                                | `Status`                                                          | -           |
| style      | Style of the root element.                                                                                                                                                    | `React.CSSProperties`                                             | -           |
| type       | Steps type: `fill`, `basic` or `nav`.                                                                                                                                         | `'fill' \| 'basic' \| 'nav'`                                      | `"fill"`    |
