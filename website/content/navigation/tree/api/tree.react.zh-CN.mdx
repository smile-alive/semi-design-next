import ApiType from '#/components/ApiType.astro';

**Props**

| 属性 | 说明 | 类型 | 默认值 |
| --- | --- | --- | --- |
| aria-label | 树的可访问名称，设在 `role="tree"` 的元素上；空状态下不渲染该元素。 | `string` | - |
| autoExpandParent | expandedKeys 变化时是否自动展开其中节点的父节点；挂载时总会展开。开启后，展开受控时要先收起所有已展开的子节点才能收起父节点。 | `boolean` | `false` |
| autoExpandWhenDragEnter | 拖拽到收起的节点上时是否自动展开该节点。 | `boolean` | `true` |
| autoMergeValue | 父节点被选中时，值不再包含它的子孙节点；leafOnly 为 false 时生效。 | `boolean` | `true` |
| blockNode | 节点显示为整行，悬浮和选中高亮整行；关闭时只高亮节点的 label。 | `boolean` | `true` |
| checkRelation | 多选时节点之间选中状态的关系：`related` 为父子联动，`unRelated` 为互不影响。 | `CheckRelation` | `"related"` |
| children | 当前实现不渲染 children，树节点请通过 treeData 提供。 | `ReactNode` | - |
| className | 根元素的类名。 | `string` | - |
| defaultExpandAll | 初始化时展开所有节点。之后 treeData 或 treeDataSimpleJson 的变化不再影响展开状态，需要时使用 expandAll。 | `boolean` | `false` |
| defaultExpandedKeys | 默认展开的节点，显示其直接子级。 | `string[]` | - |
| defaultValue | 指定默认选中的条目，只在挂载时读取。 | `Value` | - |
| directory | 目录树模式，节点默认显示文件夹或文件图标；节点数据中的 `icon` 与 `icon` 属性可以覆盖它。 | `boolean` | - |
| disabled | 禁用整棵树。节点数据中的 `disabled: false` 不会解除禁用。 | `boolean` | `false` |
| disableStrictly | 严格禁用：禁用的节点不能通过父级或子级的选中关系改变状态。 | `boolean` | `false` |
| draggable | 是否允许拖拽节点，配合 onDrop 等拖拽回调使用。 | `boolean` | `false` |
| emptyContent | 没有节点，或开启 `showFilteredOnly` 后搜索无结果时显示的内容，未设置时取语言包中的空状态文案；为 `null` 时不渲染空状态。 | `ReactNode` | - |
| expandAction | 展开逻辑：`false` 仅点击展开箭头时展开，`click` 单击节点展开，`doubleClick` 双击节点展开。键盘不使用该逻辑。 | `ExpandAction` | `false` |
| expandAll | 展开所有节点，数据变化后仍然生效。 | `boolean` | `false` |
| expandedKeys | （受控）展开的节点。传入后展开状态完全受控，配合 onExpand 使用。 | `string[]` | - |
| expandIcon | 自定义展开图标。传入元素时，Tree 为它加上展开的点击处理与展开图标的类名；传入函数时，参数为 `onClick`、`className` 和 `expanded`，返回整个图标；其他 ReactNode 原样渲染，不绑定展开点击。 | `ExpandIcon` | - |
| filterTreeNode | 是否开启搜索。传 `true` 时按 treeNodeFilterProp 匹配；传函数时自定义匹配，第三个参数为当前节点数据。 | <ApiType detail={"boolean \u007c ((inputValue: string, treeNodeString: string, data?: TreeNodeData) => boolean)"}>Type</ApiType> | `false` |
| hideDraggingNode | 拖拽时隐藏被拖拽节点的拖拽图像。 | `boolean` | - |
| icon | 自定义节点图标。传入元素时所有节点使用同一图标；传入函数时每行调用一次，参数为该行的属性（含节点数据），返回该行的图标。节点数据中的 `icon` 优先于此属性。 | <ApiType detail={"ReactNode \u007c ((props: TreeNodeProps) => ReactNode)"}>Type</ApiType> | - |
| keyMaps | 自定义节点中 key、label、value 与 children 的字段名。映射了 label 且开启搜索时，需把 treeNodeFilterProp 设为 treeData 的键之一，或自定义 filterTreeNode。 | `KeyMapProps` | - |
| labelEllipsis | label 超出时是否省略；未设置时跟随 `virtualize`，开启虚拟化则省略。 | `boolean` | - |
| leafOnly | 多选时，值和 onChange 的参数只包含叶子节点。 | `boolean` | `false` |
| loadData | 异步加载子节点，返回 Promise。叶子节点需在数据中用 isLeaf 标明；设置后未标明 isLeaf 的节点显示展开箭头。 | <ApiType detail={"(treeNode?: TreeNodeData) => Promise\u003cvoid>"}>Type</ApiType> | - |
| loadedKeys | （受控）已加载的节点，配合 loadData 使用。 | `string[]` | - |
| motion | 是否开启展开收起的动画；虚拟化时动画关闭。 | `boolean` | `true` |
| multiple | 是否多选。多选时所有子项都被选择会自动勾选其父项。 | `boolean` | `false` |
| onChange | 选中状态变化时调用，参数为当前选中的值（多选时为数组）；`onChangeWithObject` 为 `true` 时参数为节点对象。 | <ApiType detail={"(value: Value) => void"}>Type</ApiType> | - |
| onChangeWithObject | 是否把选中节点的其他属性一并给 onChange。为 `true` 时参数由 value 变为包含 value、label 及其他属性的节点对象；受控的 value 与 defaultValue 也要传对象，且必须含 value。 | `boolean` | `false` |
| onContextMenu | 在节点上右键时调用，参数为事件和节点数据。 | <ApiType detail={"(e: MouseEvent, node: TreeNodeData) => void"}>Type</ApiType> | - |
| onDoubleClick | 双击节点时调用，参数为事件和节点数据。 | <ApiType detail={"(e: MouseEvent, node: TreeNodeData) => void"}>Type</ApiType> | - |
| onDragEnd | 拖拽结束时调用，参数为 `{ event, node }`。 | <ApiType detail={"(dragProps: DragProps) => void"}>Type</ApiType> | - |
| onDragEnter | 拖拽进入节点时调用，参数为 `{ event, node, expandedKeys }`；拖拽使收起的节点展开时，`expandedKeys` 为展开后的键。 | <ApiType detail={"(dragEnterProps: DragEnterProps) => void"}>Type</ApiType> | - |
| onDragLeave | 拖拽离开节点时调用，参数为 `{ event, node }`。 | <ApiType detail={"(dragProps: DragProps) => void"}>Type</ApiType> | - |
| onDragOver | 拖拽经过节点时调用，参数为 `{ event, node }`。 | <ApiType detail={"(dragProps: DragProps) => void"}>Type</ApiType> | - |
| onDragStart | 开始拖拽节点时调用，参数为 `{ event, node }`。 | <ApiType detail={"(dragProps: DragProps) => void"}>Type</ApiType> | - |
| onDrop | 节点被放下时调用，参数为 `{ event, node, dragNode, dragNodesKeys, dropPosition, dropToGap }`。 | <ApiType detail={"(onDragProps: OnDragProps) => void"}>Type</ApiType> | - |
| onExpand | 展开或收起节点时调用，参数为展开的节点键数组和 `{ expanded, node }`。 | <ApiType detail={"(expandedKeys: string[], expandedOtherProps: ExpandedOtherProps) => void"}>Type</ApiType> | - |
| onLoad | 节点加载完毕时调用，参数为已加载节点的键集合和该节点数据。 | <ApiType detail={"(loadedKeys?: Set\u003cstring>, treeNode?: TreeNodeData) => void"}>Type</ApiType> | - |
| onSearch | 搜索框的值变化时调用，参数为搜索值和因搜索或 value、defaultValue 而展开的节点键数组；展开受控时可用它展开搜索结果。 | <ApiType detail={"(searchValue: string, filteredExpandedKeys: string[]) => void"}>Type</ApiType> | - |
| onSelect | 节点被选中或取消选中时调用，参数为节点键、是否选中和节点数据。 | <ApiType detail={"(selectedKey: string, selected: boolean, selectedNode: TreeNodeData) => void"}>Type</ApiType> | - |
| ref | 指向 `TreeHandle` 句柄，包含 `search` 与 `scrollTo` 方法，见下方 Methods。 | <ApiType detail={"React.Ref\u003cTreeHandle>"}>Type</ApiType> | - |
| renderDraggingNode | 拖拽时浏览器显示的拖拽图像。参数为节点的 HTML 元素与节点数据，返回一个 `HTMLElement`。 | <ApiType detail={"(nodeInstance: HTMLElement, node: TreeNodeData) => HTMLElement"}>Type</ApiType> | - |
| renderFullLabel | 接管整行的渲染，每行调用一次，参数为对象，含 `data`、`level`、`expandIcon`、`className`、`style`、`checkStatus`、`expandStatus`、`onClick`、`onCheck` 等。返回的元素会被 Tree 加上 `role="treeitem"`、无障碍状态属性与 `style`；返回文本、数组或 Fragment 时由 `li` 包裹。`className` 需由返回的元素自行绑定。 | <ApiType detail={"(renderFullLabelProps: RenderFullLabelProps) => ReactNode"}>Type</ApiType> | - |
| renderLabel | 自定义节点 label 的渲染，参数为 label、节点数据和当前搜索词。设置后不再自动高亮搜索词；搜索仍按 treeData 中的 label 文本匹配。 | <ApiType detail={"(label?: ReactNode, treeNode?: TreeNodeData, searchWord?: string) => ReactNode"}>Type</ApiType> | - |
| searchClassName | 搜索框容器的类名。 | `string` | - |
| searchPlaceholder | 搜索框的占位文字，未设置时取语言包中的文案。 | `string` | - |
| searchRender | 自定义搜索框，参数为对象，包含 `className`、`prefix`、`value`、`onChange`、`showClear` 和 `placeholder`，返回搜索框元素。`onChange` 的参数为输入的字符串。为 `false` 时隐藏内置搜索框。 | <ApiType detail={"((searchRenderProps: SearchRenderProps) => ReactNode) \u007c false"}>Type</ApiType> | - |
| searchStyle | 搜索框容器的样式。 | `React.CSSProperties` | - |
| showClear | 搜索框是否显示清除按钮。 | `boolean` | `true` |
| showFilteredOnly | 搜索时只展示过滤后的结果。 | `boolean` | `false` |
| showLine | 显示节点之间的连接线。 | `boolean` | `false` |
| style | 根元素的样式，类型为 `CSSProperties`。 | `React.CSSProperties` | - |
| treeData | 节点数据，key 在整棵树内唯一。只有传入新数组才会更新，原地修改数组或节点不会生效。 | `TreeNodeData[]` | - |
| treeDataSimpleJson | 简单 JSON 形式的节点数据，返回值为包含选中节点的 JSON 数据。 | `TreeDataSimpleJson` | - |
| treeNodeFilterProp | 搜索时用于匹配输入的节点属性。 | `string` | `"label"` |
| value | （受控）当前选中节点的 value（没有 value 时为 key）。传入后即为受控组件，配合 onChange 使用；多选时为数组。 | `Value` | - |
| virtualize | 列表虚拟化，用于大量节点：height、width、itemSize（必传）。开启后动画关闭。 | `Virtualize` | - |

**Methods**

| 方法     | 说明                                                                                               | 类型                                                                |
| -------- | -------------------------------------------------------------------------------------------------- | ------------------------------------------------------------------- |
| scrollTo | 在虚拟化的 Tree 中，把指定的已展开节点滚动到视图内；参数为 `{ key, align }`，align 默认 `center`。 | <ApiType detail={"(scrollData: ScrollData) => void"}>Type</ApiType> |
| search   | 手动触发搜索，效果同在搜索框中输入；需要同时设置 filterTreeNode。                                  | <ApiType detail={"(value: string) => void"}>Type</ApiType>          |
